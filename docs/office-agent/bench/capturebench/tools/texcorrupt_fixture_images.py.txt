"""texcorrupt_fixture_images.py - m53 F-SYN source images and the offline wrong-copy proofs (plan R12.1, R12.4).

Plain Python 3, standard library only, deterministic, no editor. Writes every F-SYN source image (PNG, or DDS
where a mip chain or a cube is needed) into --out, and texcorrupt_fixture_manifest.json beside them:
  - per image: file, SHA-256, size, the import settings make_texcorrupt_fixture.py applies, and its role;
  - the offline proofs: for every fault a row relies on (R12.4), the predicted per-texel error over the whole
    region the row reads, computed from the exact bytes written here, with its minimum. A fault is relied on
    only if that minimum is >= 32 levels (twice the in-leg 16-level threshold). Exit 1 if any is below.

Per-texel error = max over channels of |wrong - right| in the 8-bit value the host samples, which is the
statistic G-ID reads in the picture (per-pixel max-channel |d|).

Usage: python texcorrupt_fixture_images.py --out <dir>
Exit: 0 all relied-on proofs >= 32; 1 a proof below 32; 2 usage/IO error.
"""

def png_bytes(w, h, ch, data):
    """8-bit PNG, colour type 2 (RGB) or 6 (RGBA). data is row-major bytes, ch channels per texel."""
    ctype = {3: 2, 4: 6}[ch]
    stride = w * ch
    raw = bytearray()
    for y in range(h):
        raw.append(0)
        raw += data[y * stride:(y + 1) * stride]

    def chunk(tag, payload):
        return (struct.pack(">I", len(payload)) + tag + payload
                + struct.pack(">I", zlib.crc32(tag + payload) & 0xFFFFFFFF))

    return (b"\x89PNG\r\n\x1a\n" + chunk(b"IHDR", struct.pack(">IIBBBBB", w, h, 8, ctype, 0, 0, 0))
            + chunk(b"IDAT", zlib.compress(bytes(raw), 6)) + chunk(b"IEND", b""))


def dds_bgra8(w, h, levels, cube_faces=None):
    """Legacy DDS, 32-bit RGB masks (A8R8G8B8 -> B8G8R8A8), with a mip chain or six cube faces.
    levels: list of RGBA row-major bytearrays (mip 0 first). cube_faces: list of 6 such lists."""
    DDSD = 0x1 | 0x2 | 0x4 | 0x8 | 0x1000
    faces = cube_faces if cube_faces is not None else [levels]
    nmips = len(faces[0])
    if nmips > 1:
        DDSD |= 0x20000
    caps = 0x1000
    if nmips > 1 or cube_faces is not None:
        caps |= 0x8
    if nmips > 1:
        caps |= 0x400000
    caps2 = (0x200 | 0xFC00) if cube_faces is not None else 0
    pf = struct.pack("<8I", 32, 0x40 | 0x1, 0, 32, 0x00FF0000, 0x0000FF00, 0x000000FF, 0xFF000000)
    hdr = struct.pack("<7I", 124, DDSD, h, w, w * 4, 0, nmips) + b"\x00" * 44 + pf + struct.pack("<5I", caps, caps2, 0, 0, 0)
    body = bytearray()
    for face in faces:
        for rgba in face:
            px = len(rgba) // 4
            for i in range(px):
                r, g, b, a = rgba[4 * i:4 * i + 4]
                body += bytes((b, g, r, a))
    return b"DDS " + hdr + bytes(body)


# ---------------------------------------------------------------------------------------------- image helpers


def cells(w, h, palette, cell=CELL):
    """Row-major texels, a 'cell'-texel grid cycling through palette (tuples of equal length)."""
    ch = len(palette[0])
    per_row = []
    ncx = (w + cell - 1) // cell
    out = bytearray(w * h * ch)
    for y in range(h):
        cy = y // cell
        row = bytearray()
        for cx in range(ncx):
            col = palette[(cx + 3 * cy) % len(palette)]
            row += bytes(col) * min(cell, w - cx * cell)
        out[y * w * ch:(y + 1) * w * ch] = row
    return out, ch


def solid(w, h, col):
    return bytearray(bytes(col) * (w * h)), len(col)


def texels(data, ch):
    return [tuple(data[i:i + ch]) for i in range(0, len(data), ch)]


def enc_normal(x, y):
    z = math.sqrt(max(0.0, 1.0 - x * x - y * y))
    return tuple(int(math.floor((v * 0.5 + 0.5) * 255.0 + 0.5)) for v in (x, y, z))


def srgb_encode(c):
    c = min(1.0, max(0.0, c))
    return c * 12.92 if c <= 0.0031308 else 1.055 * (c ** (1.0 / 2.4)) - 0.055


def srgb_decode(c):
    return c / 12.92 if c <= 0.04045 else ((c + 0.055) / 1.055) ** 2.4


def q8(v):
    return int(math.floor(min(1.0, max(0.0, v)) * 255.0 + 0.5))

def uv16_colour(cx, cy):
    return (ORACLE_LEVELS[cx], ORACLE_LEVELS[cy], ORACLE_LEVELS[(cx + 2 * cy) % 4])


def uv64_colour(k):
    return (ORACLE_LEVELS[k % 4], ORACLE_LEVELS[(k // 4) % 4], ORACLE_LEVELS[k // 16])


def grid_image(w, h, ncell, colour_of):
    cell = w // ncell
    out = bytearray(w * h * 3)
    for y in range(h):
        for x in range(w):
            out[3 * (y * w + x):3 * (y * w + x) + 3] = bytes(colour_of(x // cell, y // cell))
    return out, 3

def tile_src_cell(u, v, ncell, n=DC_TILE_N):
    su, sv = (u * n) % 1.0, (v * n) % 1.0
    return min(int(su * ncell), ncell - 1), min(int(sv * ncell), ncell - 1)


def scramble_src_cell(cx, cy, a_inv, b, k=DC_SCRAMBLE_K):
    n = k * k
    j = cy * k + cx
    i = (a_inv * ((j - b) % n)) % n
    return i % k, i // k


def scramble_fwd_cell(cx, cy, a, b, k=DC_SCRAMBLE_K):
    n = k * k
    j = cy * k + cx
    i = (a * j + b) % n
    return i % k, i // k


def dc_proofs(imgs):
    """Offline, before the cook: (1) every oracle cell colour differs from every other by >= 60 in one channel at the
    texel; (2) the closed normal set's encoded values differ pairwise by >= 44 in R or G; (3) every negative pair of
    expectations (DC plan section 6) differs on >= 25 % of the kept sample positions."""
    by = {i["name"]: i for i in imgs}
    out = []

    def sep(cols):
        return min(max(abs(a - b) for a, b in zip(p, q)) for i, p in enumerate(cols) for q in cols[i + 1:])

    c16 = [uv16_colour(cx, cy) for cy in range(4) for cx in range(4)]
    c64 = [uv64_colour(k) for k in range(64)]
    out.append(dict(image="T_TC_OracleUV16", fault="cell_separation", min=sep(c16), floor=60, relied_on=sep(c16) >= 60 and len(set(c16)) == 16))
    out.append(dict(image="T_TC_OracleUV64", fault="cell_separation", min=sep(c64), floor=60, relied_on=sep(c64) >= 60 and len(set(c64)) == 64))
    encs = [enc_normal(*n) for n in ORACLE_N_SET]
    nsep = min(max(abs(p[0] - q[0]), abs(p[1] - q[1])) for i, p in enumerate(encs) for q in encs[i + 1:])
    out.append(dict(image="T_TC_OracleN", fault="closed_set_encoded_separation", min=nsep, floor=44, relied_on=nsep >= 44))
    closed = all((-x, -y) in ORACLE_N_SET and (x, -y) in ORACLE_N_SET for (x, y) in ORACLE_N_SET)
    present = sorted(set(ORACLE_N_CELLS)) == list(range(len(ORACLE_N_SET)))
    out.append(dict(image="T_TC_OracleN", fault="closed_under_invert_and_green_flip", min=int(closed and present), floor=1,
                    relied_on=closed and present))
    grid = [((x + 0.5) / 256.0, (y + 0.5) / 256.0) for y in range(256) for x in range(256)]

    def frac_differ(fa, fb):
        pts = [p for p in grid if fa(*p) is not None and fb(*p) is not None]
        return sum(1 for p in pts if fa(*p) != fb(*p)) / float(len(pts))

    ident16 = lambda u, v: (int(u * 4), int(v * 4))
    rows = [("tile x8 vs x4", lambda u, v: tile_src_cell(u, v, 4, 8), lambda u, v: tile_src_cell(u, v, 4, 4)),
            ("tile x8 vs x16", lambda u, v: tile_src_cell(u, v, 4, 8), lambda u, v: tile_src_cell(u, v, 4, 16)),
            ("tile x8 vs identity", lambda u, v: tile_src_cell(u, v, 4, 8), ident16)]
    for name, fa, fb in rows:
        f = frac_differ(fa, fb)
        out.append(dict(image="T_TC_OracleUV16", fault=name, min=round(f, 4), floor=0.25, relied_on=f >= 0.25))
    worst = dict(ordinal_pm1=1.0, forward=1.0, identity=1.0)
    import zlib
    for seed in (777, 4242, 0):
        for ordinal in range(0, 8):
            a, b, ai = scramble_pair_py(seed, ordinal, DC_SCRAMBLE_K)
            cells_all = [(cx, cy) for cy in range(8) for cx in range(8)]
            exp = {c: scramble_src_cell(c[0], c[1], ai, b) for c in cells_all}
            for other in (ordinal - 1, ordinal + 1):
                if other < 0:
                    continue
                a2, b2, ai2 = scramble_pair_py(seed, other, DC_SCRAMBLE_K)
                if (a2, b2) == (a, b):
                    continue
                f = sum(1 for c in cells_all if exp[c] != scramble_src_cell(c[0], c[1], ai2, b2)) / 64.0
                worst["ordinal_pm1"] = min(worst["ordinal_pm1"], f)
            if a != 1:
                f = sum(1 for c in cells_all if exp[c] != scramble_fwd_cell(c[0], c[1], a, b)) / 64.0
                worst["forward"] = min(worst["forward"], f)
            f = sum(1 for c in cells_all if exp[c] != c) / 64.0
            worst["identity"] = min(worst["identity"], f)
    for k, f in worst.items():
        out.append(dict(image="T_TC_OracleUV64", fault="scramble vs " + k, min=round(f, 4), floor=0.25, relied_on=f >= 0.25))
    for name, ta, tb in (("invert vs green_flip", lambda x, y: (-x, -y), lambda x, y: (x, -y)),
                         ("invert vs identity", lambda x, y: (-x, -y), lambda x, y: (x, y)),
                         ("green_flip vs identity", lambda x, y: (x, -y), lambda x, y: (x, y))):
        f = sum(1 for k in ORACLE_N_CELLS if ta(*ORACLE_N_SET[k]) != tb(*ORACLE_N_SET[k])) / float(len(ORACLE_N_CELLS))
        out.append(dict(image="T_TC_OracleN", fault=name, min=round(f, 4), floor=0.25, relied_on=f >= 0.25))
    for p in out:
        p.setdefault("row", "DC mode oracle")
        p.setdefault("texels_or_levels", 0)
        p.setdefault("region", "offline expectation, cell level")
        p.setdefault("max", p["min"])
    return out


SCRAMBLE_TAG = 0x53434D52


def scramble_pair_py(seed, ordinal, k):
    """Independent Python recomputation of TexCorruptPure::ScramblePair: CRC-32 (zlib) over 12 little-endian bytes
    {seed, ordinal, 0x53434D52}; a = the (H >> 16) % phi(N)-th unit mod N; b = H % N, bumped to 1 when (a, b) = (1, 0)."""
    import zlib
    n = k * k
    raw = struct.pack("<III", seed & 0xFFFFFFFF, ordinal & 0xFFFFFFFF, SCRAMBLE_TAG)
    h = zlib.crc32(raw) & 0xFFFFFFFF
    units = [a for a in range(1, n) if math.gcd(a, n) == 1]
    a = units[(h >> 16) % len(units)]
    b = h % n
    if a == 1 and b == 0:
        b = 1
    return a, b, pow(a, -1, n)

def per_texel_errors(img, fault):
    """Predicted per-texel max-channel error of the wrong copy against the identity copy, over the region
    the row reads. Returns (errors, region description)."""
    kind = img["kind"]
    t = texels(img["data"], img["ch"])
    if fault == "chanswap":
        if kind == "one_channel":
            return [p[0] for p in t], "mip 0, whole image: source decodes to (v,0,0,1); the swap moves v to B, the host reads R"
        if kind == "normal":
            errs = []
            for p in t:
                x, y = p[0] / 255.0 * 2 - 1, p[1] / 255.0 * 2 - 1
                r, _, b = enc_normal(x, y)
                errs.append(abs(r - b))
            return errs, "mip 0, whole image: encoded n.x (R) against encoded n.z (B) rebuilt from the stored x,y"
        return [abs(p[0] - p[2]) for p in t], "mip 0, whole image: |R - B|"
    if fault == "srgbtwice":
        return [max(abs(q8(srgb_encode(v / 255.0)) - v) for v in p[:3]) for p in t], \
            "mip 0, whole image: stored sRGB byte v against sRGB-encode(v) written through the sRGB target"
    if fault == "normal":
        errs = []
        for p in t:
            x, y = p[0] / 255.0 * 2 - 1, p[1] / 255.0 * 2 - 1
            z = math.sqrt(max(0.0, 1 - x * x - y * y))
            right = [q8(v * 0.5 + 0.5) for v in (x, y, z)]
            wrong = [q8(v) for v in (x, y, z)]
            errs.append(max(abs(a - b) for a, b in zip(right, wrong)))
        return errs, "mip 0, whole image: n written raw (UNORM clamps) against n*0.5+0.5"
    if fault == "alpha":
        return [p[3] for p in t], "mip 0, whole image: alpha channel, A = 0 (Opacity 1) against A = a"
    if fault == "texelshift":
        w, h = img["w"], img["h"]
        errs = []
        for y in range(h):
            for x in range(w):
                a = t[y * w + x]
                b = t[y * w + (x + 1) % w]
                errs.append(max(abs(i - j) for i, j in zip(a[:3], b[:3])))
        return errs, "mip 0 (magnification), whole image: texel (x,y) against (x+1,y), U-only shift (082-05 A)"
    if fault == "noclear":
        if kind == "redraw_opaque":
            return [max(min(v, 255 - v) for v in p[:3]) for p in t], \
                "whole image, linear target, A = 1: a second composite gives clamp(2v); error min(v, 255-v)"
        errs = []
        for p in t:
            a = p[3] / 255.0
            rgb = [abs(q8(srgb_encode(min(1.0, srgb_decode(v / 255.0) * (1.0 + a)))) - v) for v in p[:3]]
            errs.append(max(max(rgb), abs(q8(a * a) - p[3])))
        return errs, "whole image, sRGB target: RGB = L + L*a re-encoded, A = a^2"
    raise ValueError(fault)


def chain_a_errors(img, fault):
    cols = img["meta"]["mip_colours"]
    M = len(cols)
    rows = []
    for m in range(M):
        if fault == "mipshift":
            src = min(m + 1, M - 1)
            relied = m <= M - 2
        else:
            src = 0
            relied = m >= 1
        err = max(abs(a - b) for a, b in zip(cols[m], cols[src]))
        rows.append(dict(level=m, reads_source_mip=src, error=err, relied_on=relied))
    return rows


def run_proofs(imgs):
    out = []
    for img in imgs:
        for fault in img["faults"]:
            if img["kind"] == "chain_a":
                levels = chain_a_errors(img, fault)
                relied = [r["error"] for r in levels if r["relied_on"]]
                region = ("levels 0..M-2 (level M-1, 1x1, clamps to itself and reads 0; not relied on)" if fault == "mipshift"
                          else "levels 1..M-1 (level 0 is drawn from mip 0 and reads 0; not relied on)")
                out.append(dict(image=img["name"], row=img["role"], fault=fault, min=min(relied), max=max(relied),
                                texels_or_levels=len(relied), region=region, per_level=levels,
                                relied_on=min(relied) >= PROOF_FLOOR))
                continue
            errs, region = per_texel_errors(img, fault)
            out.append(dict(image=img["name"], row=img["role"], fault=fault, min=min(errs), max=max(errs),
                            texels_or_levels=len(errs), region=region, relied_on=min(errs) >= PROOF_FLOOR))
    return out

def main(argv):
    if "--out" not in argv:
        print("usage: texcorrupt_fixture_images.py --out <dir>")
        return 2
    out_dir = argv[argv.index("--out") + 1]
    os.makedirs(out_dir, exist_ok=True)
    imgs = build_set()
    entries = []
    for img in imgs:
        if img["fmt"] == "dds":
            if img["cube"] is not None:
                payload = dds_bgra8(img["w"], img["h"], None, cube_faces=img["cube"])
            else:
                payload = dds_bgra8(img["w"], img["h"], img["levels"])
        else:
            payload = png_bytes(img["w"], img["h"], img["ch"], img["data"])
        fn = img["name"] + "." + img["fmt"]
        with open(os.path.join(out_dir, fn), "wb") as fh:
            fh.write(payload)
        entries.append(dict(name=img["name"], file=fn, sha256=hashlib.sha256(payload).hexdigest(), bytes=len(payload),
                            width=img["w"], height=img["h"], channels=img["ch"], format=img["fmt"],
                            mips_authored=(len(img["levels"]) if img["levels"] else (1 if img["fmt"] == "dds" else None)),
                            cube=img["cube"] is not None, role=img["role"], settings=img["settings"],
                            faults_assigned=img["faults"], meta=img["meta"], dc=bool(img["meta"].get("dc"))))
    proofs = run_proofs(imgs) + dc_proofs(imgs)
    kat = [(777, 0, 29, 13, 53), (777, 1, 9, 19, 57), (4242, 0, 47, 21, 15), (4242, 7, 37, 12, 45), (0, 0, 11, 29, 35),
           (0xFFFFFFFF, 3, 1, 46, 1), (123456789, 42, 1, 46, 1)]
    kat_ok = all(scramble_pair_py(s, o, 8) == (a, b, ai) for s, o, a, b, ai in kat)
    proofs.append(dict(image="T_TC_OracleUV64", row="DC scramble pair recomputation", fault="scramble_pair_kat",
                       min=int(kat_ok), max=int(kat_ok), texels_or_levels=len(kat), relied_on=kat_ok,
                       region="the 7 C++ KAT vectors of texcorrupt_pure_test.cpp [17]"))
    doc = dict(tool="texcorrupt_fixture_images.py", proof_floor=PROOF_FLOOR,
               statistic="per-texel max-channel |wrong - right|, 8-bit, over the region the row reads",
               images=entries, proofs=proofs,
               all_relied_on=all(p["relied_on"] for p in proofs))
    with open(os.path.join(out_dir, "texcorrupt_fixture_manifest.json"), "w", encoding="utf-8", newline="\n") as fh:
        json.dump(doc, fh, indent=1)
    print("IMAGES %d written to %s" % (len(entries), out_dir))
    for p in proofs:
        print("PROOF %-20s %-11s min %4d max %4d over %6d %s  %s" % (
            p["image"], p["fault"], p["min"], p["max"], p["texels_or_levels"],
            "OK" if p["relied_on"] else "BELOW-%d" % PROOF_FLOOR, p["region"]))
    bad = [p for p in proofs if not p["relied_on"]]
    print("PROOFS %d, below %d: %d -> %s" % (len(proofs), PROOF_FLOOR, len(bad), "ALL RELIED-ON" if not bad else "STOP"))
    return 0 if not bad else 1
