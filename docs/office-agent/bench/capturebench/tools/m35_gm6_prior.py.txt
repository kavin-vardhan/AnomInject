import json
import statistics
import sys
from pathlib import Path

MEGAPIXELS = 1280 * 720 / 1_000_000.0


def leg_metrics(session_dir):
    d = Path(session_dir)
    tw = []
    with open(d / "labels.jsonl", encoding="utf-8") as f:
        for line in f:
            row = json.loads(line)
            tw.append(row["t_wall"])
    tw.sort()
    n = len(tw)
    span = tw[-1] - tw[0]
    per_frame_ms = span / (n - 1) * 1000.0
    deltas = [(tw[i + 1] - tw[i]) * 1000.0 for i in range(n - 1)]
    med_ms = statistics.median(deltas)
    rs = json.loads((d / "run_summary.json").read_text(encoding="utf-8"))
    return {
        "frames": n,
        "per_frame_ms": per_frame_ms,
        "median_delta_ms": med_ms,
        "per_mp_ms": per_frame_ms / MEGAPIXELS,
        "sustained_fps": rs.get("sustained_wall_fps"),
        "speed_ratio": rs.get("speed_ratio"),
    }


def main():
    legs = []
    for spec in sys.argv[1:]:
        name, path = spec.split("=", 1)
        m = leg_metrics(path)
        legs.append((name, m))
        print(f"{name}: frames={m['frames']} perFrame={m['per_frame_ms']:.4f} ms "
              f"medianDelta={m['median_delta_ms']:.4f} ms perMP={m['per_mp_ms']:.4f} ms/MP "
              f"sustained={m['sustained_fps']} ratio={m['speed_ratio']}")

    a = [m for n, m in legs if n.startswith("A")]
    b = [m for n, m in legs if n.startswith("B")]
    if len(a) == 2 and len(b) == 2:
        a_mean = sum(x["per_frame_ms"] for x in a) / 2
        b_mean = sum(x["per_frame_ms"] for x in b) / 2
        a_spread = abs(a[0]["per_frame_ms"] - a[1]["per_frame_ms"])
        b_spread = abs(b[0]["per_frame_ms"] - b[1]["per_frame_ms"])
        diff = b_mean - a_mean
        print(f"A mean={a_mean:.4f} ms (spread {a_spread:.4f})  B mean={b_mean:.4f} ms (spread {b_spread:.4f})")
        print(f"B-A difference={diff:.4f} ms/frame ({diff / MEGAPIXELS:.4f} ms/MP)  "
              f"within-build spread max={max(a_spread, b_spread):.4f} ms")
        if abs(diff) <= max(a_spread, b_spread):
            print("READING: the A/B difference is NOT larger than the within-build spread -> "
                  "BELOW THE RESOLUTION OF THIS INSTRUMENT (never 'no cost').")
        else:
            print("READING: the A/B difference exceeds the within-build spread. NUMBERS ONLY - "
                  "no threshold, no gate.")
        concorde_mp = 3200 * 2000 / 1_000_000.0
        print(f"EXTRAPOLATION (labelled as such, Concorde 3200x2000 = {concorde_mp:.1f} MP): "
              f"{diff / MEGAPIXELS * concorde_mp:.4f} ms/frame IF the cost scaled linearly with pixels "
              f"- an assumption, not a measurement.")


if __name__ == "__main__":
    sys.exit(main())
