import sys

import unreal

LEVEL_PATH = "/Game/CaptureBenchGate/CB_LodCalib"

FROZEN_LEVEL_PATHS = {
    "/Game/CaptureBenchGate/CB_GateLevel",
}
OVERRIDE_FLAG = "--allow-overwrite-frozen"
ALLOW_OVERWRITE_FROZEN = False

if LEVEL_PATH in FROZEN_LEVEL_PATHS and not (
        ALLOW_OVERWRITE_FROZEN or OVERRIDE_FLAG in sys.argv):
    raise RuntimeError(
        "CB_LOD_CALIB_REFUSED: refusing to author over the FROZEN gate level '%s'." % LEVEL_PATH)

ROCK_PATH = "/Game/Hidden_shrine/Meshes/Rock/SM_rock"

PLAYER_START = unreal.Vector(0, 0, 300)

RUNGS = [
    ("CB_LodRung_A", unreal.Vector(1000, 0, 300)),
    ("CB_LodRung_B", unreal.Vector(1900, 1350, 300)),
    ("CB_LodRung_C", unreal.Vector(3000, -2300, 300)),
    ("CB_LodRung_D", unreal.Vector(4600, 3400, 300)),
]

les = unreal.get_editor_subsystem(unreal.LevelEditorSubsystem)
eas = unreal.get_editor_subsystem(unreal.EditorActorSubsystem)

if unreal.EditorAssetLibrary.does_asset_exist(LEVEL_PATH):
    unreal.EditorAssetLibrary.delete_asset(LEVEL_PATH)

les.new_level(LEVEL_PATH)

rock = unreal.load_asset(ROCK_PATH)
if rock is None:
    raise RuntimeError("CB_LOD_CALIB_FAILED: rock mesh not found at %s" % ROCK_PATH)
if rock.get_num_lods() < 2:
    raise RuntimeError(
        "CB_LOD_CALIB_FAILED: %s has %d LOD(s); a LOD calibration level needs a multi-LOD mesh."
        % (ROCK_PATH, rock.get_num_lods()))

cube = unreal.load_asset("/Engine/BasicShapes/Cube")
grid_mat = unreal.load_asset("/Engine/EngineMaterials/WorldGridMaterial")

floor = eas.spawn_actor_from_class(unreal.StaticMeshActor, unreal.Vector(2000, 0, -200),
                                   unreal.Rotator(0, 0, 0))
floor.set_actor_scale3d(unreal.Vector(120, 120, 1))
floor.static_mesh_component.set_static_mesh(cube)
if grid_mat:
    floor.static_mesh_component.set_material(0, grid_mat)
floor.static_mesh_component.set_mobility(unreal.ComponentMobility.STATIC)
floor.set_actor_label("CB_LodFloor")

rungs = []
for label, loc in RUNGS:
    a = eas.spawn_actor_from_class(unreal.StaticMeshActor, loc, unreal.Rotator(0, 0, 0))
    comp = a.static_mesh_component
    comp.set_static_mesh(rock)
    comp.set_mobility(unreal.ComponentMobility.STATIC)
    a.set_actor_label(label)
    rungs.append(a)
    dx = loc.x - PLAYER_START.x
    dy = loc.y - PLAYER_START.y
    dz = loc.z - PLAYER_START.z
    dist = (dx * dx + dy * dy + dz * dz) ** 0.5
    unreal.log("[lod-calib] %s at %s distance=%.1f name=%s"
               % (label, loc, dist, a.get_name()))

sun = eas.spawn_actor_from_class(unreal.DirectionalLight, unreal.Vector(0, 0, 900),
                                 unreal.Rotator(-45, 35, 0))
sun.set_actor_label("CB_LodSun")
sun.root_component.set_mobility(unreal.ComponentMobility.MOVABLE)
sun.light_component.set_editor_property("intensity", 1.8)
sun.light_component.set_editor_property("cast_shadows", True)
sun.light_component.set_editor_property("dynamic_shadow_distance_movable_light", 8000.0)
try:
    sun.light_component.set_editor_property("atmosphere_sun_light", True)
except Exception as exc:
    unreal.log_warning("[lod-calib] atmosphere_sun_light not settable: %s" % exc)

sky = eas.spawn_actor_from_class(unreal.SkyLight, unreal.Vector(0, 0, 900),
                                 unreal.Rotator(0, 0, 0))
sky.set_actor_label("CB_LodSkyLight")
sky.root_component.set_mobility(unreal.ComponentMobility.MOVABLE)
sky.light_component.set_editor_property("intensity", 1.0)

atmo = eas.spawn_actor_from_class(unreal.SkyAtmosphere, unreal.Vector(0, 0, 0),
                                  unreal.Rotator(0, 0, 0))
atmo.set_actor_label("CB_LodSkyAtmosphere")

wall = eas.spawn_actor_from_class(unreal.StaticMeshActor, unreal.Vector(150, 0, 300),
                                  unreal.Rotator(0, 0, 0))
wall.set_actor_scale3d(unreal.Vector(2.0, 2.0, 4.0))
wall.static_mesh_component.set_static_mesh(cube)
wall.static_mesh_component.set_mobility(unreal.ComponentMobility.STATIC)
wall.set_actor_label("CB_LodNearWall")

wall_light = eas.spawn_actor_from_class(unreal.PointLight, unreal.Vector(40, 320, 520),
                                        unreal.Rotator(0, 0, 0))
wall_light.set_actor_label("CB_LodWallLight")
wall_light.root_component.set_mobility(unreal.ComponentMobility.MOVABLE)
wall_light.light_component.set_editor_property("intensity", 6000.0)
wall_light.light_component.set_editor_property("attenuation_radius", 1500.0)
wall_light.light_component.set_editor_property("cast_shadows", True)

NUM_POINT_LIGHTS = 3
for i in range(NUM_POINT_LIGHTS):
    ang = (i * 90) * 3.14159265 / 180.0
    px = 1400.0 + 700.0 * unreal.MathLibrary.cos(ang)
    py = 700.0 * unreal.MathLibrary.sin(ang)
    pl = eas.spawn_actor_from_class(unreal.PointLight, unreal.Vector(px, py, 620),
                                    unreal.Rotator(0, 0, 0))
    pl.set_actor_label("CB_LodPointLight_%02d" % i)
    pl.root_component.set_mobility(unreal.ComponentMobility.MOVABLE)
    pl.light_component.set_editor_property("intensity", 2500.0)
    pl.light_component.set_editor_property("attenuation_radius", 2400.0)
    pl.light_component.set_editor_property("cast_shadows", True)

ps = eas.spawn_actor_from_class(unreal.PlayerStart, PLAYER_START, unreal.Rotator(0, 0, 0))
ps.set_actor_label("CB_LodPlayerStart")

les.save_current_level()

if not unreal.EditorAssetLibrary.does_asset_exist(LEVEL_PATH):
    raise RuntimeError("CB_LOD_CALIB_FAILED: asset does not exist after save: %s" % LEVEL_PATH)

actors = eas.get_all_level_actors()
cb = [a for a in actors if a.get_actor_label().startswith("CB_Lod")]
expected = 1 + 1 + 1 + len(RUNGS) + 3 + NUM_POINT_LIGHTS + 1
if len(cb) != expected:
    raise RuntimeError(
        "CB_LOD_CALIB_FAILED actor-count mismatch: authored %d, found %d (total %d)."
        % (expected, len(cb), len(actors)))

unreal.log("CB_LOD_CALIB_SAVED %s cb_actors=%d total=%d rungs=%d rock_lods=%d"
           % (LEVEL_PATH, len(cb), len(actors), len(RUNGS), rock.get_num_lods()))
for a in rungs:
    unreal.log("CB_LOD_CALIB_RUNG label=%s runtime_name=%s" % (a.get_actor_label(), a.get_name()))
