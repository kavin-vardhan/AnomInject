FCaptureBenchSceneViewExtension::FCaptureBenchSceneViewExtension(const FAutoRegister& AutoRegister,
	const TSharedPtr<FCaptureBenchCapturer, ESPMode::ThreadSafe>& InCapturer)
	: FSceneViewExtensionBase(AutoRegister)
	, Capturer(InCapturer)
{
}

void FCaptureBenchSceneViewExtension::SetupViewFamily(FSceneViewFamily& InViewFamily)
{
	if (CaptureBenchProbe::IsEnabled())
	{
		UE_LOG(LogCaptureBench, Display,
			TEXT("PROBE,setupviewfamily,gfc=%llu,gfn=%u,familyFrame=%u,views=%d"),
			(unsigned long long)GFrameCounter, GFrameNumber,
			InViewFamily.FrameNumber, InViewFamily.Views.Num());
	}
}

void FCaptureBenchSceneViewExtension::BeginRenderViewFamily(FSceneViewFamily& InViewFamily)
{
	const uint32 FamilyFrame = InViewFamily.FrameNumber;

	if (CaptureBenchProbe::IsEnabled())
	{
		int32 NumGameViews = 0;
		for (const FSceneView* V : InViewFamily.Views)
		{
			if (V && !V->bIsSceneCapture && !V->bIsReflectionCapture)
			{
				++NumGameViews;
			}
		}
		UE_LOG(LogCaptureBench, Display,
			TEXT("PROBE,beginrenderviewfamily,gfc=%llu,gfn=%u,familyFrame=%u,views=%d,gameViews=%d"),
			(unsigned long long)GFrameCounter, GFrameNumber,
			FamilyFrame, InViewFamily.Views.Num(), NumGameViews);
	}

	CaptureBenchProbe::PublishKey(FamilyFrame, GFrameCounter);
}

bool FCaptureBenchSceneViewExtension::IsActiveThisFrame_Internal(const FSceneViewExtensionContext& Context) const
{
	TSharedPtr<FCaptureBenchCapturer, ESPMode::ThreadSafe> Cap = Capturer.Pin();
	return Cap.IsValid() && Cap->WantsCapture() && Cap->GetMode() == ECaptureBenchMode::SVE;
}

void FCaptureBenchSceneViewExtension::SubscribeToPostProcessingPass(EPostProcessingPass Pass,
	FAfterPassCallbackDelegateArray& InOutPassCallbacks, bool bIsPassEnabled)
{
	if (Pass == EPostProcessingPass::VisualizeDepthOfField)
	{
		InOutPassCallbacks.Add(FAfterPassCallbackDelegate::CreateLambda(
			[this](FRDGBuilder& GraphBuilder, const FSceneView& View, const FPostProcessMaterialInputs& Inputs) -> FScreenPassTexture
			{
				return AfterPass_RenderThread(GraphBuilder, View, Inputs);
			}));
	}
}

FScreenPassTexture FCaptureBenchSceneViewExtension::AfterPass_RenderThread(FRDGBuilder& GraphBuilder,
	const FSceneView& View, const FPostProcessMaterialInputs& Inputs)
{
	const FScreenPassTexture SceneColor = Inputs.GetInput(EPostProcessMaterialInput::SceneColor);

	TSharedPtr<FCaptureBenchCapturer, ESPMode::ThreadSafe> Cap = Capturer.Pin();
	if (!Cap.IsValid() || !SceneColor.IsValid() || View.bIsSceneCapture || View.bIsReflectionCapture)
	{
		return SceneColor;
	}

	CaptureBenchProbe::NoteRenderPass();
	CaptureBenchProbe::StallRenderThread();

	if (CaptureBenchProbe::IsEnabled())
	{
		const uint32 FamilyFrame = View.Family ? View.Family->FrameNumber : 0;
		CaptureBenchProbe::FRingEntry Entry;
		const bool bHit = CaptureBenchProbe::LookupKey(FamilyFrame, Entry);

		const FVector2D Jitter = View.ViewMatrices.GetTemporalAAJitter();
		const FIntRect Rect = SceneColor.ViewRect;
		const double JitterPxX = 0.5 * Jitter.X * (double)FMath::Max(1, Rect.Width());
		const double JitterPxY = 0.5 * Jitter.Y * (double)FMath::Max(1, Rect.Height());
		const FMatrix VP = View.ViewMatrices.GetViewProjectionMatrix();

		UE_LOG(LogCaptureBench, Display,
			TEXT("PROBE,rtpass,familyFrame=%u,gfcRT=%llu,gfnRT=%u,hit=%d,keyedGfc=%llu,")
			TEXT("isGameView=%d,isSceneCap=%d,rect=%dx%d,jitterNDC=(%.6f,%.6f),jitterPx=(%.4f,%.4f),vpTrace=%.6f"),
			FamilyFrame,
			(unsigned long long)GFrameCounterRenderThread, GFrameNumberRenderThread,
			bHit ? 1 : 0, (unsigned long long)(bHit ? Entry.GameFrameCounter : 0),
			View.bIsGameView ? 1 : 0, View.bIsSceneCapture ? 1 : 0,
			Rect.Width(), Rect.Height(),
			Jitter.X, Jitter.Y, JitterPxX, JitterPxY,
			VP.M[0][0] + VP.M[1][1] + VP.M[2][2] + VP.M[3][3]);
	}

	const uint32 RTFrame = GFrameNumberRenderThread;
	if (!Cap->ShouldCaptureThisRTFrame_RenderThread(RTFrame))
	{
		return SceneColor;
	}

	const double T0 = FPlatformTime::Seconds();

	FRDGTextureRef Texture = SceneColor.Texture;
	const FIntRect Rect = SceneColor.ViewRect;

	FCaptureBenchInFlight Item;
	Item.FrameIndex = Cap->NextFrameIndex();
	Item.SubmitRTFrame = RTFrame;
	Item.Rect = Rect;
	Item.Format = Texture->Desc.Format;
	Item.Readback = MakeUnique<FRHIGPUTextureReadback>(TEXT("CaptureBenchSVEReadback"));

	AddEnqueueCopyPass(GraphBuilder, Item.Readback.Get(), Texture,
		FResolveRect(Rect.Min.X, Rect.Min.Y, Rect.Max.X, Rect.Max.Y));

	const double HookSeconds = FPlatformTime::Seconds() - T0;
	Cap->SubmitInFlight_RenderThread(MoveTemp(Item), HookSeconds);

	return SceneColor;
}
