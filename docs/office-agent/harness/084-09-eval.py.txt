def rolling_median(v,w=9):

    for i in range(len(v)):out[i]=np.median(v[max(0,i-h):i+h+1])

def gate_transition(L,T,vis,span,drop=None,sigma=None,runs=None,temporal_aa=None):
    Ls=set(L);Ts=set(T);sp=set(span);V=set(vis)&sp
    unl=sorted(V-Ls);lnv=sorted((Ls&sp)-V)
    unl_x=[si for si in unl if si not in Ts];lnv_x=[si for si in lnv if si not in Ts]
    out=dict(unlabelled_visible=unl,labelled_not_visible=lnv,unlabelled_visible_without_transition=unl_x,labelled_not_visible_outside_transition=lnv_x,
             strict_0_0=not unl and not lnv,transition_used=sorted(Ts&(set(unl)|set(lnv))),transition_unused=len(Ts-(set(unl)|set(lnv))))
    decay=[]
    for a,b in (runs or runs_of(Ls)):
        after=[si for si in unl if si>b and si in Ts and all((x in V) for x in range(b+1,si+1))]
        if not after or drop is None or sigma is None:continue
        if (b+1) not in drop or b not in drop:continue
        ok=drop[b+1]<drop[b]-DECAY_K*sigma
        decay.append(dict(run_end=b,drop_last_labelled=round(drop[b],4),drop_first_after=round(drop[b+1],4),need_below=round(drop[b]-DECAY_K*sigma,4),ok=bool(ok)))
    out['offset_decay_checks']=decay
    fails=[]
    if unl_x:fails.append('unlabelled pixel-visible frame(s) without transition=1')
    if lnv_x:fails.append('labelled frame(s) not pixel-visible outside transition frames')
    if any(not d['ok'] for d in decay):fails.append('transition-excused frames after a label end whose effect had not started to decay at the first unlabelled frame (an early end, not a smear)')
    if temporal_aa is False and Ts:fails.append('transition flag(s) on a leg without temporal AA (the build emits none there)')
    out['fail_reasons']=fails;out['pass']=not fails
    if Ls and V:
        out['start']=min(V)-min(Ls);out['end']=max(V)-max(Ls)
    return out
def window_084(s,vis,span):

        b=L3.brk(s,si)

        if si+1 in vis:fv=si;break
    if fv is not None and onset_c is None:
        b=L3.brk(s,fv-1)
        if b:onset_c=f'break adjacent to the onset ({b})'
    lv=None;offset_c=None;conf=None
    if fv is not None:
        cand=fv;clean=0;sb=None

                clean=0

def estimate_lag(T,NT,mask):
    si=sorted(set(T)&set(NT))
    if len(si)<300:return None
    a=np.array([T[x] for x in si]);b=np.array([NT[x] for x in si]);m=np.array([x not in mask for x in si]);best=None
    for lag in range(-LAG_MAX,LAG_MAX+1):
        j=np.arange(len(si))+lag;ok=(j>=0)&(j<len(si))
        idx=np.nonzero(ok)[0];idx=idx[m[idx]&m[j[idx]]]
        if len(idx)<200:continue
        x=a[idx];y=b[j[idx]]
        if x.std()<1e-9 or y.std()<1e-9:continue
        cc=float(np.corrcoef(x,y)[0,1])
        if best is None or cc>best[1]:best=(lag,cc)
    return best

def local_sigma(rho,center_lo,center_hi,excl,win=NOISE_WIN):
    si=[x for x in range(center_lo-win,center_hi+win+1) if x in rho and x not in excl]
    if len(si)<12:
        si=[x for x in range(center_lo-3*win,center_hi+3*win+1) if x in rho and x not in excl]
    if len(si)<12:return None,len(si)
    v=np.array([rho[x] for x in si]);res=(v-rolling_median(v))/np.median(v)
    return robust_sigma(res),len(si)
def null_excursion(NT,lo,hi,lag):
    if NT is None:return 0.0
    si=[x for x in range(lo-20,hi+21) if (x+lag) in NT]
    if len(si)<15:return 0.0
    v=np.array([NT[x+lag] for x in si]);tr=rolling_median(v,31)
    w=[i for i,x in enumerate(si) if lo<=x<=hi]
    return float(max(0.0,max((tr[i]-v[i])/tr[i] for i in w))) if w else 0.0
def m52_measure(s,e,next_first,basis,excl_all,has_record=False,label=None,T=None,legacy_ref=None,oracle_label=None,temporal_aa=None):
    Lw=sorted(label if label is not None else m52_label(e));Tset=set(T if T is not None else m52_transitions(s,e))
    if not Lw:return dict(status='NO-LABEL')
    Lo=sorted(oracle_label) if oracle_label else Lw

    hi_cap=s.rows[-1]['session_index']
    span_hi=min(L1+SPAN_AFTER,hi_cap,(next_first-PRE_W-1) if next_first is not None else hi_cap)
    span=[si for si in range(L0-PRE_W,span_hi+1) if si in s.by_si and si in rho]
    if len(span)<PRE_W+len(Lo)+SUFFIX+2:return dict(status='SHORT-SPAN',span=[L0-PRE_W,span_hi])
    prew=[si for si in range(L0-PRE_N,L0) if si in rho]

    lag=basis.get('lag') or 0;NT=basis.get('NT')

            f=min(1.0,max(0.0,(si-cpre)/(cpost-cpre)));ref[si]=pre+(post-pre)*f
        else:ref[si]=pre
    drop={si:1.0-rho[si]/ref[si] for si in span}
    depth=max(drop[si] for si in span if L0<=si<=L1+SPAN_AFTER//2) if any(L0<=si<=L1+SPAN_AFTER//2 for si in span) else max(drop.values())
    mid=(L0+L1)/2.0

    measurable=depth>=DEPTH_K*max(thr_on,thr_off)
    drift_c=None
    if drift is not None and abs(drift)>max(thr_off,0.25*max(depth,1e-9)):
        drift_c=f'the picture does not return to its pre-event level (post/pre {drift:+.4f} against the {basis["basis"]} reference, beyond max(thr_off, 25% of depth))'
    rh=sorted(si for si,v in e['rstate'].items() if v=='held') if e.get('rstate') else []
    per={}
    for name,frac in THRESH:

        Tex,pg,tg=m52_transition_gate(s,e,Lw,Tset,vis,span,drop,thr,s_off,temporal_aa)

        if pg['fail_reasons']:fails=fails+pg['fail_reasons'];verd='FAIL'
        per[name]=dict(verdict=verd,fail_reasons=fails,two_sided_084_04a=two['verdict'],two_sided=two,transition_gate=tg,partial_guard=pg,visible=sorted(vis))
    leg_thr=None
    if legacy_ref is not None:
        leg_thr=max(K_SIGMA*legacy_ref['sigma']/max(legacy_ref['mu'],1e-9),REL_FLOOR,legacy_ref.get('drop_rel') or 0.0)
    return dict(status='OK',basis=basis['basis'],lag=basis.get('lag'),corr=basis.get('corr'),roi=basis['roi'],label=[Lw[0],Lw[-1]],label_n=len(Lw),oracle_label=[L0,L1],temporal_aa=temporal_aa,
                transition=sorted(Tset),span=[span[0],span[-1]],pre=round(pre,5),post=None if post is None else round(post,5),post_drift=None if drift is None else round(drift,4),
                sigma_on=round(s_on,5),sigma_off=round(s_off,5),noise_n=[n_on,n_off],null_excursion_on=round(ne_on,4),null_excursion_off=round(ne_off,4),own_trend_resid=round(trend_on,4),
                thr_on=round(thr_on,4),thr_off=round(thr_off,4),depth=round(depth,4),measurable=bool(measurable),
                legacy_whole_leg_thr=None if leg_thr is None else round(leg_thr,4),legacy_measurable=None if leg_thr is None else bool(depth>=DEPTH_K*leg_thr),
                drop={str(si):round(drop[si],4) for si in span},per=per,release=per[RELEASE]['verdict'])

def diff_event(s,ev,all_evs,temporal_aa=None,oracle_L=None):
    L=list(ev['L']);res=dict(type=ev['type'],node=ev['node'],ord=ev['ord'],start_frame=ev['start_frame'],label_n=len(L),transition=sorted(ev['T']))
    if not L:res['status']='NO-LABELLED-FRAMES';return res
    Ls=set(L);Ts=set(ev['T']);mine=set(ev['entries'])|Ls|Ts
    other=set()
    for o in all_evs:
        if o is ev:continue
        other.update(o['L']);other.update(o['T']);other.update(o['entries'])
    other-=mine
    lo=min(L);hi=max(L)
    span_lo=lo-ls086.SPAN_PAD;span_hi=hi+ls086.SPAN_PAD+max([x-hi for x in Ts if x>hi] or [0])
    for si in range(lo-1,lo-ls086.SPAN_PAD-1,-1):
        if si in other:span_lo=si+1;break
    for si in range(hi+1,span_hi+1):
        if si in other:span_hi=si-1;break
    span_lo=max(span_lo,s.sis[0]);span_hi=min(span_hi,s.sis[-1])
    span=[si for si in range(span_lo,span_hi+1) if si in s.rows]
    res['span']=[span_lo,span_hi]
    clean=lambda si:si in s.rows and si not in Ls and si not in Ts and si not in other and not (s.rows[si].get('anomaly_present') and si not in mine)
    refc=[si for si in range(lo-ls086.REF_BACK,lo) if clean(si)]
    if len(refc)<3:refc=[si for si in range(lo-30,lo) if clean(si)][-8:]
    if REF_AMENDMENT:
        ra=reference_amendment(s,ev,other,lo,hi,Ts,clean)
        res['reference']=ra
        if ra['status']=='CONFOUNDED':res['status']='CONFOUNDED-REFERENCE';return res
        refc=ra['frames']
    if len(refc)<2:res['status']='NO-REFERENCE';return res
    if lo<ls086.SETTLE_SKIP or max(refc)<ls086.SETTLE_SKIP:res['status']='WARMUP';return res
    if ls086.moved(s,refc+span):res['status']='CAMERA-MOVED';return res
    mvs=collections.Counter(v for si,v in ev['mv'].items() if si in Ls and v)
    mv=mvs.most_common(1)[0][0] if mvs else None
    bb=[ev['bbox'][si] for si in L if ev['bbox'].get(si)]
    f0=s.frame(refc[0])
    if f0 is None:res['status']='NO-FRAMES';return res
    roi,src,_=ls086.roi_for(s,dict(L=L,type=ev['type']),mv,bb,f0.shape)
    if roi is None:res['status']='NO-ROI';return res
    ctx=ls086.Ctx(roi,src=='fullframe');res['roi_src']=src;res['roi_px']=int(roi.sum())
    vc={}
    def V(si):
        if si not in vc:
            f=s.frame(si);vc[si]=ctx.view(f) if f is not None else None
        return vc[si]
    rv=[V(si) for si in refc]
    if any(x is None for x in rv):res['status']='NO-FRAMES';return res
    ctx.build_out(rv)
    lab_runs=runs_of(Ls);series={};run_info=[];cens={};sig_by_si={}
    for i,(a,b) in enumerate(lab_runs):
        lo_w=(lab_runs[i-1][1]+a)//2+1 if i>0 else span_lo
        hi_w=(b+lab_runs[i+1][0])//2 if i+1<len(lab_runs) else span_hi
        if i==0:ron=refc
        else:
            ron=[si for si in range(lab_runs[i-1][1]+1,a) if clean(si)]
            if len(ron)<2:ron=refc
        nxt=lab_runs[i+1][0] if i+1<len(lab_runs) else span_hi+1
        roff=[si for si in range(b+1,nxt) if clean(si)][:6]
        on_views=[V(x) for x in ron];Ron=ls086.median_view(on_views);mu_on,thD_on,thP_on,_=ls086._noise(on_views,ctx)
        off_ok=len(roff)>=2
        if off_ok:
            off_views=[V(x) for x in roff];Roff=ls086.median_view(off_views);mu_off,thD_off,thP_off,_=ls086._noise(off_views,ctx)
            d_first=ls086.metrics(off_views[0],Roff,ctx)[0]
            settled=d_first<=thD_off
            drift=ls086.metrics(Roff,Ron,ctx)[0]
        else:
            Roff=Ron;mu_off,thD_off,thP_off=mu_on,thD_on,thP_on;settled=False;drift=None
        midr=(a+b)/2.0;ser={}
        for si in range(lo_w,hi_w+1):
            if si not in s.rows:continue
            v=V(si)
            if v is None:continue
            if si<=midr:D,P,_,_=ls086.metrics(v,Ron,ctx);ser[si]=(D,P,mu_on,thD_on,thP_on)
            else:D,P,_,_=ls086.metrics(v,Roff,ctx);ser[si]=(D,P,mu_off,thD_off,thP_off)
        dD_on=max(ser[si][0] for si in ser if si<=midr) if any(si<=midr for si in ser) else 0.0
        dD_off=max(ser[si][0] for si in ser if si>midr) if any(si>midr for si in ser) else dD_on
        for si,(D,P,mu,thD,thP) in ser.items():
            dd=dD_on if si<=midr else dD_off
            series[si]=dict(D=D,P=P,frac=(D-mu)/max(dd-mu,1e-6),strict=bool(D>thD or P>thP))
            sig_by_si[si]=(thD-mu)/ls086.K_SIG
        if not settled:cens[b]=('post reference not settled (its first frame still differs from the post median beyond noise)' if off_ok else 'no clean post-reference frames before the span ends')
        run_info.append(dict(a=a,b=b,lo=lo_w,hi=hi_w,ref_on=ron,ref_off=roff,thrD_on=round(thD_on,3),thrD_off=round(thD_off,3),depth_on=round(dD_on,3),depth_off=round(dD_off,3),
                             post_settled=settled,drift_D=None if drift is None else round(drift,3)))
    if not series:res['status']='NO-FRAMES';return res
    res['runs']=run_info
    res['levels']={int(k):round(float(v['frac']),4) for k,v in series.items()};res['strict_frames']=sorted(int(k) for k,v in series.items() if v['strict'])
    peak=max(series,key=lambda si:series[si]['D']);res['peak_si']=peak
    res['confounded']=not (min(L)-1<=peak<=max(L)+SUFFIX+max([x-max(L) for x in Ts if x>max(L)] or [0]))
    depthD=max(r['depth_on'] for r in run_info);res['depth_D']=round(depthD,3)
    depthP=max(series[si]['P'] for si in series)
    res['measurable']=bool(depthP>=ls086.MEAS_P or any(r['depth_on']>=ls086.MEAS_K*r['thrD_on'] for r in run_info))
    gap={}
    for a,b in lab_runs:
        w=s.brk(a-1)
        if w and not w.startswith('engine-frame jump'):cens.setdefault(('s',a),w)
        w=s.brk(b)
        if w and not w.startswith('engine-frame jump'):cens.setdefault(('e',b),w)
        if s.brk(a-1) and s.brk(a-1).startswith('engine-frame jump'):gap[('s',a)]=s.brk(a-1)
        if s.brk(b) and s.brk(b).startswith('engine-frame jump'):gap[('e',b)]=s.brk(b)
    res['censored']=[f'{k}: {v}' for k,v in cens.items()];res['gap_bounded']=[str(k) for k in gap]
    drop={si:series[si]['frac'] for si in series}
    sig=float(np.median(list(sig_by_si.values())))/max(depthD,1e-6) if sig_by_si else None
    per={}
    for name,t in THRESH:
        vis={si for si,v in series.items() if v['strict'] and v['frac']>=t}
        tg=gate_transition(L,Ts,vis,[si for si in span if si in series],drop,sig,lab_runs,temporal_aa)
        on_c=any(isinstance(k,tuple) and k[0]=='s' and k[1]==lab_runs[0][0] for k in cens)
        off_c=(lab_runs[-1][1] in cens) or any(isinstance(k,tuple) and k[0]=='e' and k[1]==lab_runs[-1][1] for k in cens)
        fails,amb=split_censored(tg,L,on_c,off_c);tg['ambiguous_on_censored_edges']=amb
        edges=[]
        for rinfo in run_info:
            a,b=rinfo['a'],rinfo['b'];vv=sorted(si for si in vis if rinfo['lo']<=si<=rinfo['hi'])
            edges.append(dict(a=a,b=b,start=(vv[0]-a) if vv else None,end=(vv[-1]-b) if vv else None,cs=('s',a) in cens,ce=(b in cens) or (('e',b) in cens),gs=('s',a) in gap,ge=('e',b) in gap))
        strays=[si for si in tg['unlabelled_visible_without_transition'] if not any(e['a']-3<=si<=e['b']+3 for e in edges)]
        cen_any=any(e['cs'] or e['ce'] for e in edges)
        if not res['measurable']:st='NOT-MEASURABLE'
        elif res['confounded']:st='CONFOUNDED'
        elif fails:st='FAIL'
        elif cen_any:st='CENSORED'
        else:st='PASS'
        per[name]=dict(verdict=st,fail_reasons=fails,transition_gate=tg,edges=edges,strays=strays,visible=sorted(vis))
    res['per']=per;res['status']='OK';res['release']=per[RELEASE]['verdict']
    mm=[];mx=[]
    if mv:
        for si in L:
            m=s.mask(si)
            if m is None or not (m==mv).any():mm.append(si)
        for si in span:
            if si in Ls:continue
            m=s.mask(si)
            if m is not None and (m==mv).any():mx.append(si)
    else:mm=list(L)
    res['mask_value']=mv;res['mask_missing']=mm;res['mask_extra']=mx
    return res

def mask_gate_v1(sd,log='',min_events=90,bbox_pad=16,outside_tol=0.005):
    s=sess(sd);evs=anno_events(s);sm=s.summary or {}
    unavailable={int(x) for x in UNAVAIL_RX.findall(log or '')}
    cov=dict(labelled=0,masked=0,missing=[],g295=[]);alias=[];spatial=[];px_mismatch=[]
    by_frame=collections.defaultdict(list)
    for e in evs:
        for si in e['L']:
            cov['labelled']+=1;mv=e['mv'].get(si);tp=e['tp'].get(si)
            if not mv:cov['missing'].append((e['ord'],si,'no mask_value'));continue
            if tp is None or tp<0:
                (cov['g295'] if si in unavailable else cov['missing']).append((e['ord'],si,f'target_pixels {tp}'));continue
            m=s.mask(si)
            if tp>0 and (m is None or not (m==mv).any()):cov['missing'].append((e['ord'],si,'png absent or without the value'));continue
            cov['masked']+=1
            if m is not None and tp>0:
                hit=(m==mv);n=int(hit.sum())
                if n!=tp:px_mismatch.append((e['ord'],si,tp,n))
                bb=e['bbox'].get(si)
                if bb and len(bb)==4 and bb[2]>0 and bb[3]>0:
                    H,W=m.shape;x0=max(0,int(bb[0])-bbox_pad);y0=max(0,int(bb[1])-bbox_pad);x1=min(W,int(bb[0]+bb[2])+bbox_pad);y1=min(H,int(bb[1]+bb[3])+bbox_pad)
                    inside=int(hit[y0:y1,x0:x1].sum());outn=n-inside
                    if n and outn/n>outside_tol:spatial.append(dict(ord=e['ord'],si=si,value=mv,px=n,outside=outn,bbox=bb))
        for si,mv in e['mv'].items():
            if mv:by_frame[si].append((e['ord'],mv))
    for si,lst in by_frame.items():
        vals=collections.Counter(v for _,v in lst)
        for v,k in vals.items():
            if k>1:alias.append(dict(si=si,value=v,events=[o for o,x in lst if x==v]))
    holders=collections.defaultdict(list)
    for e in evs:
        vs={v for si,v in e['mv'].items() if v}
        fr=[si for si,v in e['mv'].items() if v]
        for v in vs:holders[v].append((e['ord'],min(fr),max(fr)))
    overlap=[]
    for v,h in holders.items():
        h=sorted(h,key=lambda x:x[1])
        for (o1,a1,b1),(o2,a2,b2) in zip(h,h[1:]):
            if a2<=b1:overlap.append(dict(value=v,first=o1,second=o2,first_range=[a1,b1],second_range=[a2,b2]))
    rec_lines=len(RECYCLE_RX.findall(log or ''));viol=len(OWNER_VIOL_RX.findall(log or ''));vsum=[int(x) for x in OWNER_SUM_RX.findall(log or '')];viol=max([viol]+vsum)
    reused=sum(1 for v,h in holders.items() if len(h)>1)
    fails=[]
    if len(evs)<min_events:fails.append(f'{len(evs)} events < {min_events}')
    if cov['missing']:fails.append(f'{len(cov["missing"])} labelled frame(s) without a measured mask (G295-signature frames counted apart: {len(cov["g295"])})')
    if alias:fails.append(f'{len(alias)} frame(s) where two events carry one mask value')
    if overlap:fails.append(f'{len(overlap)} recycled value(s) whose holders overlap in time')
    if spatial:fails.append(f'{len(spatial)} labelled frame(s) whose mask pixels lie outside their own target box (aliasing onto another object)')

    if shared:fails.append(f'{len(shared)} frame(s) where one mask value is carried by two holders (TAG-OWNERS shared>0; first {shared[:5]})')
    g.update(tag_owner_lines=len(own),tag_owner_shared_frames=shared[:20],tag_owner_missing=missing[:20],peak_live=peak,
             prior_collision=sm.get('mask_prior_collision'),prior_collision_quarantined=sm.get('mask_prior_collision_quarantined'),
             retire_quarantined=sm.get('mask_tag_retire_quarantined'),invalid=invalid,fail_reasons=fails,
             verdict='INVALID' if invalid else ('PASS' if not fails else 'FAIL'))
    g['pass_']=g['verdict']=='PASS'
    return g
def _flag(r,key):
    if r.get(key):return True
    for a in (r.get('anomalies') or []):
        if a.get(key) or ('camera_clipping_unconfirmed' in reasons_of(a) and key=='camera_clipping.unconfirmed'):return True
    return False
def cc_gate(on_sd,null_sd,on_log,null_log,nullB_sd=None,nullB_log=None,frames=200):
    r=cc_pair(on_sd,null_sd,on_log,null_log,nullB_sd,nullB_log,frames)
    if not r.get('valid'):r['gate']='INVALID';return r
    on=sess(on_sd);unc={k:_flag(on.by_si[k],'camera_clipping.unconfirmed') for k in range(frames)}
    lab={k:bool(on.by_si[k].get('camera_clipping.slab')) for k in range(frames)}
    D={int(k):v for k,v in r['D'].items()};theta=r['theta'];clipped={k:D[k]>theta for k in range(frames)}
    fr,_=cc_schedule(on_log);des={k:fr[k]['designed'] for k in range(frames)}
    edges=[]
    for k0 in [k for k in range(1,frames) if des[k]!=des[k-1]]:
        lo=max(0,k0-6);hi=min(frames-1,k0+6)
        lf=next((k for k in range(lo+1,hi+1) if lab[k]!=lab[k-1]),None);of=next((k for k in range(lo+1,hi+1) if clipped[k]!=clipped[k-1]),None)
        edges.append(dict(designed=k0,to=des[k0],label_flip=lf,pixel_flip=of,off=None if lf is None or of is None else lf-of))
    dis=r['rules'][RELEASE]['label_vs_clipped_disagree']
    r.update(unconfirmed_frames=sum(unc.values()),unconfirmed_disagreeing=[k for k in dis if unc[k]],confirmed_disagreeing=[k for k in dis if not unc[k]],
             edges=edges,edge_offsets=[e['off'] for e in edges])
    r['gate']='UNJUDGEABLE' if r['unjudgeable'] else ('PASS' if not dis and all((e['off'] in (0,None)) for e in edges) else 'FAIL')
    return r
def cc_mw_gate(on_sd,null_sd,nullB_sd=None,frames=200):
    base=cc_mw_pair(on_sd,null_sd,nullB_sd,frames)
    on=sess(on_sd);nu=sess(null_sd);nb=sess(nullB_sd) if nullB_sd else None
    if nb is None:base['gate']='INVALID';base['gate_note']='no second null: the per-frame floor needs null-vs-null';return base
    n=base['frames'];per=[]
    for k in range(n):
        d=dcount(frame_arr(on,k),frame_arr(nu,k));nn=dcount(frame_arr(nb,k),frame_arr(nu,k))
        r=on.by_si.get(k) or {};lab=bool(r.get('camera_clipping.slab'));unc=_flag(r,'camera_clipping.unconfirmed')
        per.append(dict(k=k,D=d,N=nn,ratio=None if d is None or nn is None else round(d/max(nn,1),3),labelled=lab,unconfirmed=unc))
    over=[p['k'] for p in per if p['labelled'] and not p['unconfirmed'] and p['ratio'] is not None and p['ratio']<=2.0]
    over_unc=[p['k'] for p in per if p['labelled'] and p['unconfirmed'] and p['ratio'] is not None and p['ratio']<=2.0]
    under=[p['k'] for p in per if (not p['labelled']) and p['ratio'] is not None and p['ratio']>2.0]
    base.update(per_frame_floor='labelled confirmed frame must have D_k > 2 * N_k (ON-vs-null over null-vs-null, per frame)',
                labelled_confirmed=sum(1 for p in per if p['labelled'] and not p['unconfirmed']),labelled_unconfirmed=sum(1 for p in per if p['labelled'] and p['unconfirmed']),
                over_label_confirmed=over[:30],over_label_confirmed_n=len(over),over_label_unconfirmed=over_unc[:30],over_label_unconfirmed_n=len(over_unc),
                unlabelled_above_floor_G352=len(under),per_frame=per)
    base['gate']='PASS' if not over else 'FAIL'
    return base
FADE_LO=0.10
FADE_HI=0.90
FADE_MIN=2
FADE_PRE=3
FADE_POST_OFF=9
LOD_MIN_JUDGED=16
LOD_FADE_RULE='084-08b: at each pop edge of a judged event (onset window a-3..mid, offset window mid+1..b+9, clipped to the run window), a frame is INTERMEDIATE when its per-frame effect level (the E1 frac: 0 = the unforced reference, 1 = the run peak) is in [0.10, 0.90); an edge with >= 2 intermediate frames is a MULTI-FRAME FADE. For variant D it is a FINDING, reported and never excused (084-08 ruling 1); for variant N it is the control reading. Transition flags do not remove a frame from this count.'
def lod_edges(r):
    lv={int(k):float(v) for k,v in (r.get('levels') or {}).items()};out=[]
    for run in r.get('runs') or []:
        a,b=run['a'],run['b'];mid=(a+b)//2
        for kind,lo,hi in (('onset',a-FADE_PRE,mid),('offset',mid+1,b+FADE_POST_OFF)):
            lo=max(lo,run['lo']);hi=min(hi,run['hi'])
            seq=[(si,lv[si]) for si in range(lo,hi+1) if si in lv]
            inter=[si for si,v in seq if FADE_LO<=v<FADE_HI]
            out.append(dict(kind=kind,a=a,b=b,window=[lo,hi],levels=[[si,round(v,3)] for si,v in seq],intermediate=inter,
                            n_intermediate=len(inter),multi_frame_fade=len(inter)>=FADE_MIN))
    return out
def hist_of(v):
    return dict(sorted(collections.Counter(v).items()))
def lod_gate(events,variant=None,min_judged=LOD_MIN_JUDGED):
    ok=[r for r in events if r.get('status')=='OK']
    judged=[r for r in ok if ((r.get('per') or {}).get(RELEASE) or {}).get('verdict') in ('PASS','FAIL')]
    off=[]
    for r in judged:
        tg=r['per'][RELEASE]['transition_gate']
        off+=[dict(start_frame=r.get('start_frame'),si=si,kind='unlabelled-visible') for si in tg['unlabelled_visible_without_transition']]
        off+=[dict(start_frame=r.get('start_frame'),si=si,kind='labelled-not-visible') for si in tg['labelled_not_visible_outside_transition']]
    fails=[dict(start_frame=r.get('start_frame'),reasons=r['per'][RELEASE]['fail_reasons']) for r in judged if r['per'][RELEASE]['verdict']=='FAIL']
    edges=[dict(start_frame=r.get('start_frame'),**e) for r in judged for e in lod_edges(r)]
    fades=[e for e in edges if e['multi_frame_fade']]
    ih={k:hist_of([e['n_intermediate'] for e in edges if e['kind']==k]) for k in ('onset','offset')}
    st=collections.Counter(r.get('status') if r.get('status')!='OK' else 'OK:'+str(((r.get('per') or {}).get(RELEASE) or {}).get('verdict')) for r in events)
    verdict='INSUFFICIENT' if len(judged)<min_judged else ('FAIL' if (off or fails) else 'PASS')
    return dict(verdict=verdict,variant=variant,judged=len(judged),min_judged=min_judged,events=len(events),status=dict(st),
                off_frames_t50_outside_transition=len(off),off=off[:40],fail_events=fails[:20],
                start=hist_of([e['start'] for r in judged for e in r['per'][RELEASE]['edges'] if e['start'] is not None and not e['cs']]),
                end=hist_of([e['end'] for r in judged for e in r['per'][RELEASE]['edges'] if e['end'] is not None and not e['ce']]),
                edges=len(edges),intermediate_hist=ih,fade_edges=len(fades),fades=fades[:24],
                fade_finding=bool(fades) and variant=='D',fade_control=(bool(fades) if variant=='N' else None),fade_rule=LOD_FADE_RULE,
                rule='gate: >= %d judged events (t50 PASS or FAIL) AND 0 off frames at t50 outside transition frames; fewer judged = INSUFFICIENT, never PASS'%min_judged)
def lod_fade_compare(gD,gN):
    if not gD or not gN:return None
    fd=gD['fade_edges']/max(1,gD['edges']);fn=gN['fade_edges']/max(1,gN['edges'])
    return dict(D_fade_edges=gD['fade_edges'],D_edges=gD['edges'],N_fade_edges=gN['fade_edges'],N_edges=gN['edges'],
                D_rate=round(fd,3),N_rate=round(fn,3),D_exceeds_N=fd>fn,
                reading='a D fade with no N fade at the same condition points at the dithered material; fades in both point at the capture/TAA path, which is still reported')
