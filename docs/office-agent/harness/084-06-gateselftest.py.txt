def rng(a,b):return list(range(a,b+1))
def rule(L,kon,koff):L=sorted(L);return set(L[:kon])|set(range(L[-1]+1,L[-1]+koff+1))

SEGS=[('control_start',0,29,lambda k:0),('wall_far',30,39,lambda k:0),('wall_in',40,59,lambda k:1 if k>=50 else 0),('wall_near',60,69,lambda k:1),('wall_out',70,89,lambda k:1 if k<=79 else 0),
      ('wall_far_2',90,99,lambda k:0),('behind',100,119,lambda k:0),('beside',120,139,lambda k:0),('prop_noncolliding',140,159,lambda k:1),('pawn_mesh',160,179,lambda k:0),('control_end',180,199,lambda k:0)]
def seg_of(k):
    for n,a,b,f in SEGS:
        if a<=k<=b:return n,f(k)
def write_cc(d,kind,rngs,label_shift=0,proxy='predicted',same_as_null=False):
    if d.exists():shutil.rmtree(d)
    (d/'Actual_Frames').mkdir(parents=True);rows=[];H,W=24,40;base=np.full((H,W,3),100,np.int16);base[:,:W//2]=140
    lines=['IAI-CC-SCHEDULE ARMED version=cc_v1 frames=200 wall_face_x=-1400.0 eye_z=260.0 pawn=None pawn_mesh=0 view_owner=x local_offset=0 gfc=1']
    for k in range(200):
        n,des=seg_of(k);img=base+rngs.integers(-3,4,size=base.shape)
        if kind=='on' and not same_as_null:
            if des:img[4:14,5:25]+=60
            elif n in ('control_start','wall_far','wall_far_2','control_end'):img[20:22,0:6]+=20
        Image.fromarray(np.clip(img,0,255).astype(np.uint8),'RGB').save(d/'Actual_Frames'/f'frame_{k:05d}.png')
        lab=seg_of(k-label_shift)[1] if 0<=k-label_shift<200 else des
        r=dict(session_index=k,frame_index=5000+k,image=f'Actual_Frames/frame_{k:05d}.png',anomalies=[])
        if kind=='on':
            px=(1 if (n in ('behind','beside')) else (0 if n=='prop_noncolliding' else des)) if proxy=='predicted' else des
            r['camera_clipping.slab']=bool(lab);r['camera_clipping.sphere_proxy']=bool(px);r['camera_clipping.eye_inside_box']=0
            if lab:r['anomalies']=[dict(id='camera_clipping',target_name='StaticMeshActor_49',start_frame=4999)]
        rows.append(r)
        lines.append(f'IAI-CC-SCHEDULE FRAME k={k} seg={n} designed={des} pawn_mesh=0 want=(0,0,0,yaw 0) got=(0,0,0,yaw 0) err_cm=0.0100 err_deg=0.0000 exhausted=0 gfc={5000+k}')
    (d/'labels.jsonl').write_text('\n'.join(json.dumps(x) for x in rows)+'\n',encoding='utf-8')
    (d/'run_summary.json').write_text(json.dumps({'camera_clipping_label_rule':'view_slab_render_bounds_v1'}),encoding='utf-8')
    E._SC.pop(str(d.resolve()),None)
    return d,'\n'.join(lines)
def cc_gate():

    on,lon=write_cc(t/'on',"on",rg);na,lna=write_cc(t/'na',"null",rg);nb,lnb=write_cc(t/'nb',"null",rg)

    check('C2 sphere proxy wrong on behind/beside/prop: can-fail proven',r.get('gate2_canfail_proven'),r.get('proxy_can_fail'))
    check('C3 designed verdicts and the oracle agree on every frame',r.get('designed_vs_oracle_disagree')==[],r.get('designed_vs_oracle_disagree'))
    on2,lon2=write_cc(t/'on_shift',"on",np.random.default_rng(7),label_shift=1)
    r2=E.cc_pair(on2,na,lon2,lna,nb,lnb);check('C4 label one frame late at every edge (a stale camera, G41 shape): gate 1 FAIL',r2.get('valid') and not r2['gate1_label_equals_clipped'],r2.get('rules',{}).get('t50'))
    on3,lon3=write_cc(t/'on_proxyok',"on",np.random.default_rng(7),proxy='designed')
    r3=E.cc_pair(on3,na,lon3,lna,nb,lnb);check('C5 proxy equal to the oracle everywhere: can-fail NOT proven (the gate refuses to call it evidence)',r3.get('valid') and not r3['gate2_canfail_proven'],r3.get('proxy_can_fail'))
    on4,lon4=write_cc(t/'on_same',"on",np.random.default_rng(7),same_as_null=True)
    r4=E.cc_pair(on4,na,lon4,lna,nb,lnb);check('C6 ON identical to its null (the push did nothing): UNJUDGEABLE, not PASS',r4.get('valid') and r4['unjudgeable'] and not r4['gate1_label_equals_clipped'],{k:r4.get(k) for k in ('F','P','unjudgeable')})
    lbad=lon.replace('err_cm=0.0100','err_cm=0.9000',3)
