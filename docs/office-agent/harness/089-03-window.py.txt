LUMA_MIN_MEAN=20.0;LUMA_MIN_ROI_STD=2.0
def luma_check(sd,every=15):
    import numpy as np
    from PIL import Image
    s=E.sess(sd);bb=[]
    for r in s.rows:
        for a in r.get('anomalies') or []:
            b=a.get('bbox_px')
            if b and len(b)==4 and b[2]>8 and b[3]>8:bb.append(b)
    roi=[int(np.median([b[i] for b in bb])) for i in range(4)] if bb else None
    means=[];stds=[]
    for r in s.rows[::every]:
        a=np.asarray(Image.open(s.dir/r['image']).convert('L'),dtype=np.float64);means.append(float(a.mean()))
        if roi:x,y,w,h=roi;stds.append(float(a[y:y+h,x:x+w].std()))
    mm=float(np.median(means)) if means else 0.0;ms=float(np.median(stds)) if stds else 0.0
    return dict(pass_=bool(mm>=LUMA_MIN_MEAN and (roi is None or ms>=LUMA_MIN_ROI_STD)),frames_sampled=len(means),median_frame_mean=round(mm,2),min_frame_mean=round(min(means),2) if means else None,
                target_roi=roi,median_roi_std=round(ms,2),rule=f'G402 first premise check: median whole-frame mean luma >= {LUMA_MIN_MEAN} and median target-ROI luma std >= {LUMA_MIN_ROI_STD} (not black, target non-uniform)')

    out=dict(pass_=bool(lu['pass_'] and elig and frac>=0.8),luma=lu,eligible=len(elig),measurable=len(meas),fraction=round(frac,3),
             status=dict(collections.Counter(r.get('status') for r in ev)),rule='luma first (G402), then >= 80 % of non-warm-up, camera-still (status OK) events MEASURABLE')
    if c.STUB and DRY_FORCE_PREMISE and not out['pass_']:
        out=dict(out,pass_=True,forced=True,real_pass=False,note='DRY ONLY: premise FORCED to pass so the dry run launches and evaluates the gate legs on stand-in data; the real stand-in verdict is real_pass')

def postflight():
    problems=[]
    def chk(ok,msg):
        if not ok:problems.append(msg)
    ours,foreign=c.classify()
    chk(not [p for p in ours if p['name']!='UnrealTraceServer.exe'],'our engine/game processes present at close')
    st=c.staged_state();chk(st==c.M53,'staged set is not the m53 set: '+json.dumps(st))
    chk(c.git(c.RP,'rev-parse','HEAD')==c.M51_HEAD and c.git(c.RP,'branch','--show-current')=='m51','main checkout moved off m51 53bf725')
    fh=c.git(c.RP,'rev-parse',c.FIX_BRANCH);fo=c.git(c.RP,'rev-parse','refs/remotes/origin/'+c.FIX_BRANCH);chk(fh==fo,f'fix branch {fh[:7]} != origin {fo[:7]}')
    led=L.ledger();acc=[x for x in led if x.get('status')=='ACCEPTED' and x.get('kind')!='part0'];links=0
    for x in acc:
        a=c.BANK/x['label'];al=c.BANK/(c.BANK_PREFIX+x['leg'])
        for p in [p for p in a.rglob('*') if p.is_file()]:
            q=al/p.relative_to(a)
            if not q.exists() or os.stat(q).st_ino!=os.stat(p).st_ino:problems.append(f'alias not a hardlink: {q}');break
            links+=1
    left=[d.name for d in c.EXE_SIDE.iterdir() if d.is_dir() and d.name.startswith(c.BANK_PREFIX)] if c.EXE_SIDE.exists() else []
    chk(not left,f'leg copies left after banking: {left[:5]}')
    bk={k:c.sha(c.M53_BACKUP/k) for k in c.KEYS if (c.M53_BACKUP/k).exists()};chk(bk==c.M53,'m53 backup does not verify')
    sw=L.security_windows()
    out=dict(utc=c.now(),problems=problems,accepted_legs=len(acc),attempts=len([x for x in led if x.get('label')]),not_accepted=[x for x in led if x.get('status')!='ACCEPTED'],
             alias_hardlinks_verified=links,left=left,staged=st,fix_head=fh,foreign_present=[(p['pid'],p['name']) for p in foreign],security_windows_open=sw,verdict='PASS' if not problems else 'FAIL')
    L.jsave(c.RUN/'postflight.json',out);return out
