        log=L.leg_log(Lg['leg']);r=dict(group=Lg['group'],role=Lg['role'],session=str(sd))
        try:

                else:r.update(eval_m52_leg(Lg,sd,log,sess_of(Lg['null']) if Lg.get('null') else None))

            elif Lg['group'] in ('BCCMW','BCCLY'):
                if Lg['role']=='CC_ON':
                    na,nb=Lg['pair'];sa=sess_of(na);sb=sess_of(nb) if nb else None
                    r['cc_mw']=E.cc_mw_pair(sd,sa,sb,Lg['frames']) if sa else dict(verdict='NO-NULL')
                else:r['role_note']='null leg: read through its ON pair'
            elif Lg['group']=='MASK55':
                r['mask']=E.mask_gate(sd,log);r.update(eval_diff_leg(Lg,sd,log))
            else:r.update(eval_diff_leg(Lg,sd,log))
        except Exception:
            import traceback;r['error']=traceback.format_exc()[-1500:]

def smear_values(rb):
    on=[];off=[];off_cens=[];src=[]
    for leg in ('B0_FIX','B3_FIX','B7_FIX'):
        v=rb['legs'].get(leg)
        if not isinstance(v,dict):continue
        for r in v['rows']:
            if r.get('status')!='OK' or not r.get('measurable') or r.get('camera_moved'):continue
            t=r['per']['t50']['two_sided'];L1=r['label'][1]
            if t['start'] is not None and not any(x.startswith('onset') for x in t['censored_edges']):on.append(t['start'])
            if t['end'] is not None and not any(x.startswith('offset') for x in t['censored_edges']):off.append(t['end'])
            elif any(x.startswith('offset') for x in t['censored_edges']):
                aft=[si-L1 for si in r['per']['t50']['transition_gate']['unlabelled_visible'] if si>L1]
                if aft:off_cens.append(max(aft))
        src.append(leg)
    import numpy as np
    def pct(v,q):return None if not v else float(np.percentile(v,q,method='higher'))
    res=dict(sources=src,threshold='t50 (the release rule)',onset=dict(n=len(on),hist=E.hist(on),p95=pct(on,95),p100=pct(on,100)),
             offset=dict(n=len(off),hist=E.hist(off),p95=pct(off,95),p100=pct(off,100)),
             offset_censored_lower_bounds=dict(n=len(off_cens),hist=E.hist(off_cens),max=max(off_cens) if off_cens else None,
                                               note='offsets censored because the picture never returned to its pre-event level against the null ratio; the last unlabelled visible frame after the label is a lower bound of the tail, not certified'))
    kon=int(res['onset']['p95'])+1 if on else None
    koff=max(int(res['offset']['p95'])+1,int(max(off_cens)) if off_cens else 0) if off else None
    res['proposed']=dict(TransitionOnFrames=kon,TransitionOffFrames=koff,
                         rule='K_on = p95 of the judged onset smear + 1; K_off = max(p95 of the judged offset smear + 1, the largest censored lower bound)',
                         outliers_above_proposed=dict(onset=[x for x in on if kon is not None and x>kon],offset=[x for x in off+off_cens if koff is not None and x>koff]))
    return res
