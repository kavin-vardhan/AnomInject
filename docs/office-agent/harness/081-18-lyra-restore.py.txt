def sha(p):
 with Path(p).open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()
def git(*a):return subprocess.check_output(['git','-C',str(L),*a]).decode().strip()
def classify():
    s=subprocess.check_output(['powershell','-NoProfile','-Command',"@(Get-CimInstance Win32_Process | Where-Object { $_.Name -in 'UnrealEditor.exe','StackOBot.exe','LyraGame.exe','UnrealBuildTool.exe','ShaderCompileWorker.exe' } | Select-Object ProcessId,Name,ExecutablePath,CommandLine) | ConvertTo-Json -Compress"]).decode().strip()
    v=[] if not s or s=='[]' else json.loads(s) if s.startswith('[') else [json.loads(s)]
    ours=[];foreign=[]
    for x in v:
        t=((x.get('ExecutablePath') or '')+' '+(x.get('CommandLine') or '')).lower()
        (ours if ('uesource' in t or 'intrusiveanomalies' in t or 'lyrastartergame' in t or 'stackobot' in t or x['Name'] in ('StackOBot.exe','LyraGame.exe','UnrealBuildTool.exe')) else foreign).append(dict(pid=x['ProcessId'],name=x['Name'],path=x.get('ExecutablePath'),cmd=(x.get('CommandLine') or '')[:200]))
    return ours,foreign
