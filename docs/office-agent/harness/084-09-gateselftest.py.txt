def mk_entry(labelled=True,reasons=None,levels=None,transition=None):
    a=dict(id='stuck_low_mip',start_frame=SF,labelled=labelled)

def mk_lod_event(sf,a,b,levels=None,unl=(),lnv=(),verdict='PASS'):

    lv.update(levels or {})
    tg=dict(unlabelled_visible_without_transition=list(unl),labelled_not_visible_outside_transition=list(lnv))
    per={E.RELEASE:dict(verdict=verdict,fail_reasons=['x'] if verdict=='FAIL' else [],transition_gate=tg,edges=[dict(start=0,end=0,cs=False,ce=False)])}
