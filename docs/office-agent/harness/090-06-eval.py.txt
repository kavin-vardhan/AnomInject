def e1_adjusted(events,interrupted_sis=frozenset(),temporal_aa=None):
    """Per-fire E1 t50 verdicts with the declared F1 adjudication; a fire without transition frames is unchanged."""
    out=[];isis=set(interrupted_sis)
    for r in events:
        st=r.get('status');row=dict(ord=r.get('ord'),start_frame=r.get('start_frame'),status=st,label=[min(r['L']),max(r['L'])] if r.get('L') else None)
        if st!='OK':row.update(verdict=st);out.append(row);continue
        p=(r.get('per') or {}).get(E.RELEASE) or {};T=set(r.get('transition') or []);vis=set(p.get('visible') or [])
        raw=list(p.get('fail_reasons') or []);exc=bool(T) and T<=isis and temporal_aa is False
        adj=[f for f in raw if not (exc and f==F1_T_REASON)]
        vi=sorted(vis&T&isis)
        if vi:adj.append(f'effect_interrupted frame(s) pixel-visible at t50: {vi[:8]}')
        ed=p.get('edges') or [];start=ed[0].get('start') if ed else None;end=ed[-1].get('end') if ed else None
        cen=any(e.get('cs') or e.get('ce') for e in ed)
        if not r.get('measurable'):v='NOT-MEASURABLE'
        elif r.get('confounded'):v='CONFOUNDED'
        elif adj:v='FAIL'
        elif cen:v='CENSORED'
        else:v='PASS'
        tg=p.get('transition_gate') or {}
        row.update(verdict=v,raw_verdict=p.get('verdict'),raw_fails=raw,adjusted_fails=adj,excused_temporal_aa_flag=exc and F1_T_REASON in raw,
                   start=start,end=end,visible=sorted(vis),transition=sorted(T),measurable=r.get('measurable'),
                   labelled_not_visible=list(tg.get('labelled_not_visible_outside_transition') or []),unlabelled_visible=list(tg.get('unlabelled_visible_without_transition') or []))
        out.append(row)
    return out
def e1_summary(adj):
    judged=[x for x in adj if x.get('verdict') in ('PASS','FAIL','CENSORED')]
    fails=[x for x in adj if x.get('verdict')=='FAIL']
    S=collections.Counter(x['start'] for x in judged if x.get('start') is not None);EE=collections.Counter(x['end'] for x in judged if x.get('end') is not None and x['verdict']!='CENSORED')
    verdict='FAIL' if fails else ('NO-JUDGEABLE-EVENT' if not judged else ('PASS-WITH-CENSORED' if any(x['verdict']=='CENSORED' for x in judged) else 'PASS'))
    starts=[x.get('start') for x in judged];ends=[x.get('end') for x in judged if x['verdict']!='CENSORED']
    return dict(verdict=verdict,judged=len(judged),passed=sum(1 for x in judged if x['verdict']=='PASS'),fails=len(fails),censored=sum(1 for x in judged if x['verdict']=='CENSORED'),
                all_starts_zero=bool(judged) and all(s==0 for s in starts),all_ends_zero=bool(ends) and all(e_==0 for e_ in ends),
                start='{'+', '.join(f'{k:+d}:{v}' for k,v in sorted(S.items()))+'}',end='{'+', '.join(f'{k:+d}:{v}' for k,v in sorted(EE.items()))+'}',
                status=dict(collections.Counter(x.get('verdict') for x in adj)),fail_events=[{k:x.get(k) for k in ('ord','label','start','end','adjusted_fails')} for x in fails][:6])

def lever_lines(log,kind):
    out=[]
    for m in LEVER_RX[kind].finditer(log or ''):
        rest=m.group('rest');d=dict(line=m.group(0)[:300])
        for k in ('session_index','actor','id'):
            mm=re.search(k+r'=(\S+)',rest);d[k]=mm.group(1) if mm else None

        except ValueError:d['si']=None

    return out

def f1_pixels(sd,log,st,expect):
    anomaly='missing_texture';events,taa,src=e1_run(sd,log,{anomaly})
    adj=e1_adjusted(events,set(st.get('interrupted_all') or []),taa)
    W=st['window_sis'];si=st['si']
    tgt=[(r,a) for r,a in zip(events,adj) if r.get('L') and set(r['L'])&set(W)]
    out=dict(temporal_aa=taa,temporal_aa_source=src,leg=e1_summary(adj),adjudication=F1_ADJ_RULE)
    if not tgt:out.update(verdict='UNJUDGEABLE',why='E1 found no fire overlapping the lever window');return out,adj
    r,a=tgt[0];out['fire']={k:a.get(k) for k in ('ord','label','status','verdict','raw_verdict','raw_fails','adjusted_fails','excused_temporal_aa_flag','start','end','visible','transition',
                                                 'labelled_not_visible','unlabelled_visible')}
    if a.get('status')!='OK' or a.get('verdict') in ('NOT-MEASURABLE','CONFOUNDED'):
        out.update(verdict='UNJUDGEABLE',why=f'E1 reads the lever fire {a.get("verdict") or a.get("status")}: the pixels cannot judge it');return out,adj
    vis=set(a.get('visible') or []);labset=set(st.get('labelled_sis') or [])
    vl=sorted(vis&labset);out['labelled_window_edge']=(vl[-1]-max(labset)) if vl and labset else None
    out['e1_run_window_end']=a.get('end')
    if expect=='new':
        ok=a['verdict'] in ('PASS','CENSORED') and a.get('start')==0 and a.get('end')==0 and si not in vis and not (vis&set(st.get('interrupted_sis') or []))
        out.update(verdict='PASS' if ok else 'FAIL',si_visible=si in vis,rule=f'fire verdict PASS after the F1 adjudication, onset +0, offset +0 against si-1, si below t50, no interrupted frame visible')
    else:
        want=-(W[-1]-si+1);want_lnv=list(range(si,W[-1]+1))
        ok=(a.get('raw_verdict')=='FAIL' and LNV_REASON in (a.get('raw_fails') or []) and a.get('labelled_not_visible')==want_lnv
            and out['labelled_window_edge']==want and a.get('start')==0)
        out.update(verdict='CANFAIL' if ok else 'NOT-CANFAIL',want_edge=want,want_labelled_not_visible=[si,W[-1]],
                   rule=(f'E1 FAIL on the lever fire with "{LNV_REASON}" on exactly si {si}..{W[-1]}; the last pixel-visible LABELLED frame is si-1, so the '
                         f'offset edge inside the label window is {want}; onset +0. E1\'s own run-window end is reported, not judged: with the whole labelled '
                         'off-half effect-free, E1 scales its off-half by noise and can read post-window noise frames as visible (the G419 class)'))
    return out,adj
def f1_check(sd,log,L,expect):
    st=f1_structure(sd,log,L,expect);out=dict(kind='F1',expect=expect,structure=st)
    if st['invalid']:
        out.update(verdict='INVALID',why=st['invalid']);return out
    px,adj=f1_pixels(sd,log,st,expect);out['pixels']=px;out['label_rule']=label_rule(sd)
    struct_ok=all(v['pass_'] for v in st['checks'].values());others=[x for x in adj if x.get('verdict')=='FAIL' and not (x.get('label') and set(range(x['label'][0],x['label'][1]+1))&set(st['window_sis']))]
    out['other_fires']=dict(fails=[{k:x.get(k) for k in ('ord','label','adjusted_fails')} for x in others][:6],judged=px['leg']['judged'])
    if expect=='new':
        reasons_=[k for k,v in st['checks'].items() if not v['pass_']]
        if px['verdict']=='UNJUDGEABLE':v='UNJUDGEABLE-PIXEL' if struct_ok else 'FAIL'
        elif struct_ok and px['verdict']=='PASS' and not others and px['leg']['judged']>=8 and out['label_rule']['pass_']:v='PASS'
        else:v='FAIL'
        if px['verdict']=='FAIL':reasons_.append('pixels: '+px['rule'])
        if others:reasons_.append(f'{len(others)} other fire(s) FAIL E1')
        if px['leg']['judged']<8:reasons_.append(f'judged {px["leg"]["judged"]} < 8')
        if not out['label_rule']['pass_']:reasons_.append('verify_capture --label-rule: '+str(out['label_rule']['verdict']))
        out.update(verdict=v,fail_reasons=reasons_)
    else:
        reasons_=[k for k,v in st['checks'].items() if not v['pass_']]
        if px['verdict']!='CANFAIL':reasons_.append('pixels: '+px.get('rule',px.get('why','')))
        out.update(verdict='CANFAIL-PROVEN' if (struct_ok and px['verdict']=='CANFAIL') else 'CANFAIL-NOT-PROVEN',fail_reasons=reasons_,
                   note='the label-rule is structural and reads a consistent old-binary session as NO MISMATCH; only the pixels see this can-fail')
    return out
def nanite_check(sd,log,mode,allow_reason_lines=None):
    rs=jl(Path(sd)/'run_summary.json');rows=read_rows(sd);an=jl(Path(sd)/'annotation.json')
    un=UNMEAS_NANITE_RX.findall(log or '');ref=REFUSED_RX.findall(log or '');end=NANITE_END_RX.findall(log or '')
    rb=READBACK_RX.findall(log or '');cur=CURRENT_RX.findall(log or '');unav={int(x) for x in UNAVAIL_RX.findall(log or '')}
    refused_actors=sorted({x[0] for x in ref});sites=dict(collections.Counter(x[2] for x in ref))
    res=dict(kind='NANITE',mode=mode,refused_nanite=rs.get('refused_nanite'),nanite_target_policy=rs.get('nanite_target_policy'),
             unmeasurable_targets_admitted=rs.get('unmeasurable_targets_admitted'),census_unmeasurable_nanite=rs.get('census_unmeasurable_nanite'),
             reason_nanite_lines=len(un),reason_nanite_actors=sorted(set(un)),refused_lines=len(ref),refused_actors=refused_actors,refused_sites=sites,
             run_end=[dict(distinct=int(a),refusals=int(b),setting=s) for a,b,s in end],readback=rb,current=cur,checks={},notes=[])
    def chk(name,ok,detail=None):res['checks'][name]=dict(pass_=bool(ok),detail=detail)
    missing=[k for k in ('refused_nanite','nanite_target_policy','unmeasurable_targets_admitted') if k not in rs]
    if missing:
        res.update(verdict='INVALID',why=f'run_summary lacks {missing}: not a 090-05 artifact (a pre-090-05 build writes no Nanite policy keys)');return res
    lab_rows=[r for r in rows if any(a.get('labelled') is True for a in (r.get('anomalies') or []))]
    un_rows=[r for r in lab_rows if r.get('mask_state') not in ('present','empty')]

def evaluate_leg(L,sd,log,sessions,logs,kit_dir=None):
    """Dispatch one leg: sessions / logs map leg id -> accepted session dir / log text (for nulls and the Nanite pair)."""
    g=L['group'];t0=time.time()
    if sd is None:return dict(verdict='NO-SESSION')
    try:
        if g=='A_REAL':r=eval_real(L,sd,log)
        elif g=='B_M53F':r=eval_m53f(L,sd,log)
        elif g=='B_GMODE':r=eval_mode(L,sd,log,sessions.get(L.get('null')))
        elif g in ('C_F1','C_F1_OLD'):r=f1_check(sd,log,L,'new' if L['binary']=='NEW' else 'old')
        elif g=='D_NANITE':
            al=UNMEAS_NANITE_RX.findall(logs.get('NANITE_ALLOW1') or '') if L['leg']=='NANITE_SKIP_DEFAULT' else None
            r=nanite_check(sd,log,'skip' if L['leg']=='NANITE_SKIP_DEFAULT' else 'allow',allow_reason_lines=len(al) if al is not None else None)
            r['label_rule']=label_rule(sd)
            if not r['label_rule']['pass_'] and r.get('verdict') in ('PASS','PASS-WITH-G295'):r['verdict']='FAIL';r.setdefault('fail_reasons',[]).append('label-rule '+str(r['label_rule']['verdict']))
        elif g=='E_M52':
            r=eval_b5(L,sd,log,[sessions.get('B0'),sessions.get('B9')]) if L['role']=='NULL' else eval_m52(L,sd,log,sessions.get(L.get('null')))
        else:r=dict(verdict='NOT-EVALUATED')
    except BaseException as ex:
        import traceback;r=dict(verdict='ERROR',error=traceback.format_exc()[-1500:])
    if kit_dir:c.guard_write(Path(kit_dir)/'x');Path(kit_dir).mkdir(parents=True,exist_ok=True)
    if kit_dir and g in ('A_REAL','B_M53F','C_F1','C_F1_OLD'):
        r['kit_office']=kit_office(sd,Path(kit_dir)/f'{L["leg"]}.office.json')
    if kit_dir and g=='E_M52' and L['role']!='NULL' and sessions.get(L.get('null')):
        r['kit_gt']=kit_aligned(sd,sessions.get(L['null']),Path(kit_dir)/f'{L["leg"]}.aligned.json')
    r['seconds']=round(time.time()-t0,1);r['evaluator']=EVALUATOR;return r
