def rows_of(sd):
    return {json.loads(l)['session_index']: json.loads(l) for l in Path(sd, 'labels.jsonl').read_text(encoding='utf-8-sig').splitlines() if l.strip()}

def sharp(sd):
    lb = Path(sd) / 'labels.jsonl'
    n = sum(1 for l in lb.read_text(encoding='utf-8-sig').splitlines() if l.strip())
    key = hashlib.sha1((hashlib.sha1(lb.read_bytes()).hexdigest() + '|' + ','.join(map(str, ROI)) + '|' + str(n)).encode()).hexdigest()[:20]
    return {int(k): v for k, v in json.loads((C84 / f'sharp_{key}.json').read_text()).items()}

ROWS = {k: rows_of(v) for k, v in S.items() if k != 'M55'}

tN = {si: r['t'] for si, r in ROWS['B5'].items()}
ks = sorted(tN, key=lambda k: tN[k]); NT = np.array([tN[k] for k in ks]); NV = np.array([SH['B5'][k] for k in ks])

def null_at(tv):
    i = bisect.bisect_left(NT, tv)
    if i <= 0 or i >= len(NT) or NT[i] - NT[i - 1] > 4.5 / 30:
        return None
    return float(np.interp(tv, NT, NV))

def aligned(leg):
    T = SH[leg]; t = {si: r['t'] for si, r in ROWS[leg].items()}
    rho = {}
    for si in T:
        v = null_at(t[si])
        if v:
            rho[si] = T[si] / v
    return rho, t

def gt_event(rho, L, W):
    L0, L1 = min(L), max(L)
    pre = [rho[x] for x in range(L0 - 8, L0) if x in rho]
    post = list(range(L1 + 1, L1 + 31))
    if len(pre) < 6 or sum(x in rho for x in L) < len(L) - 1 or sum(x in rho for x in post) < 26:
        return None
    pm = float(np.median(pre)); r = np.array(pre) / pm - 1
    sig = 1.4826 * float(np.median(np.abs(r - np.median(r)))); thr = max(5 * sig, 0.02)
    dep = max(1 - rho[x] / pm for x in L if x in rho)
    tail = [(1 - rho[x] / pm) if x in rho else None for x in post]
    n = [None if v is None else v / dep for v in tail]
    below = min([k + 1 for k, v in enumerate(n) if v is not None and v < 0.5], default=None)
    half = [k + 1 for k, v in enumerate(n) if v is not None and v >= 0.5 and k + 1 > W]
    sl = max([k + 1 for k, v in enumerate(tail) if v is not None and v > thr], default=0)
    dr = float(np.median([rho[x] for x in range(L1 + 24, L1 + 31) if x in rho]) / pm - 1)
    return dict(dep=dep, below=below, half=half, strict_last=sl, drift=dr, n1=n[0])
