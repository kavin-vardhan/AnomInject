def rank(v):
    s=sorted(range(len(v)),key=lambda i:v[i]);r=[0.0]*len(v);i=0
    while i<len(s):
        j=i
        while j+1<len(s) and v[s[j+1]]==v[s[i]]:j+=1
        for k in range(i,j+1):r[s[k]]=(i+j)/2+1
        i=j+1
    return r
def spearman(x,y):
    if len(x)<3:return None
    rx,ry=rank(x),rank(y);mx,my=st.fmean(rx),st.fmean(ry)
    num=sum((a-mx)*(b-my) for a,b in zip(rx,ry));den=(sum((a-mx)**2 for a in rx)*sum((b-my)**2 for b in ry))**0.5
    return None if den==0 else num/den

def g(a,*path):
    for k in path:
        if a is None:return None
        a=a[k] if isinstance(a,dict) and k in a else (a[k] if isinstance(a,list) and isinstance(k,int) and k<len(a) else None)
    return a

        p50=w['overall']['p50'],spread_m2_m10=(max(d['p50'] for d in dec[1:])-min(d['p50'] for d in dec[1:])) if all(dec[1:]) else None,
        game_cores=g(a,'process','game_cpu_logical_cores'),busy=g(a,'process','system_busy_pct'),priv_growth=(g(a,'process','private_gb_at_last_arm') or 0)-(g(a,'process','private_gb','first') or 0),
        tail_s=a['wall']['last_arm_to_last_png_s'],backlog=a['wall']['backlog_frames_at_last_arm'],drops=a['files']['missing_png_count']+a['files']['png_zero_bytes']+sum(a['writer_counters']['frames_failed_encode_write'])+sum(a['writer_counters']['frames_unresolved_at_run_end']))

    if len(on)<1 or len(off)<1:return dict(on=on,off=off,verdict='INSUFFICIENT')
    overlap=not (max(on)<min(off) or max(off)<min(on))
    return dict(on=on,off=off,mean_on=st.fmean(on),mean_off=st.fmean(off),diff_on_minus_off=st.fmean(on)-st.fmean(off),
        within_on_range=max(on)-min(on),within_off_range=max(off)-min(off),verdict=('NOT DISTINGUISHABLE' if overlap else 'DISTINGUISHABLE') if len(on)==2 and len(off)==2 else 'PARTIAL (n<2 in a condition)')
