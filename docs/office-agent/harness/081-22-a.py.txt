def census_check(s,text):
    errs=[];peak=s.get('change_bytes_peak_census');hw=s.get('change_bytes_high_water')
    if not isinstance(peak,dict):return ['peak census absent'],None
    if peak.get('high_water')!=hw:errs.append('peak.high_water != change_bytes_high_water')
    if peak.get('bytes_held')!=peak.get('high_water'):errs.append('peak.bytes_held != peak.high_water')
    if peak.get('unaccounted_bytes')!=0:errs.append('unaccounted_bytes nonzero')
    if peak.get('colours_unowned')!=0:errs.append('colours_unowned nonzero')
    if peak.get('colours')!=sum(peak.get(k,0) for k in ('colours_pending','colours_previous','colours_phase_ref','colours_in_hand','colours_unowned')):errs.append('colour split does not sum')
    if peak.get('colour_bytes',0)+peak.get('mask_bytes',0)+peak.get('unaccounted_bytes',0)!=peak.get('bytes_held'):errs.append('bytes do not sum')
    lines=re.findall(r'HIGH-WATER-PEAK bytes=(\d+) max=(\d+) census=(\{.*\})',text)
    if len(lines)!=1:errs.append(f'HIGH-WATER-PEAK lines={len(lines)}')
    elif int(lines[0][0])!=hw or json.loads(lines[0][2])!=peak:errs.append('HIGH-WATER-PEAK line disagrees with summary')
    return errs,peak
