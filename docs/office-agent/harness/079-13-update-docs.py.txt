"""Apply bounded UTF-8 byte replacements, preserving unrelated document bytes."""

def replace_span(path, start, stop, text):
    raw = path.read_bytes()
    a, b = raw.index(start.encode()), raw.index(stop.encode(), raw.index(start.encode()))
    path.write_bytes(raw[:a] + text.replace("\n", "\r\n").encode("utf-8") + raw[b:])
