def gate_stat(a,b):
    nA,nB=len(a),len(b)
    if nA<2 or nB<2:return None
    mA,mB=st.fmean(a),st.fmean(b);df=nA+nB-2;ss=sum((x-mA)**2 for x in a)+sum((x-mB)**2 for x in b);sp=math.sqrt(ss/df);se=sp*math.sqrt(1/nA+1/nB);t=T95[df]
    return dict(nA=nA,nB=nB,meanA=round(mA,4),meanB=round(mB,4),delta=round(mB-mA,4),pooled_sd=round(sp,4),se=round(se,4),df=df,t95_one_sided=t,
                upper=round(mB-mA+t*se,4),lower=round(mB-mA-t*se,4),spread_AA=round(max(a)-min(a),4),spread_BB=round(max(b)-min(b),4))
def verdict_for(res,legs):
    A=[x for x in legs if x['cond']=='A'];B=[x for x in legs if x['cond']=='B'];why=[]
    if len(A)!=4 or len(B)!=4:return dict(resolution=res,verdict='UNRESOLVED',why=[f'incomplete: {len(A)} A and {len(B)} B counted legs, 4 and 4 required (two complete ABBA blocks)'])
    held=[]
    for grp in (A,B):
        med=st.median(x['arm_rate_fps'] for x in grp)
        for x in grp:
            if x['arm_rate_fps']<FPS_PRIOR_FRAC*med:held.append(dict(label=x['label'],arm_rate_fps=x['arm_rate_fps'],condition_median=med,stalls_in_window=x.get('stalls_in_window'),load=x.get('load_summary')))
    gA=[x['gt_ms'] for x in A];gB=[x['gt_ms'] for x in B];wA=[x['writer'] for x in A];wB=[x['writer'] for x in B]
    g=gate_stat(gA,gB);w=gate_stat(wA,wB)
    gt_pass=g['upper']<=GT_MS_GATE
    if not gt_pass:why.append(f'game thread: upper bound {g["upper"]} ms > +{GT_MS_GATE} ms (delta {g["delta"]}, se {g["se"]}, t {g["t95_one_sided"]})')
    drop=w['upper']<0;hw=w['t95_one_sided']*w['se'];res_ok=hw<=WRITER_RES_FRAC*w['meanA']
    if drop:why.append(f'writer throughput dropped beyond uncertainty: upper bound of the change {w["upper"]} PNG/s < 0 (delta {w["delta"]})')
    if not res_ok:why.append(f'writer throughput under-resolved: one-sided 95% half-width {round(hw,3)} PNG/s > {WRITER_RES_FRAC:.0%} of the A mean {w["meanA"]} (a drop that size could not be seen)')
    if held:v='HELD-FPS-PRIOR';why.append(f'{len(held)} counted leg(s) below {FPS_PRIOR_FRAC} x their condition median arm rate: the gate is not read until each is explained (081-05 carry-forward)')
    else:v='PASS' if (gt_pass and not drop and res_ok) else 'UNRESOLVED'
    return dict(resolution=res,verdict=v,why=why,game_thread_ms=g,writer_throughput_pngps=w,writer_half_width=round(hw,4),writer_resolution_limit=round(WRITER_RES_FRAC*w['meanA'],4),held=held,
                legs=[{k:x[k] for k in ('label','cond','block','gt_ms','writer','arm_rate_fps')} for x in legs])
def verdict_P(res,legs):
    A=[x for x in legs if x['cond']=='A'];B=[x for x in legs if x['cond']=='B'];why=[];base=dict(resolution=res,mode='paced block P (AMENDMENT 1, headline)',pacer_note=PACER_NOTE)
    if len(A)!=4 or len(B)!=4:return dict(base,verdict='UNRESOLVED',why=[f'incomplete: {len(A)} A and {len(B)} B counted legs, 4 and 4 required (two complete ABBA blocks)'])
    held=[]
    for grp in (A,B):
        med=st.median(x['arm_rate_fps'] for x in grp)
        for x in grp:
            if x['arm_rate_fps']<FPS_PRIOR_FRAC*med:held.append(dict(label=x['label'],arm_rate_fps=x['arm_rate_fps'],condition_median=med,stalls_in_window=x.get('stalls_in_window'),load=x.get('load_summary')))
    g=gate_stat([x['gt_ms'] for x in A],[x['gt_ms'] for x in B]);w=gate_stat([x['writer'] for x in A],[x['writer'] for x in B]);r=gate_stat([x['arm_rate_fps'] for x in A],[x['arm_rate_fps'] for x in B])
    gt_pass=g['upper']<=GT_MS_GATE
    if not gt_pass:why.append(f'game thread: upper bound {g["upper"]} ms > +{GT_MS_GATE} ms (delta {g["delta"]}, se {g["se"]}, t {g["t95_one_sided"]})')
    drop=w['upper']<0;hw=w['t95_one_sided']*w['se'];res_ok=hw<=WRITER_RES_FRAC*w['meanA']
    if drop:why.append(f'writer throughput dropped beyond uncertainty: upper bound of the change {w["upper"]} PNG/s < 0 (delta {w["delta"]})')
    if not res_ok:why.append(f'writer throughput under-resolved: one-sided 95% half-width {round(hw,3)} PNG/s > {WRITER_RES_FRAC:.0%} of the A mean {w["meanA"]} (a drop that size could not be seen)')
    pace_drop=r['upper']<0
    if pace_drop:why.append(f'pacing not held: B arm rate below A beyond uncertainty, upper bound of the change {r["upper"]} fps < 0 (delta {r["delta"]})')
    drops=[(x['label'],x.get('drops_total')) for x in legs if x.get('drops_total')!=0]
    if drops:why.append(f'drops: every counted leg must carry 0 drops (the 081-37 integrity set); not 0 or not read on {drops}')
    vac=[(x['label'],x.get('pairs_measured'),x.get('pairs_required')) for x in B if not x.get('yield_ok')]
    if vac:why.append(f'B vacuity: measured-pair yield below 0.9 of required, or not read, on {vac}')
    if held:v='HELD-FPS-PRIOR';why.append(f'{len(held)} counted leg(s) below {FPS_PRIOR_FRAC} x their condition median arm rate: the gate is not read until each is explained (081-05 carry-forward)')
    else:v='PASS' if (gt_pass and not drop and res_ok and not pace_drop and not drops and not vac) else 'UNRESOLVED'
    return dict(base,verdict=v,why=why,game_thread_ms=g,writer_throughput_pngps=w,writer_half_width=round(hw,4),writer_resolution_limit=round(WRITER_RES_FRAC*w['meanA'],4),
                arm_rate_fps=r,arm_rate_half_width=round(r['t95_one_sided']*r['se'],4),held=held,
                legs=[{k:x.get(k) for k in ('label','cond','block','gt_ms','writer','arm_rate_fps','drops_total','pairs_measured','pairs_required','yield_ok')} for x in legs])

def run_block(res,bi,names,invocation,t_inv0):
    legs=[leg_id(res,n) for n in names]
    have=[counted_valid(l) for l in legs]
    if all(have):return 'complete'
    prev=[x['label'] for h in have for x in h]
    if prev:void_rows(set(prev),f'VOID-RESUME: block {res}/{bi} was incomplete when invocation {invocation} began; its legs re-run together so drift cancels inside one contiguous ABBA block');c.log_event('block-voided',res=res,block=bi,labels=prev)
    for n in names:run_attempt(res,n,invocation,t_inv0)
    return 'ran'
def run_campaign(invocation):

        for r,bi,ns in todo:state['blocks'].append((r,bi,run_block(r,bi,ns,invocation,t0)))
    r,bi,ns=DIAG_BLOCK
    if not all(counted_valid(leg_id(r,n)) for n in ns):state['blocks'].append((r,bi,run_block(r,bi,ns,invocation,t0)))
    state['finished']=c.now();state['bench_s']=round(time.time()-t0,1);return state
def _block_legs(rows,blocks,res):
    legs=[]
    for r,bi,ns in blocks:
        if r!=res:continue
        for n in ns:
            v=[x for x in rows if x['leg']==leg_id(r,n)]
            if v:legs.append(dict(label=v[-1]['label'],cond=cond_of(n),block=bi,gt_ms=v[-1]['gt_ms'],writer=v[-1]['writer'],arm_rate_fps=v[-1]['arm_rate_fps'],stalls_in_window=v[-1].get('stalls_in_window'),load_summary=v[-1].get('load_summary'),
                                  drops_total=v[-1].get('drops_total'),pairs_measured=v[-1].get('pairs_measured'),pairs_required=v[-1].get('pairs_required'),yield_ok=v[-1].get('yield_ok')))
    return legs
def stress_readings(res,legs,halts):
    ys=[];reasons=collections.Counter()
    for x in legs:
        if x['cond']!='B':continue
        y=((jload(ROOT/'legs'/f'readings-{x["label"]}.json',{}) or {}).get('m55') or {}).get('yield_') or {}
        if y.get('required'):ys.append(round(y['measured']/y['required'],4));reasons.update(y.get('unmeasured_window_reasons') or {})
    v=verdict_for(res,legs);h=[dict(label=x.get('label'),status=x['status'],problems=x.get('problems')) for x in halts if x.get('res')==res]
    lab=f'pacing-off stress (081-38), measured-pair yield {min(ys):.2f}-{max(ys):.2f}' if ys else 'pacing-off stress (081-38), measured-pair yield not read'
    cause=None
    if h:cause='writer saturation -> m55 payload at the 256 MiB cap -> honest budget_exceeded refusals -> B vacuity LEG-FAILURE; not re-run and not reshaped'
    return dict(v,mode='pacing OFF (081-37 blocks, retired by AMENDMENT 1)',label=lab,ruling=RULING_STRESS,headline=False,b_yields=ys,b_unmeasured_reasons=dict(reasons),halting_rows=h,cause=cause)
