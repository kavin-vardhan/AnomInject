import importlib.util,json,os,sys
from pathlib import Path
W=sys.argv[1];assert W in ('c0','c1','c2','c3');os.environ['IAI_R23_WINDOW']=W
spec=importlib.util.spec_from_file_location('c23',r'D:\IntrusiveAnomalies\_reviews\081-23-common.py');c=importlib.util.module_from_spec(spec);spec.loader.exec_module(c)
R,E,E23=c.R,c.E,c.E23;E.mkdir(parents=True,exist_ok=True)
post22=json.loads((c.E22/'postflight.json').read_text());prep=json.loads((E23/'prep-boundary.json').read_text())
problems=[]
def chk(ok,msg):
    if not ok:problems.append(msg)
ours,foreign=c.classify()
chk(not [p for p in ours if p['name']!='UnrealTraceServer.exe'],'our engine/game processes present')
for name,r in {'AnomalyInjector':c.RP,'Lyra':c.L,'CaptureBench':c.C}.items():
    chk(not c.git(r,'diff','--name-only') and not c.git(r,'diff','--cached','--name-only'),f'{name} tracked modifications')
    st=c.state(r);old=post22['repos'][name]
    chk(st['head']==old['head'] and st['branch']==old['branch'],f'{name} head/branch moved {st["head"][:7]} {st["branch"]}')
chk(c.git(c.RP,'rev-parse',c.FEATURE)==prep['feature_head'],'feature head != prep boundary')
chk(sorted(p.name for p in (c.L/'Binaries/Win64').iterdir() if p.is_file())==post22['lyra_module_names'],'Lyra module set')
for p,h in post22['verified_binaries'].items():chk(c.sha(p)==h,f'binary changed {p}')
b3=c.binary_v3()
for key in ('built','staged','archive'):chk(c.sha(b3[key])==b3['sha256'],f'002805CF not {key}')
for r in b3['containers']:chk(c.sha(r['path'])==r['sha256'],f'container {r["path"]}')
chk(c.sha(c.PRE_M55).upper().startswith('0844220E'),'pre-m55 archive')
for p,h in prep['harness'].items():chk(c.sha(R/p)==h,f'harness file changed {p}')
for w in ('c0','c1','c2','c3'):
    pf=E23/w/'postflight.json'
    if w!=W and pf.exists():chk(json.loads(pf.read_text())['verdict'].startswith('PASS'),f'window {w} postflight not PASS')
q=c.quiet()
out=dict(utc=c.now(),window=W,problems=problems,quiet_now=q['quiet'],blocking=q['blocking'],foreign=[(p['pid'],p['name']) for p in foreign])
n=len(list(E.glob('preflight*.json')));(E/(f'preflight-{n+1}.json')).write_text(json.dumps(out,indent=2,default=str))
print(json.dumps(out,default=str)[:2000]);sys.exit(1 if problems else 0)
