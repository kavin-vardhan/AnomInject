def link(label, path, line=None):
    target = str(path).replace("\\", "/") + (":" + str(line) if line else "")
    return "[%s](%s)" % (label, target)


def source(token):
    path = F / "verify_capture.py"
    lines = path.read_text(encoding="utf-8").splitlines()
    line = next(i for i, s in enumerate(lines, 1) if token in s)
    return link("verify_capture.py:%d" % line, path, line)


def fence(lines):
    return "```text\n" + (lines if isinstance(lines, str) else "\n".join(lines)).rstrip() + "\n```\n"


def summary(item):
    return next(s.strip() for s in item["lines"] if s.startswith("  CONSISTENT"))


def verdict(item):
    return next(s.strip() for s in item["lines"] if s.startswith("  VERDICT"))


def counts(cells):
    got = Counter(c["outcome"] for c in cells)
    return [got[k] for k in ("recovered", "wrong", "no-transition", "unassessable", "unsupported")]
