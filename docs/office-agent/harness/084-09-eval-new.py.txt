def detect_label_rule(rows):
    n=0;m=0
    for r in rows:
        for a in (r.get('anomalies') or []):
            n+=1;m+=1 if 'labelled' in a else 0
    if n==0:return 'empty'
    return 'new' if m==n else ('old' if m==0 else 'mixed')
def entry_of(s,si,typ,start_frame):
    rows=s.by_si if hasattr(s,'by_si') else s.rows
    r=rows.get(si)
    for a in ((r.get('anomalies') or []) if r else []):
        if TYPE_MAP.get(a.get('id'),a.get('id'))==typ and a.get('start_frame')==start_frame:return a
    return None
def reasons_of(a):
    if not a:return set()
    rs=a.get('transition_reason')
    if rs is None:return {LEGACY_T} if a.get('transition')==1 else set()
    if isinstance(rs,str):rs=[rs]
    return set(rs)
def held_textures(a):
    if not a:return 0
    return sum(1 for t in (a.get('stuck_mip.render_textures') or []) if str(t.get('level'))=='held')
def partial_guard(s,e,Lw,Tset,vis,span,drop,thr,temporal_aa):
    Ls=set(Lw);sp=set(span);V=set(vis)&sp
    ents={si:entry_of(s,si,ID52,e['start_frame']) for si in span}
    R={si:reasons_of(a) for si,a in ents.items()}
    info=dict(partial=[],unresolved=[],confirmed_partial=[],flagged_invisible=[],flagged_invisible_counted=[],partial_without_held_texture=[],
              partial_on_unlabelled=[],unresolved_only_off=[],legacy_transition=[],fail_reasons=[])
    Tex=set()
    for si in sorted(span):
        rs=R.get(si,set())
        if not rs:continue
        if 'temporal_aa' in rs or 'hide_return' in rs or LEGACY_T in rs:Tex.add(si)
        if LEGACY_T in rs:info['legacy_transition'].append(si)
        if 'unresolved' in rs:info['unresolved'].append(si)
        if 'partial' in rs:
            info['partial'].append(si)
            if si not in Ls:info['partial_on_unlabelled'].append(si);continue
            above=drop.get(si,0.0)>thr.get(si,0.0)
            (info['confirmed_partial'] if above else info['flagged_invisible']).append(si)
            if held_textures(ents[si])>=1:
                Tex.add(si)
                if not above:info['flagged_invisible_counted'].append(si)
            else:info['partial_without_held_texture'].append(si)
    off=((Ls&sp)-V)|(V-Ls)
    for si in sorted(off):
        rs=R.get(si,set())
        if rs and rs<={'unresolved'}:info['unresolved_only_off'].append(si)
    P=[si for si in Lw if si in set(info['partial'])];nonp=[si for si in Lw if si not in set(P)]
    if nonp:on=[si for si in P if si<nonp[0]];offp=[si for si in P if si>nonp[-1]];mid=[si for si in P if nonp[0]<si<nonp[-1]]
    else:on=list(P);offp=[];mid=[]
    info['edges']=dict(onset=len(on),offset=len(offp),mid=len(mid))
    info['edges_over_investigate']=[k for k,v in info['edges'].items() if v>PARTIAL_EDGE_INVESTIGATE]
    if info['partial_without_held_texture']:info['fail_reasons'].append('partial flag on a labelled frame whose render record shows NO texture at the held level - the flag cannot excuse it (084-07 decision 1)')
    if info['partial_on_unlabelled']:info['fail_reasons'].append('partial flag on an unlabelled frame - partial is a labelled-frame reason and cannot excuse a late label')
    if info['unresolved_only_off']:info['fail_reasons'].append('off frame backed only by unresolved - FAIL to investigate (084-07b N4)')
    tmp=[si for si in sorted(span) if R.get(si,set())&{'temporal_aa','hide_return',LEGACY_T}]
    info['temporal_flags']=tmp
    if temporal_aa is False and tmp:info['fail_reasons'].append('temporal transition flag(s) on a leg without temporal AA (the build emits temporal_aa / hide_return only under TAA or TSR; partial and unresolved are exempt)')
    return Tex,info
def m52_transition_gate(s,e,Lw,Tset,vis,span,drop,thr,sig,temporal_aa):
    Tex,pg=partial_guard(s,e,Lw,Tset,vis,span,drop,thr,temporal_aa)
    tg=gate_transition(Lw,Tex,vis,span,drop,sig,None,None)
    return Tex,pg,tg
def reference_amendment(s,ev,other,lo,hi,Ts,clean):
    prev=[x for x in other if x<lo];nxt=[x for x in other if x>hi]
    prev_last=max(prev) if prev else None
    start=max(lo-ls086.REF_BACK,(prev_last+PREV_SETTLE+1) if prev_last is not None else lo-ls086.REF_BACK)
    pre=[si for si in range(start,lo) if clean(si)]
    out=dict(prev_last=prev_last,pre=pre,post=[],basis='pre',frames=pre,status='OK',rule=AMENDMENT)
    if len(pre)>=REF_MIN:return out
    tail=max([hi]+[x for x in Ts if x>hi]);lim=min(nxt) if nxt else tail+1+POST_SKIP+POST_REF_MAX
    post=[si for si in range(tail+1+POST_SKIP,lim) if clean(si)][:POST_REF_MAX]
    out.update(post=post,basis='pre+post',frames=sorted(pre+post))
    if len(pre)+len(post)<REF_MIN:out.update(status='CONFOUNDED',basis='none')
    return out
def labelled_membership(sd):
    s=sess(sd);rule=detect_label_rule(s.rows);evs=anno_events(s)
    if rule!='new':
        return dict(rule=rule,pass_=rule in ('old','empty'),mismatch=[],mismatch_n=0,
                    note='read under the OLD rule: no entry carries labelled, and visible_positive meant event active + a box (084-07 changed it); membership is not checkable' if rule=='old' else ('no entries' if rule=='empty' else 'MIXED rules in one session: refused'))
    mism=[]
    for e in evs:
        L=set(e['L'])
        for si,a in e['entries'].items():
            lab=a.get('labelled') is True
            if lab!=(si in L):mism.append(dict(ord=e['ord'],type=e['type'],si=si,labelled=lab,in_annotation=si in L))
    return dict(rule='new',pass_=not mism,mismatch=mism[:30],mismatch_n=len(mism),
                note='new rule: every entry labelled == its frame is in its event annotation list (N1: one authority); the 086-01 checker judges exactly that list')

def mask_gate(sd,log='',min_events=56,require_peak_live=2,bbox_pad=16,outside_tol=0.005):
    g=mask_gate_v1(sd,log,min_events=min_events,bbox_pad=bbox_pad,outside_tol=outside_tol)
    s=sess(sd);sm=s.summary or {}
    own=[(int(a),int(b),int(c_)) for a,b,c_ in TAGOWN_RX.findall(log or '')]
    have={si for si,_,_ in own};shared=sorted({si for si,_,sh in own if sh>0})
    missing=[si for si in sorted(s.by_si) if si not in have]
    peak=sm.get('mask_tag_peak_live') or 0
    invalid=[]
    if missing:invalid.append(f'{len(missing)} captured frame(s) without a TAG-OWNERS line (first {missing[:5]}): the ownership instrument was not live')
    if peak<require_peak_live:invalid.append(f'mask_tag_peak_live {peak} < {require_peak_live}: no two mask records were ever live together, so the concurrent path was NOT exercised (not a pass)')
    fails=list(g['fail_reasons'])
    if shared:fails.append(f'{len(shared)} frame(s) where one mask value is carried by two holders (TAG-OWNERS shared>0; first {shared[:5]})')
    g.update(tag_owner_lines=len(own),tag_owner_shared_frames=shared[:20],tag_owner_missing=missing[:20],peak_live=peak,
             prior_collision=sm.get('mask_prior_collision'),prior_collision_quarantined=sm.get('mask_prior_collision_quarantined'),

    return g
def _flag(r,key):
    if r.get(key):return True
    for a in (r.get('anomalies') or []):
        if a.get(key) or ('camera_clipping_unconfirmed' in reasons_of(a) and key=='camera_clipping.unconfirmed'):return True
    return False

    if not r.get('valid'):r['gate']='INVALID';return r
    on=sess(on_sd);unc={k:_flag(on.by_si[k],'camera_clipping.unconfirmed') for k in range(frames)}
    lab={k:bool(on.by_si[k].get('camera_clipping.slab')) for k in range(frames)}
    D={int(k):v for k,v in r['D'].items()};theta=r['theta'];clipped={k:D[k]>theta for k in range(frames)}
    fr,_=cc_schedule(on_log);des={k:fr[k]['designed'] for k in range(frames)}

        lf=next((k for k in range(lo+1,hi+1) if lab[k]!=lab[k-1]),None);of=next((k for k in range(lo+1,hi+1) if clipped[k]!=clipped[k-1]),None)
        edges.append(dict(designed=k0,to=des[k0],label_flip=lf,pixel_flip=of,off=None if lf is None or of is None else lf-of))
    dis=r['rules'][RELEASE]['label_vs_clipped_disagree']

             edges=edges,edge_offsets=[e['off'] for e in edges])
    r['gate']='UNJUDGEABLE' if r['unjudgeable'] else ('PASS' if not dis and all((e['off'] in (0,None)) for e in edges) else 'FAIL')

FADE_LO=0.10
FADE_HI=0.90
FADE_MIN=2
FADE_PRE=3
FADE_POST_OFF=9
LOD_MIN_JUDGED=16
LOD_FADE_RULE='084-08b: at each pop edge of a judged event (onset window a-3..mid, offset window mid+1..b+9, clipped to the run window), a frame is INTERMEDIATE when its per-frame effect level (the E1 frac: 0 = the unforced reference, 1 = the run peak) is in [0.10, 0.90); an edge with >= 2 intermediate frames is a MULTI-FRAME FADE. For variant D it is a FINDING, reported and never excused (084-08 ruling 1); for variant N it is the control reading. Transition flags do not remove a frame from this count.'
def lod_edges(r):
    lv={int(k):float(v) for k,v in (r.get('levels') or {}).items()};out=[]
    for run in r.get('runs') or []:
        a,b=run['a'],run['b'];mid=(a+b)//2
        for kind,lo,hi in (('onset',a-FADE_PRE,mid),('offset',mid+1,b+FADE_POST_OFF)):
            lo=max(lo,run['lo']);hi=min(hi,run['hi'])
            seq=[(si,lv[si]) for si in range(lo,hi+1) if si in lv]
            inter=[si for si,v in seq if FADE_LO<=v<FADE_HI]
            out.append(dict(kind=kind,a=a,b=b,window=[lo,hi],levels=[[si,round(v,3)] for si,v in seq],intermediate=inter,
                            n_intermediate=len(inter),multi_frame_fade=len(inter)>=FADE_MIN))
    return out
def hist_of(v):
    return dict(sorted(collections.Counter(v).items()))
def lod_gate(events,variant=None,min_judged=LOD_MIN_JUDGED):

    judged=[r for r in ok if ((r.get('per') or {}).get(RELEASE) or {}).get('verdict') in ('PASS','FAIL')]
    off=[]

        tg=r['per'][RELEASE]['transition_gate']
        off+=[dict(start_frame=r.get('start_frame'),si=si,kind='unlabelled-visible') for si in tg['unlabelled_visible_without_transition']]
        off+=[dict(start_frame=r.get('start_frame'),si=si,kind='labelled-not-visible') for si in tg['labelled_not_visible_outside_transition']]
    fails=[dict(start_frame=r.get('start_frame'),reasons=r['per'][RELEASE]['fail_reasons']) for r in judged if r['per'][RELEASE]['verdict']=='FAIL']
    edges=[dict(start_frame=r.get('start_frame'),**e) for r in judged for e in lod_edges(r)]
    fades=[e for e in edges if e['multi_frame_fade']]
    ih={k:hist_of([e['n_intermediate'] for e in edges if e['kind']==k]) for k in ('onset','offset')}
    st=collections.Counter(r.get('status') if r.get('status')!='OK' else 'OK:'+str(((r.get('per') or {}).get(RELEASE) or {}).get('verdict')) for r in events)
    verdict='INSUFFICIENT' if len(judged)<min_judged else ('FAIL' if (off or fails) else 'PASS')
    return dict(verdict=verdict,variant=variant,judged=len(judged),min_judged=min_judged,events=len(events),status=dict(st),
                off_frames_t50_outside_transition=len(off),off=off[:40],fail_events=fails[:20],
                start=hist_of([e['start'] for r in judged for e in r['per'][RELEASE]['edges'] if e['start'] is not None and not e['cs']]),
                end=hist_of([e['end'] for r in judged for e in r['per'][RELEASE]['edges'] if e['end'] is not None and not e['ce']]),
                edges=len(edges),intermediate_hist=ih,fade_edges=len(fades),fades=fades[:24],
                fade_finding=bool(fades) and variant=='D',fade_control=(bool(fades) if variant=='N' else None),fade_rule=LOD_FADE_RULE,
                rule='gate: >= %d judged events (t50 PASS or FAIL) AND 0 off frames at t50 outside transition frames; fewer judged = INSUFFICIENT, never PASS'%min_judged)
