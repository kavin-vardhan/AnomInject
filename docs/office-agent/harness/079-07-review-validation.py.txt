def module(name, raw, filename):
    m = types.ModuleType(name)
    m.__file__ = str(filename)
    sys.modules[name] = m
    exec(compile(raw, str(filename), "exec"), m.__dict__)
    return m

def junction(src, dst):
    src, dst = pathlib.Path(src).resolve(), pathlib.Path(dst).absolute()
    assert dst.is_relative_to(scratch)
    assert src.is_relative_to(scratch) or src.is_relative_to(BANK_RESOLVED)

        os.symlink(src, dst, target_is_directory=True)
    except OSError:
        subprocess.run(["cmd", "/c", "mklink", "/J", str(dst), str(src)], check=True,
                       stdout=subprocess.DEVNULL, stderr=subprocess.PIPE)
    links.append(dst)

def gate(path, **kwargs):
    detail = []
    code, lines = vc.label_pixel_gate(str(path), 8, 4, 1, quiet=False, out_detail=detail, **kwargs)
    r = {"path": str(path), "code": code, "lines": lines, "detail": detail}
    print(pathlib.Path(path).name, "exit", code, flush=True)
    for line in lines:
        if line.startswith("idx=") or line.strip().startswith(("run[", "onset ", "end ", "VERDICT")):
            print(line, flush=True)
    return r

def fixture(name, runs=((60, 67),), label_runs=None, burst_at=None, onset_delta=0,
            full_boxes=(), total=100, patch_sides=None):
    """RGB truth is declared by runs; labels may differ. One persistent 10px scroll burst."""
    from PIL import Image, ImageDraw
    p = scratch / name
    (p / "Actual_Frames").mkdir(parents=True)
    w, h = 160, 120
    x, y, rw, rh = 40, 30, 50, 50
    bg = vc._synth_backdrop(w, h, total, 10, 16, 0)
    lr = list(label_runs if label_runs is not None else runs)
    if onset_delta:
        lr[0] = (lr[0][0] + onset_delta, lr[0][1])
    idxs = [k for a, b in lr for k in range(a, b + 1)]
    rows = []
    for k in range(total):
        offset = 10 if burst_at is not None and k >= burst_at else 0
        im = Image.new("RGB", (w, h), (90, 90, 90))
        im.paste(bg.crop((offset, 0, offset + rw, rh)).convert("RGB"), (x, y))
        for n, (a, b) in enumerate(runs):
            if a <= k <= b:
                side = patch_sides[n] if patch_sides else 20
                ImageDraw.Draw(im).rectangle((x + 5, y + 5, x + 4 + side, y + 4 + side), fill=(255, 255, 255))
        im.save(p / "Actual_Frames" / ("frame_%05d.png" % k))
        box = [0, 0, w, h] if k in full_boxes else [x, y, rw, rh]
        entry = {"id": "missing_texture", "target_name": "SynthTarget", "start_frame": 1000 + min(idxs),
                 "bbox_valid": True, "bbox_px": box}
        rows.append({"frame_index": 1000 + k, "session_index": k,
                     "image": "Actual_Frames/frame_%05d.png" % k, "width": w, "height": h,
                     "anomaly_present": k in idxs, "visible_positive": k in idxs,
                     "anomalies": [entry] if k in idxs else []})
    (p / "labels.jsonl").write_text("".join(json.dumps(r) + "\n" for r in rows), encoding="utf-8")
    ann = {"anomalies": [{"anomaly_type": "missing_texture", "manifested": True,
                          "affected_frames": {"start_frame": min(idxs), "end_frame": max(idxs),
                                              "frame_count": len(idxs), "frame_indices": idxs},
                          "affected_objects": {"primary_index": 0, "nodes": [{"name": "SynthTarget"}]}}]}
    (p / "annotation.json").write_text(json.dumps(ann), encoding="utf-8")
    return p

def key(c):
    return tuple(c[k] for k in ("session", "event", "ordinal", "kind", "which", "delta", "variant"))

def do_cohort():
    saved = json.loads((BANK / "M079_VERIFIER_EVIDENCE/079-06/cohort.json").read_text(encoding="utf-8"))
    RESULT["audit"] = {}
    expected_keys = set()
    for name in ("A2L_LEGA", "LYRA_SMOKE_01", "A1L_LEGA", "M50L_LG9"):
        src = sc.session_of(str(BANK), name)
        cohort = sc.cohort_of(src, mlo)
        for c in cohort:
            for which, delta in sc.PERTURBATIONS:
                for variant in sc.VARIANTS:
                    if delta == 0 and variant == "coherent":
                        continue
                    expected_keys.add((name, c["event"], c["ordinal"], c["kind"], which, delta, variant))
    for side in ("guard_off", "guard_on"):
        cells = saved[side]
        keys = [key(c) for c in cells]
        errors = []
        for c in cells:
            want = -c["delta"] if c["which"] in ("both", c["kind"]) else 0
            if c["expected"] != want:
                errors.append(["expected", c])
            if c["outcome"] != "refused":
                outcome = "recovered" if c["offset"] == want else "wrong"
                if c["outcome"] != outcome:
                    errors.append(["classification", c])
        RESULT["audit"][side] = {"count": len(cells), "unique": len(set(keys)),
            "missing_keys": sorted(expected_keys - set(keys)), "extra_keys": sorted(set(keys) - expected_keys),
            "counts": dict(Counter(c["outcome"] for c in cells)), "errors": errors,
            "by_session": dict(Counter(c["session"] for c in cells)),
            "refusal_reasons": dict(Counter(c["reason"] for c in cells if c["outcome"] == "refused"))}
    RESULT["audit"]["derived_cap"] = sc.derive_cap(saved["guard_off"])
    RESULT["audit"]["minimum_wrong"] = sorted([c for c in saved["guard_off"] if c["outcome"] == "wrong"], key=lambda c: c["m_edge"])
    successes = [c for c in saved["guard_on"] if c["outcome"] == "recovered"]
    RESULT["audit"]["success_edges"] = sorted(set((c["session"], c["event"], c["ordinal"], c["kind"]) for c in successes))
    RESULT["audit"]["success_m_range"] = [min(c["m_edge"] for c in successes), max(c["m_edge"] for c in successes)]
    targets = []
    for side, outcome, cap in (("guard_off", "wrong", 1.0), ("guard_on", "recovered", vc.REGION_CAP)):
        groups = sorted(set((c["session"], c["which"], c["delta"], c["variant"]) for c in saved[side] if c["outcome"] == outcome))
        targets.extend((side, cap, group) for group in groups)
    RESULT["audit"]["rerun_groups"] = targets
    print("COHORT_AUDIT", json.dumps(RESULT["audit"]), flush=True)
    RESULT["reruns"] = []
    for n, (side, cap, group) in enumerate(targets, 1):
        name, which, delta, variant = group
        src = sc.session_of(str(BANK), name)
        cells = sc.run_cell(src, str(scratch / "bank_shadows"), name, which, delta, variant,
                            cap, mlo, sc.cohort_of(src, mlo))
        want = {key(c): c for c in saved[side] if (c["session"], c["which"], c["delta"], c["variant"]) == group}
        diffs = []
        for c in cells:
            old = want[key(c)]
            for k in ("expected", "m_edge", "tau", "offset", "outcome", "reason"):
                if c[k] != old[k]:
                    diffs.append({"key": key(c), "field": k, "was": old[k], "now": c[k]})
        RESULT["reruns"].append({"guard": side, "group": group, "cells": cells, "differences": diffs})
        print("RERUN", n, "of", len(targets), side, group, "cells", len(cells), "differences", len(diffs), flush=True)
