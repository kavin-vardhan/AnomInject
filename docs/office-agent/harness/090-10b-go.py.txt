def manifest(tag):
    p = c.WORK / f'content-{tag}-{int(time.time())}.json'
    r = subprocess.run([sys.executable, str(MANIFEST), 'hash', str(p)], capture_output=True, text=True)
    return p if r.returncode == 0 else None


def run(label):
    spec = LEGS[label] if label in LEGS else LEGS[label.split('__')[0]]
    g = c.wait_quiet()
    rec = dict(label=label, gate=g)
    if not g['clear']:
        rec['status'] = 'NOT-QUIET'
        c.jsave(c.WORK / 'runs' / f'{label}.json', rec)
        return rec
    if spec['mode'] == 'staged':
        rec['stage'] = c.stage(spec['build'])
    else:
        rec['dlls'] = c.swap_editor(spec['build'])
        rec['manifest_before'] = str(manifest('pre'))
    rec['run'] = c.run_leg(label, spec)
    if spec['mode'] != 'staged':
        after = manifest('post')
        d = subprocess.run([sys.executable, str(MANIFEST), 'diff', rec['manifest_before'], str(after), '--rule', 'none'],
                           capture_output=True, text=True)
        rec['content_diff_rc'] = d.returncode
        rec['content_diff'] = (d.stdout + d.stderr)[-600:]
    rec['reading'] = c.read_leg(label)
    c.jsave(c.WORK / 'runs' / f'{label}.json', rec)
    return rec


def brief(rec):
    r = rec.get('reading') or {}
    px = [(e.get('type'), e.get('mode'), e.get('verdict'), e.get('median'), e.get('null')) for e in r.get('pixels', [])]
    s = r.get('summary') or {}
    keep = {k: v for k, v in s.items() if v not in (0, None, [], {}, False, '') }
    return dict(label=rec['label'], status=rec.get('status') or (rec.get('run') or {}).get('stdout', '')[-200:],
                events=r.get('events'), pixels=px, summary=keep, content_diff_rc=rec.get('content_diff_rc'))


if __name__ == '__main__':
    if sys.argv[1] == 'restore':

    elif sys.argv[1] == 'gate':

    else:
        for lab in sys.argv[1:]:
            print(json.dumps(brief(run(lab)), default=str), flush=True)
