    anom=[]
    for ln in log.splitlines():
        m=ANOM_CONTAM_RX.search(ln)
        if m:anom.append(dict(reason=m.group('reason'),owner=m.group('owner'),revert_logged=REVERT_NOW in ln))
    closed=collections.defaultdict(list)
    for m in TRAIL_CLOSED_RX.finditer(log):closed[m.group('event')].append(int(m.group('si')))
    unresolved={m.group('event') for m in UNRESOLVED_RX.finditer(log)}
    evs={e['start_frame']:e for e in events(s)} if s is not None else {}
    recs=[];used=set();logged=set()
    for c in cap:
        key=c['event'];fsi=int(c['si']);r=dict(event=key,from_si=fsi,reason=c['reason'],problems=[])
        if not re.fullmatch(re.escape(ID)+r'@\d+',key):r['problems'].append('contamination not attributable to an event')
        mi=next((i for i,a in enumerate(anom) if i not in used and a['reason']==c['reason']),None)
        if mi is None:r['problems'].append('no anomaly-side HOLD CONTAMINATED line with this reason')
        else:
            used.add(mi);r['revert_logged']=anom[mi]['revert_logged']
            if not anom[mi]['revert_logged']:r['problems'].append(f'anomaly-side line does not say {REVERT_NOW}')
        cs=[x for x in closed.get(key,[]) if x>=fsi];r['trail_closed_at']=cs[0] if cs else None
        if key in unresolved:r['problems'].append('RESTORE UNRESOLVED after contamination: the revert is not confirmed back at baseline')
        elif not cs:r['problems'].append('no RESTORE TRAIL CLOSED at or after the contamination: the revert is not confirmed back at baseline')
        if s is None:r['problems'].append('no session supplied: per-frame stuck_mip.contaminated flags not verified')
        else:
            sf=int(key.split('@')[1]) if '@' in key and key.split('@')[1].isdigit() else None;e=evs.get(sf)
            if e is None:r['problems'].append('event not present in labels.jsonl')
            else:
                logged.add(sf);mem=sorted(si for si,st in e['rstate'].items() if st in MEMBER_STATES)
                need=[si for si in mem if si>=fsi];r['member_frames_from_si']=len(need)
                r['unflagged_member_frames']=[si for si in need if not e['contam'].get(si)]
                r['wrongly_flagged_frames']=sorted(si for si,v in e['contam'].items() if v and (si<fsi or si not in mem))
                r['flagged_frames']=len(need)-len(r['unflagged_member_frames'])
                if r['unflagged_member_frames']:r['problems'].append('member frame(s) from the contamination si without stuck_mip.contaminated = 1')
                if r['wrongly_flagged_frames']:r['problems'].append('stuck_mip.contaminated = 1 on frame(s) before the contamination si or outside membership')
                if not need:r['problems'].append('no member frame at or after the contamination si to carry the flag')
        r['handled_as_ruled']=not r['problems'];recs.append(r)
    orphan_flags=[]
    for sf,e in evs.items():
        if sf not in logged and any(e['contam'].values()):orphan_flags.append(dict(event=f'{ID}@{sf}',frames=sorted(si for si,v in e['contam'].items() if v)))
    unmatched_anom=[a for i,a in enumerate(anom) if i not in used]
    sm=(s.summary if s is not None else {}) or {}
    return dict(events=recs,flags_without_contamination_line=orphan_flags,anomaly_lines_without_capture_line=unmatched_anom,
                counters=dict(capture_lines=len(cap),anomaly_lines=len(anom),summary_hold_contaminations=sm.get('stuck_mip_hold_contaminations'),
                              summary_contaminated_frames=sm.get('stuck_mip_contaminated_frames'),
                              row_flags=sum(1 for e in evs.values() for v in e['contam'].values() if v) if s is not None else None),
                failed=[r['event'] for r in recs if not r['handled_as_ruled']]+[x['event'] for x in orphan_flags]+(['(anomaly-side contamination with no capture-side event)'] if unmatched_anom else []))

    for i,hs in enumerate(fires):
        got=sorted(ev[i][1]) if i<len(ev) else None

    if bad:v='FAIL'

                per_event_join=join[:20],join_ok=join_ok,wrong_scope=wrong_scope[:5],bad=bad[:20],refused_shared_world=ref[:20],
                inactive_level_users=sum(int(t.get('inact') or 0) for t in tex),unregistered_users=sum(int(t.get('unreg') or 0) for t in tex),
                apply_census_verdict=census,contaminations=cc['events'],contamination=cc,contaminated_events=[r['event'] for r in cc['events']],
                lifetime_note=('the Apply census is joined per fire; lifetime purity is judged only through the contaminations the build '
                               'itself detects and logs (HOLD CONTAMINATED). A new user the hold monitor misses (084-02b Codex F4 routes) '
                               'is invisible to this gate'),verdict=v)

def closure_rule(after,before):
    if not after:return dict(rule='NO-JUDGEABLE-EVENT')
    out=dict(after_min=min(after),after_max=max(after),before_min=min(before) if before else None,before_max=max(before) if before else None,n=len(after))
    onset_ok=bool(before) and min(before)==0 and max(before)==0
    if min(after)==max(after) and min(after)>=0:
        out['rule']=f'constant settle tail {min(after)} frame(s) gives 0 at the offset on every judged event'
        out['constant_tail']=min(after)
    else:
        out['rule']='no constant tail gives 0 at the offset (the residual varies by event); an in-engine settling detector is needed'
        out['constant_tail']=None
    out['onset_exact_from_record']=onset_ok
    if not onset_ok:out['onset_note']='the first record-held frame is not the first pixel-visible frame on every event; the onset needs its own rule'
    return out
def null_check(null_sess):
    s=null_sess;evs=events(s);out=dict(events=len(evs),labelled_held_rows=0,measurable_depth_events=0,frames=len(s.rows),render_held_rows=0)
    for e in evs:
        out['labelled_held_rows']+=sum(1 for v in e['held'].values() if v)
        out['render_held_rows']+=len(render_held(e))
    firsts=[e['rows'][0] for e in evs]
    for t in sorted({e['target'] for e in evs}):
        ref=null_ref(s,t)
        if not ref:continue
        out.setdefault('refs',{})[t]=dict(mu=round(ref['mu'],4),sigma_resid=round(ref['sigma'],4),worst_drop_rel=round(ref['drop_rel'],4) if ref['drop_rel'] is not None else None,
                                         null_events_used=ref['null_events'],floor_rel=REL_FLOOR,n=ref['n'])
    out['note']='the null CALIBRATES the threshold (its own worst drop below its pre-event level under the same procedure), so its own events cannot exceed it by construction; the null gate is: no labelled or render-held row, and at least one settled null event to calibrate from'
    out['pass']=out['events']>0 and out['labelled_held_rows']==0 and out['render_held_rows']==0 and any((v.get('null_events_used') or 0)>0 for v in (out.get('refs') or {}).values())
    return out
def manifest(root,exclude=('_leg',)):
    root=Path(root);out={}
    for p in sorted(root.rglob('*')):
        if p.is_file() and not any(part in exclude for part in p.relative_to(root).parts):out[str(p.relative_to(root)).replace('\\','/')]=[p.stat().st_size,c.sha(p)]
    return out
class BankError(Exception):pass
def bank_attempt(label,ev_dir,exe_side=None,bank=None):
    exe_side=Path(exe_side or c.EXE_SIDE);bank=Path(bank or c.BANK);src=exe_side/label;dst=bank/label;ev_dir=Path(ev_dir)
    rec=dict(label=label,src=str(src),dst=str(dst),utc=c.now())
    if dst.exists():
        prev=jload(ev_dir/'bank-manifest.json')
        if prev and prev.get('verified') and manifest(dst)==prev['manifest'] and not src.exists():return dict(prev,resumed=True)
        raise BankError(f'bank destination exists without a verified record: {dst}')
    dst.mkdir(parents=True)
    if src.exists():shutil.copytree(src,dst,dirs_exist_ok=True)
    legd=dst/'_leg';legd.mkdir()
    for f in ('game.log','launch.json','runner.log','command.json'):
        if (ev_dir/f).exists():shutil.copy2(ev_dir/f,legd/f)
    ms=manifest(src) if src.exists() else {};md=manifest(dst)
    rec.update(manifest=md,src_files=len(ms),dst_files=len(md),verified=(ms==md))
    if not rec['verified']:
        rec['mismatch']=sorted(k for k in set(ms)|set(md) if ms.get(k)!=md.get(k))[:20]
        jsave(ev_dir/'bank-manifest.json',rec);raise BankError(f'bank manifest mismatch for {label}: exe-side copy KEPT')
    if src.exists():
        shutil.rmtree(src)
        if src.exists():raise BankError(f'exe-side copy not deleted: {src}')
        rec['exe_side_deleted']=True

class PersonMonitor:
    def __init__(self,label,interval=PP_SAMPLE_S):
        self.label=label;self.interval=interval;self.samples=[];self._stop=threading.Event();self._th=None;self.start=self.end=None
    def _loop(self):
        while True:
            try:self.samples.append(_desktop_sample())
            except Exception:pass

def person_evidence(launch,mon):
    if not launch.get('sampling_started_utc'):return dict(person_evidence=False,reasons=[],note='no steady-state span')
    lo=_epoch(launch['sampling_started_utc']);hi=_epoch((launch.get('anchor_end') or {}).get('utc') or launch.get('finished_utc') or launch['sampling_started_utc'])
    pid=launch.get('pid');ms=[x for x in mon.samples if lo<=x['t']<=hi];reasons=[]
    ticks=[x for x in ms if x.get('input_tick') is not None]
    inputs=sum(1 for a,b in zip(ticks,ticks[1:]) if a['input_tick']!=b['input_tick'] and a.get('fg_pid')==pid and b.get('fg_pid')==pid)
    foreign_fg=sum(1 for x in ms if x.get('fg_pid') not in (pid,None))
    if inputs:reasons.append(f'non-harness input while the game held focus: {inputs}')
    if foreign_fg>PP_FORCE_SAMPLES:reasons.append(f'foreground left the game for {foreign_fg} samples')
    return dict(person_evidence=bool(reasons),reasons=reasons,samples=len(ms))

def backup_m53():
    dst=c.M53_BACKUP
    if dst.exists():
        st={k:c.sha(dst/k) for k in c.KEYS if (dst/k).exists()}
        if st!=c.M53:raise Stop(10,f'M53 backup exists and does not match the m53 set: {st}')
        return dict(resumed=True,path=str(dst))
    if c.staged_state()!=c.M53:raise Stop(10,'refusing to back up: the staged set is not the m53 set')
    for k in c.KEYS:(dst/k).parent.mkdir(parents=True,exist_ok=True);shutil.copy2(c.STAGED_ROOT/k,dst/k)
    st={k:c.sha(dst/k) for k in c.KEYS}
    if st!=c.M53:raise Stop(10,'M53 backup copy does not verify')

def stage(name,tag):
    rec=dict(set=name,tag=tag,utc=c.now(),stub=c.STUB)
    if c.STUB:rec.update(note='STUB: nothing staged');jsave(c.RUN/f'receipt-{tag}.json',rec);return rec
    want={'BASELINE':c.BASELINE,'FIX':c.fix_set() if name=='FIX' else None,'M53':c.M53}[name]
    q=c.quiet()
    if not q['quiet']:raise Stop(4,f'NOT-QUIET at staging {name}: {q["blocking"]}')
    before=c.staged_state();src=c.sources(name);copied=[]
    for k in c.KEYS:
        if before[k]==want[k]:continue
        if c.sha(src[k])!=want[k]:raise Stop(10,f'source for {k} does not match {name}: {src[k]}')
        shutil.copy2(src[k],c.STAGED_ROOT/k);copied.append(k)
    after=c.staged_state()
    rec.update(before=before,after=after,copied=copied,verified=(after==want))
    jsave(c.RUN/f'receipt-{tag}.json',rec)
    if after!=want:raise Stop(10,f'staging {name} did not verify: {[k for k in c.KEYS if after[k]!=want[k]]}')
    return rec
