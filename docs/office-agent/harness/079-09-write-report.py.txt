def link(path, label=None, line=None):
    return "[%s](%s%s)" % (label or path.name, path.as_posix(), ":%d" % line if line else "")

def loc(filename, needle):
    p = F / filename
    line = next(i for i, s in enumerate(p.read_text(encoding="utf-8").splitlines(), 1) if needle in s)
    return link(p, filename + ":" + str(line), line)
