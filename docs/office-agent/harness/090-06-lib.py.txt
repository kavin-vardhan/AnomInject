def _copy(src,dst,log=None,tries=None,backoff=None,sleep=time.sleep):

                raise Stop(11,f'copy {src} -> {dst} failed {tries} times: {errs[-1]}')
            sleep(backoff[min(i,len(backoff)-1)])

def hold_exclusive(path,seconds):
    """Dry-only G355 proof: hold the first staged sandbox file open without sharing so the restore must retry."""
    ready=threading.Event();info={}
    def run():
        h=_K32.CreateFileW(str(path),0x80000000,0,None,3,0x80,None)
        info['handle_ok']=h not in (None,0,-1,0xFFFFFFFFFFFFFFFF);ready.set();time.sleep(seconds)
        if info['handle_ok']:_K32.CloseHandle(h)
    threading.Thread(target=run,daemon=True).start();ready.wait(5);return info

def stage(name,tag,prerun=None):
    """Stage NEW / OLD (the pinned container + that exe) or PRERUN (the receipted pre-run pairing), copying only differing files,
    hashing all six before and after, with a receipt. NEW with a placeholder exe pin is refused live; in a dry run the container is
    staged into the sandbox and the exe is recorded as NOT staged (pin pending)."""
    rec=dict(set=name,tag=tag,utc=c.now(),dry=c.DRY,root=str(c.STAGED_ROOT));copies=[]
    if not c.DRY:
        q=c.idle_gate()
        if not q['idle']:raise Stop(4,f'NOT-IDLE at staging {name}: {q["blocking"][:4]}')
    before=c.staged_state();rec['before']=before;rec['before_set']=which_set(before)
    if name=='PRERUN':
        want=dict(prerun['state']);src={k:Path(v) for k,v in prerun['restore_source']['sources'].items()}
    else:
        want=c.binary_set(name);src=c.binary_sources(name)
    skipped=[]
    for k in c.KEYS:
        if want.get(k) is None:
            if c.DRY and name=='NEW' and k==c.EXE_KEY:skipped.append(k);continue
            raise Stop(10,f'staging {name}: {k} has no pinned hash (placeholder)')
        if before[k]==want[k]:continue
        if c.sha(src[k])!=want[k]:raise Stop(10,f'source for {k} does not match {name}: {src[k]}')
        _copy(src[k],c.STAGED_ROOT/k,copies)
    after=c.staged_state();ok=all(after[k]==want[k] for k in c.KEYS if k not in skipped)
    rec.update(want=want,after=after,after_set=which_set(after),copies=[dict(x,src=Path(x['src']).name) for x in copies],skipped=skipped,verified=ok,
               note=('DRY: exe NOT staged, the NEW pin is PENDING; the container files were staged into the sandbox' if skipped else None))
    jsave(c.RUN/f'receipt-{tag}.json',rec)
    if not ok:raise Stop(10,f'staging {name} did not verify: {[k for k in c.KEYS if k not in skipped and after[k]!=want[k]]}')
    return rec

        if not got or got[-1]!='0':errs.append(f'read-back: r.AntiAliasingMethod echoes {got[-1] if got else "nothing"}, declared 0')
    if 'arbiter' in L['readback']:
        for k in ('r.AntiAliasingMethod','r.Lumen.DiffuseIndirect.Allow','r.DynamicGlobalIlluminationMethod','r.ReflectionMethod'):
            got=re.findall(r'\]\s*'+re.escape(k)+r' = "([^"]*)"',log or '')
            if not got or got[-1]!='0':errs.append(f'arbiter read-back: {k} echoes {got[-1] if got else "nothing"}')
    if 'showflags' in L['readback']:
        for sf in [x for x in L['levers'] if x.startswith('ShowFlag.')]:
            nme,val=sf.split()[0],sf.split()[-1];got=re.findall(re.escape(nme)+r' = "([^"]*)"',log or '')
            if not got or got[-1]!=val:errs.append(f'show-flag read-back: {nme} echoes {got[-1] if got else "nothing"}, declared {val}')
    if 'off16' in L['readback'] and rs.get('label_transition_off_frames')!=16:errs.append(f'read-back: label_transition_off_frames {rs.get("label_transition_off_frames")} != 16')
    if 'nanite_policy' in L['readback'] and 'IAI.Targets.AllowNanite 1' in L['levers']:
        if rs.get('nanite_target_policy')!='allowed(console)' or 'EFFECTIVE READ-BACK = allowed(console)' not in (log or ''):
            errs.append(f'read-back: IAI.Targets.AllowNanite 1 did not reach the run (nanite_target_policy {rs.get("nanite_target_policy")!r})')
    if 'lever' in L['readback']:
        kind,si=L['lever'];rx=re.compile(r'Capture\(bench\): '+re.escape(kind)+r'\b[^\r\n]*session_index=(\d+)(?![^\r\n]*NOTHING)')
        if not any(int(m)==si for m in rx.findall(log or '')):errs.append(f'lever: no "Capture(bench): {kind} ... session_index={si}" line that acted (INVALID: a clean reading would be blindness)')
    present=[x for x in NEW_CODE_LINES if x in (log or '')]
    if L['binary']=='NEW' and len(present)!=len(NEW_CODE_LINES):errs.append(f'binary identity: the NEW exe must log {NEW_CODE_LINES}; seen {present}')
    if L['binary']=='OLD' and present:errs.append(f'binary identity: the OLD exe logged 090-05 lines {present}')
    return errs
def validity(L,launch,sess,log):
    errs=[]
    if launch.get('status')!='COMPLETE':errs.append('protocol: '+str(launch.get('error')))
    want=c.binary_set(L['binary'])[c.EXE_KEY]
    if (launch.get('exe_sha256') or '').lower()!=want:errs.append(f'binary: the launched exe {str(launch.get("exe_sha256"))[:16]} != the {L["binary"]} pin {str(want)[:16]}')
    fa=launch.get('focus_at_s')
    if fa is None or fa<0 or fa>=20.0:errs.append(f'A63: focus not held within 20 s (focus_at_s={fa})')
    cf=launch.get('capture_focus') or [];cf=[cf] if isinstance(cf,dict) else cf
    if cf and not all(x.get('foreground_pid')==launch.get('pid') for x in cf):errs.append('whole-leg foreground failed')
    if sess is None:errs.append('artifact: no session with run_summary');return errs
    rows=[l for l in (sess/'labels.jsonl').read_text(encoding='utf-8-sig').splitlines() if l.strip()] if (sess/'labels.jsonl').exists() else []
    pngs=list((sess/'Actual_Frames').glob('*.png'))
    if len(rows)!=L['frames']:errs.append(f'artifact: {len(rows)} label rows != {L["frames"]} frames')
    if len(pngs)!=len(rows):errs.append(f'artifact: {len(pngs)} PNGs != {len(rows)} rows')
    if any(p.stat().st_size==0 for p in pngs):errs.append('artifact: zero-byte PNG')
    if '-IAIBench' in (L.get('extra') or '') and 'IAI bench levers: ENABLED' not in (log or ''):errs.append('G377: -IAIBench given but no "IAI bench levers: ENABLED" line')
    return errs+readback_errors(L,log,sess)

def run_leg(L,runner=real_runner):
    rows=[x for x in ledger() if x['leg']==L['leg']]
    if any(x['status'] in ('ACCEPTED','DRY-STUB','REFUSED-PENDING-PIN','NOT-RUNNABLE') for x in rows):return 'done'
    ok,why=leg_runnable(L)
    if not ok:
        row=dict(leg=L['leg'],group=L['group'],status=why.split(':')[0],reason=why,recorded=c.now(),minutes=L['minutes'])
        if c.DRY and L['kind']=='capture':
            lab=label_for(L);row['exec_cmds']=exec_cmds(L,lab);row['exec_chars']=len(row['exec_cmds'])
            st=attach_standin(L,lab,c.RUN/lab)
            if st:row.update(standin=st,label=lab,session=st['session'],log=str(c.RUN/lab/'game.log') if st.get('log') else None)
        ledger_add(row);print(json.dumps(dict(leg=L['leg'],status=row['status'],reason=why[:160]),default=str),flush=True);return 'refused'
    while True:
        rows=[x for x in ledger() if x['leg']==L['leg']];bad=[x for x in rows if x['status']=='INVALID'];env=[x for x in rows if x['status']=='ENV-INTERRUPTED']
        if len(env)>=PP_CAP:raise Stop(4,f'NOT-QUIET: {L["leg"]} has {len(env)} ENV-INTERRUPTED attempts')
        if len(bad)>=ATTEMPTS:ledger_add(dict(leg=L['leg'],group=L['group'],status='NOT-RUN',recorded=c.now(),reason='; '.join('|'.join(x.get('errors') or []) for x in bad)));return 'not-run'
        pp_gate('before-attempt',L['leg']);ig=idle_wait(L['leg'])
        if not c.DRY:
            staged=c.staged_state()[c.EXE_KEY];want=c.binary_set(L['binary'])[c.EXE_KEY]
            if staged!=want:raise Stop(10,f'{L["leg"]}: the staged exe {str(staged)[:16]} is not its {L["binary"]} pin {str(want)[:16]}')
        n=len([x for x in rows if x.get('label')])+1;label=label_for(L,n);ev=c.RUN/label
        if ev.exists() and not c.DRY:raise Stop(10,f'evidence dir exists for a new attempt: {ev}')
        cmds=build_cmds(L,label);args=runner_args(L,label,ev,cmds);t0=c.now()
        rc=runner(args,c.RUN/f'runner-{label}.log')
        ev.mkdir(parents=True,exist_ok=True);jsave(ev/'command.json',dict(args=args,cmds=cmds,leg=L))
        launch=jload(ev/'launch.json',{}) or {}
        row=dict(leg=L['leg'],group=L['group'],label=label,attempt=n,started=t0,finished=c.now(),runner_exit=rc,exec_chars=len(', '.join(cmds)),exec_cmds=', '.join(cmds),minutes=L['minutes'],idle=ig)
        if c.DRY:
            st=attach_standin(L,label,ev)
            row.update(status='DRY-STUB',launch=dict(status=launch.get('status'),exe=launch.get('exe'),exe_exists=launch.get('exe_exists'),exe_sha256=launch.get('exe_sha256')),
                       standin=st,session=(st or {}).get('session'),log=str(ev/'game.log') if (st or {}).get('log') else None)
            ledger_add(row);print(json.dumps(dict(leg=L['leg'],status='DRY-STUB',launch=launch.get('status')),default=str),flush=True);return 'dry'
        if 'FOREIGN-EDITOR-APPEARED' in str(launch.get('error') or ''):row.update(status='INVALID-FOREIGN',errors=['foreign editor appeared']);ledger_add(row);raise Stop(5,f'INVALID-FOREIGN {label}')
        if launch.get('status')=='REFUSED-NOT-IDLE':row.update(status='INVALID-NOT-IDLE');ledger_add(row);raise Stop(4,f'NOT-IDLE at launch of {label}')
        try:bk=bank_attempt(label,ev,c.EXE_SIDE)
        except BankError as ex:row.update(status='INVALID-BANK',errors=[str(ex)]);ledger_add(row);raise Stop(6,str(ex))
        bsd=find_session(c.BANK/label);log=(ev/'game.log').read_text(encoding='utf-8-sig',errors='replace') if (ev/'game.log').exists() else ''
        errs=validity(L,launch,bsd,log)
        if errs:row.update(status='INVALID',errors=errs);ledger_add(row);print(json.dumps(dict(label=label,status='INVALID',errors=errs)),flush=True);continue
        try:al=alias_hardlinks(label,c.BANK_PREFIX+L['leg'])
        except BankError as ex:row.update(status='INVALID-BANK',errors=['alias: '+str(ex)]);ledger_add(row);raise Stop(6,'alias: '+str(ex))
        row.update(status='ACCEPTED',alias=al.get('alias'),session=str(bsd) if bsd else None,log=str(ev/'game.log'),bank_files=bk.get('dst_files'));ledger_add(row)
        print(json.dumps(dict(label=label,status='ACCEPTED')),flush=True);return 'accepted'

def eval_pool(legs):
    d=c.RUN/'eval';d.mkdir(parents=True,exist_ok=True);out={}
    def one(k):
        p=d/f'{k}.json'
        if p.exists():p.unlink()
        cp=subprocess.run([c.PYTHON,'-B',str(R/'090-06-lib.py'),'evalleg',k],capture_output=True,text=True,encoding='utf-8',errors='replace',env=dict(os.environ))
        if p.exists():return json.loads(p.read_text(encoding='utf-8'))
        return dict(verdict='ERROR',error='evalleg exit %s: %s'%(cp.returncode,((cp.stdout or '')+(cp.stderr or ''))[-1500:]))

def postflight(prerun):
    problems=[]
    st=c.staged_state()
    if st!=prerun['state']:problems.append('staged set is not the pre-run pairing: '+json.dumps({Path(k).name:(v or '')[:8] for k,v in st.items()}))
    left=[d.name for d in c.EXE_SIDE.iterdir() if d.is_dir() and d.name.startswith(c.BANK_PREFIX)] if c.EXE_SIDE.exists() else []
    if left:problems.append(f'leg copies left after banking: {left[:5]}')
    links=0
    for x in [x for x in ledger() if x.get('status')=='ACCEPTED']:
        a=c.BANK/x['label'];al=c.BANK/(c.BANK_PREFIX+x['leg'])
        for p in [p for p in a.rglob('*') if p.is_file()]:
            q=al/p.relative_to(a)
            if not q.exists() or os.stat(q).st_ino!=os.stat(p).st_ino:problems.append(f'alias not a hardlink: {q}');break
            links+=1
    ig=c.idle_gate();ours=[b for b in ig['blocking'] if (b.get('name') or '').lower()=='stackobot.exe']
    if ours:problems.append(f'a StackOBot process is still running: {ours}')
    out=dict(utc=c.now(),problems=problems,alias_hardlinks_verified=links,staged=which_set(st),verdict='PASS' if not problems else 'FAIL')
    jsave(c.RUN/'postflight.json',out);return out

                           copies=[(Path(x['dst']).name,x['attempt']) for x in (restore or {}).get('copies',[])]),postflight=pf2.get('verdict'),postflight_problems=pf2.get('problems'),
