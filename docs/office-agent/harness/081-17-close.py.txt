from pathlib import Path
import datetime,hashlib,json,subprocess,sys
B=Path(r'D:\IntrusiveAnomalies');E=B/'_reviews/081-17-evidence';R=B/'StackOBot/Plugins/AnomalyInjector'
L=Path(r'E:\UEProjects\LyraStarterGame\Plugins\AnomalyInjector');C=B/'StackOBot/Plugins/CaptureBench';FEATURE='feat/m55-change-evidence'
def sha(p):
    with Path(p).open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()
def git(r,*a):return subprocess.check_output(['git','-C',str(r),*a]).decode().strip()
def idle():
    s=subprocess.check_output(['powershell','-NoProfile','-Command','@(Get-Process -Name UnrealEditor,StackOBot,LyraGame,UnrealBuildTool -ErrorAction SilentlyContinue | Select-Object Id,ProcessName) | ConvertTo-Json -Compress']).decode().strip()
    return [] if not s or s=='[]' else json.loads(s) if s.startswith('[') else [json.loads(s)]
def state(r):
    return {'head':git(r,'rev-parse','HEAD'),'branch':git(r,'branch','--show-current'),
            'refs':git(r,'for-each-ref','--format=%(refname) %(objectname)','refs/heads','refs/remotes'),
            'untracked':{p:sha(r/p) for p in git(r,'ls-files','--others','--exclude-standard').splitlines()}}
pre=json.loads((E/'preflight.json').read_text());problems=[]
procs=idle();foreign=[p for p in procs]
if foreign:problems.append(['bench processes present at close',foreign])
feature=git(R,'rev-parse',FEATURE)
if feature!=git(R,'rev-parse','origin/'+FEATURE):problems.append(['feature not pushed',feature])
for name,r in {'AnomalyInjector':R,'Lyra':L,'CaptureBench':C}.items():
    if git(r,'diff','--name-only') or git(r,'diff','--cached','--name-only'):problems.append([name,'tracked modifications'])
if '--checkout' in sys.argv:git(R,'checkout','m51')
repos={}
for name,r in {'AnomalyInjector':R,'Lyra':L,'CaptureBench':C}.items():
    st=state(r);old=pre['repos'][name]
    for k in ('head','branch','untracked'):
        if st[k]!=old[k]:problems.append([name,k,old[k] if k!='untracked' else None,st[k] if k!='untracked' else sorted(set(st[k])^set(old[k]))])
    a=dict(l.split(' ',1) for l in old['refs'].splitlines());z=dict(l.split(' ',1) for l in st['refs'].splitlines())
    delta={k:[a.get(k),z.get(k)] for k in a.keys()|z.keys() if a.get(k)!=z.get(k)}
    allowed={'refs/heads/'+FEATURE,'refs/remotes/origin/'+FEATURE} if name in ('AnomalyInjector','Lyra') else set()
    if set(delta)-allowed:problems.append([name,'unexpected ref delta',delta])
    st['ref_delta']=delta;repos[name]=st
m=json.loads((E/'binary-v2.json').read_text());main=json.loads((E/'main-editor-binary.json').read_text())
changed_live={m['built']:m['sha256'],m['staged']:m['sha256']}
for row in main:changed_live[row['path']]=row['sha256']
bins={}
for p,h in pre['verified_binaries'].items():
    a=sha(p);bins[p]=a
    if a!=changed_live.get(p,h):problems.append(['binary',p])
for p,h in changed_live.items():
    a=sha(p);bins[p]=a
    if a!=h:problems.append(['changed live binary',p])
bins[m['archive']]=sha(m['archive'])
if bins[m['archive']]!=m['sha256']:problems.append(['archive',m['archive']])
for row in m['lyra_modules']:
    p=str(Path(m['lyra_archive'])/row['name']);bins[p]=sha(p)
    if bins[p]!=row['sha256']:problems.append(['lyra archive',p])
for row in main:
    bins[row['archive']]=sha(row['archive'])
    if bins[row['archive']]!=row['sha256']:problems.append(['main editor archive',row['archive']])
lyra_names=sorted(p.name for p in (L/'Binaries/Win64').iterdir() if p.is_file())
if lyra_names!=pre['lyra_module_names']:problems.append(['Lyra module set',lyra_names])
for p,h in pre['prior_artifacts'].items():
    if sha(B/p)!=h:problems.append(['artifact',p])
for p,h in pre['bank_files'].items():
    if sha(B/p)!=h:problems.append(['bank',p])
cur=sha(B/'ToCodex/CURRENT.md')
if cur!=pre['current_sha256']:problems.append(['ToCodex CURRENT.md changed'])
remote=git(R,'ls-remote','origin','refs/heads/'+FEATURE,'refs/heads/master')
if feature+'\trefs/heads/'+FEATURE not in remote or '031a103b18b90b2ce9dd626b22e9498717c60815\trefs/heads/master' not in remote:problems.append(['remote',remote])
if repos['AnomalyInjector']['head']!='53bf725bd406f8ce3e38d6ed153b7ff77e3a3e67' or repos['AnomalyInjector']['branch']!='m51':problems.append(['main checkout not m51'])
if repos['Lyra']['head']!='caa68c6799db023f7a24c0d6cfc56124ce27b542':problems.append(['Lyra head'])
bank={str(p.relative_to(B)):sha(p) for d in (B/'_bench_sessions_bank').glob('M55S2R17_*') for p in d.rglob('*') if p.is_file()}
(E/'bank-manifest.json').write_text(json.dumps(bank,indent=2))
report=B/'_reviews/081-17-code-m55-build2-report.md'
out=dict(utc=datetime.datetime.now(datetime.timezone.utc).isoformat(),feature_commit=feature,repos=repos,verified_binaries=bins,
    lyra_module_names=lyra_names,prior_artifacts_verified=len(pre['prior_artifacts']),prior_bank_files_verified=len(pre['bank_files']),
    new_bank_files_hashed=len(bank),current_sha256_unchanged=cur==pre['current_sha256'],remote_refs=remote,bench_processes=procs,
    report_sha256=sha(report) if report.exists() else None,problems=problems,
    verdict='PASS: boundary restored (main m51 53bf725, Lyra caa68c6 with original 9 modules)' if not problems else 'FAIL: see problems')
(E/'postflight.json').write_text(json.dumps(out,indent=2))
man={str(p.relative_to(B)):sha(p) for p in E.rglob('*') if p.is_file() and p.name not in ('evidence-manifest.json',)}
for p in sorted((B/'_reviews').glob('081-17-*')):
    if p.is_file():man[str(p.relative_to(B))]=sha(p)
(E/'evidence-manifest.json').write_text(json.dumps(man,indent=2))
print(json.dumps({k:v for k,v in out.items() if k not in ('repos','verified_binaries')},indent=2))
sys.exit(1 if problems else 0)
