def coalesced_unmeasured(text):

        u=TM_UNAV_RE.search(l)

    for n,si,rid,px,line in unav:
        host=[p for p in passes if p[0]<n and rid in p[1]]
        rec=dict(si=si,request_id=rid,pixels=px,unavailable_line=line,served_line=host[-1][3] if host else None,signature=False,why='')
        if px!=0:rec['why']='pixels != 0';out.append(rec);continue

        ids=host[-1][1];tg=[i for i in ids if is_target_arm(i)];pos=ids.index(rid);low=(1<<TARGET_ARM_BIT)-1

def colour_file_meta(session,rel):
    p=Path(session)/rel;m=dict(file=str(rel).replace('\\','/'))
    if not p.is_file():m['missing']=True;return m
    try:
        with p.open('rb') as f:h=f.read(33)
        if h[:8]==PNG_SIG and h[12:16]==b'IHDR':
            m.update(format='PNG',width=int.from_bytes(h[16:20],'big'),height=int.from_bytes(h[20:24],'big'),bit_depth=h[24],colour_type=h[25]);return m
        from PIL import Image
        with Image.open(p) as im:m.update(format=im.format,width=im.size[0],height=im.size[1],mode=im.mode)
    except Exception as ex:m['unreadable']=repr(ex)[:200]
    return m
def _a6_sig(m):return None if m.get('missing') or m.get('unreadable') else tuple(m.get(k) for k in A6_KEYS)
def colour_condition(l,si,names,readings,flats):
    def meta(x):return colour_file_meta(readings[x]['session'],flats[x].get(f'labels/{si}/image') or f'Actual_Frames/frame_{si:05d}.png')
    own=meta(l);comp={m:meta(m) for m in names if m!=l};why=[];so=_a6_sig(own)
    if so is None:why.append(f"{l}: colour file of si {si} {'missing' if own.get('missing') else 'unreadable'} ({own['file']})")
    for m,v in comp.items():
        sv=_a6_sig(v)
        if sv is None:why.append(f"{m} (comparison): colour file of si {si} {'missing' if v.get('missing') else 'unreadable'} ({v['file']})")
        elif so is not None and sv!=so:why.append(f'{l} vs {m}: colour file of si {si} differs in dimensions/pixel format '+json.dumps({k:[own.get(k),v.get(k)] for k in A6_KEYS if own.get(k)!=v.get(k)}))
    return dict(leg=l,si=si,holds=not why,colour=own,comparison=comp,why=why,ruling=A6_RULING+' (A.6)')

def peak_census_check(s,text):
    errs=[];peak=s.get('change_bytes_peak_census');hw=s.get('change_bytes_high_water')
    if hw==0 and peak is None:return [],None
    if not isinstance(peak,dict):return ['peak census absent'],None

    if peak.get('bytes_held')!=peak.get('high_water'):errs.append('peak.bytes_held != peak.high_water')
    errs+=census_entry_errors(peak,'peak')
    errs+=peak_lines_errors(s,text,peak)

def peak_lines_errors(s,text,peak):
    errs=[];hw=s.get('change_bytes_high_water');resets=s.get('change_epoch_resets',0)
    lines=re.findall(r'HIGH-WATER-PEAK bytes=(\d+) max=(\d+) census=(\{.*\})',text)
    want=1+resets if isinstance(resets,int) and resets>=0 else None
    if want is None:return [f'change_epoch_resets {resets!r} not a count']
    if len(lines)!=want:return [f'HIGH-WATER-PEAK lines={len(lines)} (want 1 + change_epoch_resets = {want})']
    for i,(b,mx,cj) in enumerate(lines):
        cz=json.loads(cj);tag=f'HIGH-WATER-PEAK line {i+1}/{want}'
        if cz.get('high_water')!=int(b) or cz.get('bytes_held')!=int(b):errs.append(f'{tag}: census not closed at its bytes')
        errs+=census_entry_errors(cz,tag)
    if int(lines[-1][0])!=hw or json.loads(lines[-1][2])!=peak:errs.append('HIGH-WATER-PEAK line disagrees with summary')
    return errs
def census_entry_errors(x,tag):
    e=[]
    if x.get('unaccounted_bytes')!=0:e.append(f'{tag}: unaccounted_bytes nonzero')
    if x.get('colours_unowned')!=0:e.append(f'{tag}: colours_unowned nonzero')
    if x.get('colours')!=sum(x.get(k,0) for k in ('colours_pending','colours_previous','colours_phase_ref','colours_in_hand','colours_unowned')):e.append(f'{tag}: colour split does not sum')
    if x.get('colour_bytes',0)+x.get('mask_bytes',0)+x.get('unaccounted_bytes',0)!=x.get('bytes_held'):e.append(f'{tag}: bytes do not sum')
    if x.get('bytes_held',0)>x.get('max_bytes',0):e.append(f'{tag}: bytes_held above max_bytes')
    for k in ('cursor','latest_index','open_phases','head'):
        if k not in x:e.append(f'{tag}: {k} absent')
    if 'head' in x and not ({'si','state','waits_on'}<=set(x['head']) or {'si','state'}<=set(x['head'])):e.append(f'{tag}: head incomplete')
    return e
def refusal_census_check(records,text):
    errs=[];lines=re.findall(r'Capture\(m55\): BUDGET-EXCEEDED si=(-?\d+) kind=(\w+) requested=(\d+) held=(\d+) max=(\d+) census=(\{.*\})',text)
    entries=[(x['session_index'],y) for x in records for y in x.get('budget_census',[])]
    for si,kind,req,held,mx,cj in lines:
        cz=json.loads(cj)
        if cz.get('trigger')!='budget_exceeded' or cz.get('kind')!=kind or cz.get('si')!=int(si) or cz.get('bytes_held')!=int(held) or cz.get('max_bytes')!=int(mx):errs.append(f'BUDGET-EXCEEDED si={si}: line fields disagree with its census')
        errs+=census_entry_errors(cz,f'refusal si={si}')
    for si,y in entries:
        if y.get('trigger')!='budget_exceeded':errs.append(f'row si={si}: budget_census trigger {y.get("trigger")}')
        errs+=census_entry_errors(y,f'row si={si}')
    refused_rows=[x['session_index'] for x in records if x.get('reason')=='budget_exceeded']
    for si in refused_rows:
        if not any(e[0]==si for e in entries):errs.append(f'refused row si={si} carries no budget_census')
    if len(entries)>len(lines):errs.append(f'more row census entries ({len(entries)}) than BUDGET-EXCEEDED lines ({len(lines)})')
    return errs,dict(lines=len(lines),row_entries=len(entries),refused_rows=refused_rows)

def tag_bijection(rx,fx,ry,fy,mapkind):
    fwd={};rev={};why=[]
    def add(a,b,where):
        a=int(a);b=int(b)
        if fwd.setdefault(a,b)!=b:why.append(f'inconsistent across frames: {a}->{fwd[a]} and {a}->{b} ({where})')
        if rev.setdefault(b,a)!=a:why.append(f'many-to-one: {rev[b]}->{b} and {a}->{b} ({where})')
    add(0,0,'background')
    lp=[(p,fx[p],fy[p]) for p in fx if p.startswith('labels/') and p.endswith('/mask_value') and p in fy]
    for p,a,b in lp:add(a,b,p)
    tx_=rx.get('mask_tags') or {};ty_=ry.get('mask_tags') or {};common=sorted(set(tx_)&set(ty_))
    for n in common:
        if len(tx_[n]['order'])!=len(ty_[n]['order']):why.append(f'mask/{n}: tag partition differs');continue
        for a,b in zip(tx_[n]['order'],ty_[n]['order']):add(a,b,f'mask/{n}')

    if identity and not why:res['valid']=True;return res
    if why:return res
    table=bytearray(range(256))
    for a,b in fwd.items():table[a]=b
    for n in common:
        raw=_decode_mask(Path(rx['session'])/'target_mask'/n)
        if hashlib.sha256(raw.translate(bytes(table))).hexdigest()!=ty_[n]['pix_sha']:why.append(f'mask/{n}: pixels differ after remap')
        if {str(fwd.get(int(k),int(k))):v for k,v in tx_[n]['counts'].items()}!=ty_[n]['counts']:why.append(f'mask/{n}: per-tag pixel counts differ after remap')
    if mapkind=='CB' and rx.get('occupancy')!=ry.get('occupancy'):why.append('occupancy differs')
    tx_t=[list(t) for t in rx.get('mask_ties') or []];ty_t=[list(t) for t in ry.get('mask_ties') or []]
    if mapkind=='MAIN':tx_t=sorted([t[0],str(fwd.get(int(t[1]),int(t[1]))),*t[2:]] for t in tx_t)

    for a,s in ax.items():
        if ay.get(fwd.get(a,a))!=s:why.append(f'pi breaks actor identity in labels: {a}->{fwd.get(a,a)} ({sorted(map(str,s))} vs {sorted(map(str,ay.get(fwd.get(a,a),set())))})')

    def __init__(self,label,interval=PP_SAMPLE_S):

def _pp_prongs(span,streams,monitor,game,names,phases=()):
    lo,hi=span;force=[]
    inside=lambda x:x is None or lo<=x<=hi
    nongame=lambda s:'fg' in s and s.get('fg') not in game
    name=lambda s:s.get('fg_name') or names.get(str(s.get('fg'))) or ''
    streams=[[s for s in st if inside(s.get('x'))] for st in streams]
    for st in streams:
        force+=[(s['x']-PP_FORCE_S,s['x']+PP_FORCE_S) for s in st if s.get('x') is not None and nongame(s)]
    force+=list(phases)
    near=lambda a,b:any(f0<=b and a<=f1 for f0,f1 in force)
    shell=collections.Counter();moves=[];inputs=[];h_moves=0;h_inputs=0;n_rs=0
    for st in streams:
        n_rs+=len(st)
        for s in st:
            if nongame(s) and name(s).lower() in SHELL_PROCS:shell[name(s)]+=1
        prev=None
        for i,s in enumerate(st):
            if not s.get('cursor'):continue
            if prev is not None and st[prev]['cursor']!=s['cursor']:
                if s.get('x') is None:h=any(nongame(t) for t in st[max(0,i-PP_FORCE_SAMPLES):i+PP_FORCE_SAMPLES+1])
                else:h=near(st[prev]['x'] if st[prev].get('x') is not None else s['x'],s['x'])
                if h:h_moves+=1
                else:moves.append([st[prev]['cursor'],s['cursor']])
            prev=i
    tick=sorted((s for s in monitor if s.get('input_tick') is not None and s.get('x') is not None),key=lambda s:s['x'])
    for a,b in zip(tick,tick[1:]):
        if a['input_tick']==b['input_tick']:continue
        x=b['x']-b['idle_ms']/1000.0
        if not lo<=x<=hi:continue
        if near(x,x):h_inputs+=1
        else:inputs.append(round(x,3))
    reasons=[]
    if shell:reasons.append('foreground taken by a shell process: '+', '.join(f'{k} x{v}' for k,v in shell.items()))
    if moves:reasons.append(f'cursor moved while the harness was not moving it: {len(moves)} move(s), {moves[0][0]} -> {moves[-1][1]}')
    if inputs:reasons.append(f'non-harness input registered: {len(inputs)} input event(s) while the game held the foreground and no runner focus action was near')
    return dict(person_evidence=bool(reasons),reasons=reasons,prongs=dict(shell=dict(shell),cursor_moves=len(moves),cursor_moves_near_focus_action=h_moves,
                input_events=len(inputs),input_events_near_focus_action=h_inputs,samples_in_span=n_rs,monitor_samples=len(monitor)))

        status,errors=lyra_fixture(launch,text,run.returncode);row.update(fixture_status=status,fixture_errors=errors)
        if status=='HARNESS-INVALID':row['status']=status;ledger.append(row);ledger_save(ledger_name,ledger);raise Stop(3,f'HARNESS-INVALID {label}')
        if status!='FIXTURE-VALID':
            row['status']=pp_invalid_status(status,errors,pe)
            if row['status']!=status:row.update(original_status=status,counts_toward_budget=False)
            ledger.append(row);ledger_save(ledger_name,ledger);print(json.dumps({k:row[k] for k in ('label','status','fixture_errors','person_evidence')}),flush=True)
            if row['status']=='ENV-INTERRUPTED':pp_after_interrupt(leg,[x for x in ledger if x['leg']==leg],label,pe)
            continue
        ev=lyra_g270_eval(launch['session'],d/'game.log',outdir=d) if kind=='g270' else lyra_twin_eval(launch['session'],d/'game.log',kind,outdir=d)
        (d/'measurement-audit.json').write_text(json.dumps(ev,indent=2,default=str))
        row['status']=ev['branch'] if kind=='g270' else ev['status'];row['result']={k:v for k,v in ev.items() if k!='onsets'}
        if row['status'] in ('NEEDS-DECISION','FAIL') and pe['person_evidence']:row['needs_decision']=PP_ND_NOTE
        ledger.append(row);ledger_save(ledger_name,ledger)
        print(json.dumps({k:row.get(k) for k in ('label','status','result','person_evidence')},default=str),flush=True)
        if row['status'] in ('NEEDS-DECISION','FAIL'):raise Stop(1,f'Lyra {leg} {row["status"]}'+pp_flag(pe))
        if pe['person_evidence']:c.log_event('person-evidence-on-valid-attempt',leg=leg,label=label,status=row['status'],reasons=pe['reasons'])
        return

def run_sequence(steps,summary_name):
    out=dict(started=c.now(),steps=[]);code=0;msg='complete'
    try:
        for s in steps:
            s();out['steps'].append(getattr(s,'__name__','step'))
    except Stop as ex:code=ex.code;msg=ex.msg
    out.update(finished=c.now(),exit=code,message=msg);(c.E/summary_name).write_text(json.dumps(out,indent=2))
    print(json.dumps(dict(exit=code,message=msg)),flush=True);return code
