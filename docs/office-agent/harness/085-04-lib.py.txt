def alias_hardlinks(accepted,alias,bank=None):

class PersonMonitor:
    def __init__(self,label,interval=PP_SAMPLE_S):
        self.label=label;self.interval=interval;self.samples=[];self._stop=threading.Event();self._th=None
    def _loop(self):
        while True:
            try:self.samples.append(_desktop_sample())
            except Exception:pass
            if self._stop.wait(self.interval):return

        self._th=threading.Thread(target=self._loop,daemon=True);self._th.start();return self
    def __exit__(self,*a):

def person_evidence(launch,mon):
    if not launch.get('sampling_started_utc'):return dict(person_evidence=False,reasons=[],note='no steady-state span')
    lo=_epoch(launch['sampling_started_utc']);hi=_epoch((launch.get('anchor_end') or {}).get('utc') or launch.get('finished_utc') or launch['sampling_started_utc'])
    pid=launch.get('pid');ms=[x for x in mon.samples if lo<=x['t']<=hi];reasons=[]
    ticks=[x for x in ms if x.get('input_tick') is not None]
    inputs=sum(1 for a,b in zip(ticks,ticks[1:]) if a['input_tick']!=b['input_tick'] and a.get('fg_pid')==pid and b.get('fg_pid')==pid)
    foreign_fg=sum(1 for x in ms if x.get('fg_pid') not in (pid,None))
    if inputs:reasons.append(f'non-harness input while the game held focus: {inputs}')
    if foreign_fg>PP_FORCE_SAMPLES:reasons.append(f'foreground left the game for {foreign_fg} samples')
    return dict(person_evidence=bool(reasons),reasons=reasons,samples=len(ms))

def resolve(L,lk=None):
    """Substitute the premise-lock values: @H1/@H2 targets, @CAP (the declared bench-only cap raise, else nothing), the DEFER
    counts. A leg that needs a lock value that does not exist is NOT runnable (returns None, reason)."""
    L=json.loads(json.dumps(L));lk=lk if lk is not None else lock()
    need=bool(L.get('premise')) and (str(L.get('target') or '').startswith('@') or any(str(x).startswith('@') for x in L['levers']+L['post_cmds']) or L.get('runs'))
    if need and not lk:return None,'no premise lock'
    if str(L.get('target') or '').startswith('@'):
        t=(lk or {}).get(L['target'][1:])
        if not t:return None,f'premise named no {L["target"][1:]}'
        L['target']=t
    lv=[]
    for x in L['levers']:
        if x=='@CAP':
            if lk and lk.get('cap_declared') and lk.get('cap_bytes'):lv.append(f'IAI.Anomaly.TexCorruptMaxRtBytes {int(lk["cap_bytes"])}')
        else:lv.append(x)
    L['levers']=lv
    pc=[]
    for x in L['post_cmds']:
        m=re.match(r'@DEFER_(\w+) (.*)',x)
        if x=='@DEFERPROBES':
            pc+=[defer(n,cv) for n,cv in PROBES]
        elif m:
            key='defer_'+m.group(1).lower()
            if not lk or key not in lk:return None,f'premise lock has no {key}'
            pc.append(defer(lk[key],m.group(2)))
        else:pc.append(x)
    L['post_cmds']=pc
    if L.get('runs'):
        d=L['runs'].get('defer')
        if isinstance(d,str) and d.startswith('@'):
            key=d[1:].lower()
            if not lk or key not in lk:return None,f'premise lock has no {key}'
            L['runs']['defer']=int(lk[key])
            if key=='excl_a_defer' and lk.get('excl_a_pre0'):L['runs']['cfg2']='2 0 8 14 0'
    return L,None

def validity(L,launch,sess,log,extra_sessions=()):
    errs=[]
    if launch.get('status')!='COMPLETE':errs.append('protocol: '+str(launch.get('error')))
    if L['kind']=='census':
        n=len(re.findall(re.escape(L['wait_pattern']),log or ''))
        want=L.get('wait_count',1)
        if n<want:errs.append(f'artifact: {n} "{L["wait_pattern"]}" lines < {want}')
        return errs+bench_flag_errors(L,log)
    fa=launch.get('focus_at_s')
    if fa is None or fa<0 or fa>=20.0:errs.append(f'A63: focus not held within 20 s (focus_at_s={fa})')
    cf=launch.get('capture_focus') or [];cf=[cf] if isinstance(cf,dict) else cf
    if cf and not all(x.get('foreground_pid')==launch.get('pid') for x in cf):errs.append('whole-leg foreground failed')
    if sess is None:errs.append('artifact: no session with run_summary');return errs
    rows=[l for l in (sess/'labels.jsonl').read_text(encoding='utf-8-sig').splitlines() if l.strip()] if (sess/'labels.jsonl').exists() else []
    n=len(rows);pngs=list((sess/'Actual_Frames').glob('*.png'))
    if n!=L['frames']:errs.append(f'artifact: {n} label rows != {L["frames"]} frames')
    if len(pngs)!=n:errs.append(f'artifact: {len(pngs)} PNGs != {n} rows')
    if any(p.stat().st_size==0 for p in pngs):errs.append('artifact: zero-byte PNG')
    if L.get('runs'):
        r1=L['runs']['first']
        if not extra_sessions or extra_sessions[0] is None:errs.append('artifact: run 1 left no session')
        else:
            n1=sum(1 for l in (extra_sessions[0]/'labels.jsonl').read_text(encoding='utf-8-sig').splitlines() if l.strip())
            if n1!=r1['frames']:errs.append(f'artifact: run 1 {n1} rows != {r1["frames"]}')
        if log and re.search(r'IAI\.Capture\.Start.*(ignored|refused).*mid-run',log,re.I):errs.append('protocol: run 2 start was refused mid-run (DEFER too short)')
    return errs+bench_flag_errors(L,log)

def copy_retry(src,dst,tries=None,backoff=None,sleep=time.sleep,log=None):
    tries=tries or c.RESTORE_TRIES;backoff=backoff or c.RESTORE_BACKOFF_S;errs=[]
    for i in range(tries):
        try:
            shutil.copy2(src,dst)
            if log is not None:log.append(dict(dst=str(dst),attempt=i+1,ok=True,errors=errs[:]))
            return i+1
        except Exception as ex:
            errs.append(f'{type(ex).__name__}: {ex}')
            if i+1>=tries:
                if log is not None:log.append(dict(dst=str(dst),attempt=i+1,ok=False,errors=errs[:]))
                raise Stop(11,f'copy {src} -> {dst} failed {tries} times: {errs[-1]}')
            sleep(backoff[min(i,len(backoff)-1)])
def hold_exclusive(path,seconds):
    ready=threading.Event();done=threading.Event();info={}
    def run():
        h=_K32.CreateFileW(str(path),0x80000000,0,None,3,0x80,None)
        info['handle_ok']=h not in (None,ctypes.c_void_p(-1).value) and h!=-1 and h!=0xFFFFFFFFFFFFFFFF
        info['err']=ctypes.get_last_error();ready.set();time.sleep(seconds)
        if info['handle_ok']:_K32.CloseHandle(h)
        done.set()
    threading.Thread(target=run,daemon=True).start();ready.wait(5);return info,done
def _stub_sources():
    root=c.RUN/'_stub_sets';want={}
    for name in ('M53','DC','FIX'):
        s={}
        for k in c.KEYS:
            p=root/name/k;p.parent.mkdir(parents=True,exist_ok=True)
            if not p.exists():p.write_bytes((f'{name}|{k}|stub bytes for the staging and G355 restore proof|'*64).encode())
            s[k]=p
        want[name]=(s,{k:c.sha(s[k]) for k in c.KEYS})
    return want
def stage(name,tag,restore=False,stub_start='FIX'):
    """Stage a pinned pairing (only the files that differ), hash-verified, with a receipt. It never assumes what was
    staged before: the before-state is read and recorded, and any of M53 / FIX / DC / unknown is accepted as a start."""
    rec=dict(set=name,tag=tag,utc=c.now(),stub=c.STUB);copies=[]
    if c.STUB:
        sets=_stub_sources();src,want=sets[name];root=c.RUN/'_stub_staged'
        if not (root/c.KEYS[0]).exists():
            for k in c.KEYS:(root/k).parent.mkdir(parents=True,exist_ok=True);shutil.copy2(sets[stub_start][0][k],root/k)
        before={k:c.sha(root/k) for k in c.KEYS};lock_=None
        rec['before_set']=next((n for n,(s,w) in sets.items() if w==before),None)
        inj=float(os.environ.get('IAI_R854_INJECT_LOCK_S','0') or 0)
        if restore and inj>0:
            info,done=hold_exclusive(root/c.KEYS[0],inj);lock_=dict(file=c.KEYS[0],seconds=inj,**info);rec['injected_lock']=lock_
        for k in c.KEYS:
            if before[k]==want[k]:continue
            copy_retry(src[k],root/k,log=copies)
        after={k:c.sha(root/k) for k in c.KEYS}
        rec.update(before=before,after=after,copies=copies,verified=(after==want),note='STUB: staged into the dry root, real copies, real retry path')
        jsave(c.RUN/f'receipt-{tag}.json',rec)
        if after!=want:raise Stop(10,f'stub staging {name} did not verify')
        return rec
    want=c.verify_set(name)
    q=c.ours_only_quiet() if restore else c.quiet()
    if not q['quiet']:raise Stop(4,f'NOT-QUIET at staging {name}: {q["blocking"]}')
    rec['foreign_at_staging']=q.get('foreign')
    before=c.staged_state();src=c.sources(name);rec['before_set']=c.which_set(before)
    for k in c.KEYS:
        if before[k]==want[k]:continue
        if c.sha(src[k])!=want[k]:raise Stop(10,f'source for {k} does not match {name}: {src[k]}')
        copy_retry(src[k],c.STAGED_ROOT/k,log=copies)
    after=c.staged_state()
    rec.update(before=before,after=after,copies=copies,verified=(after==want))
    jsave(c.RUN/f'receipt-{tag}.json',rec)
    if after!=want:raise Stop(10,f'staging {name} did not verify: {[k for k in c.KEYS if after[k]!=want[k]]}')
    return rec
def restore_m53(tag='99-restore-m53',tries=None,backoff=None,sleep=time.sleep):
    tries=tries or c.RESTORE_TRIES;backoff=backoff or c.RESTORE_BACKOFF_S;hist=[]
    for i in range(tries):
        try:
            r=stage('M53',tag,restore=True);r['restore_attempts']=i+1;r['restore_history']=hist;jsave(c.RUN/f'receipt-{tag}.json',r);return r
        except BaseException as ex:
            if isinstance(ex,KeyboardInterrupt):raise
            hist.append(dict(attempt=i+1,utc=c.now(),error=f'{type(ex).__name__}: {getattr(ex,"msg",ex)}'))
            c.log_event('restore-retry',attempt=i+1,error=hist[-1]['error'])
            if i+1>=tries:
                jsave(c.RUN/f'receipt-{tag}-FAILED.json',dict(history=hist));raise Stop(12,f'RESTORE FAILED after {tries} attempts: {hist[-1]["error"]}')
            sleep(backoff[min(i,len(backoff)-1)])
