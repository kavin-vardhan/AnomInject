def read(path):
    return json.loads(path.read_text(encoding="utf-8"))


def link(label, path):
    return f"[{label}]({path.as_posix()})"


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def table(headers, rows):
    def row(values):
        return "| " + " | ".join(str(v).replace("|", " / ").replace("\n", " ") for v in values) + " |"
    return "\n".join([row(headers), row(["---"] * len(headers)), *(row(r) for r in rows)])


def event_tokens(case):
    return ", ".join(s.split(" runs{")[0].split()[-1] for s in case["lines"] if s.startswith("idx="))


def runs(case):
    return ", ".join(e["run_outcome"] for e in case["detail"] if e["kind"] == "onset")


def edges(case):
    values = []
    for e in case["detail"]:
        if e["observation"] == "TRANSITION":
            values.append(f"T{e['best_k']}({e['offset']:+d})")
        elif e["observation"] == "NO-TRANSITION":
            values.append("N")
        else:
            values.append("U(" + e["reason"] + ")")
    return "; ".join(values)
