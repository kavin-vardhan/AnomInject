def _ft(f):return (f.dwHighDateTime<<32)|f.dwLowDateTime
def snapshot_processes():
    h=K32.CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS,0);out={}
    if not h or h==INVALID_HANDLE:return out
    try:
        e=PROCESSENTRY32W();e.dwSize=ctypes.sizeof(e);ok=K32.Process32FirstW(h,ctypes.byref(e))
        while ok:
            out[int(e.th32ProcessID)]=(e.szExeFile,int(e.th32ParentProcessID));ok=K32.Process32NextW(h,ctypes.byref(e))
    finally:K32.CloseHandle(h)
    return out
def threads_of(pid):
    h=K32.CreateToolhelp32Snapshot(TH32CS_SNAPTHREAD,0);out=[]
    if not h or h==INVALID_HANDLE:return out
    try:
        e=THREADENTRY32();e.dwSize=ctypes.sizeof(e);ok=K32.Thread32First(h,ctypes.byref(e))
        while ok:
            if e.th32OwnerProcessID==pid:out.append(int(e.th32ThreadID))
            ok=K32.Thread32Next(h,ctypes.byref(e))
    finally:K32.CloseHandle(h)
    return out
def thread_description(h):
    p=ctypes.c_wchar_p()
    if K32.GetThreadDescription(h,ctypes.byref(p))<0:return None
    try:return p.value
    finally:K32.LocalFree(ctypes.cast(p,ctypes.c_void_p))
def thread_cycles(h):

def thread_times(h):
    f=[W.FILETIME() for _ in range(4)]
    return (_ft(f[0]),_ft(f[2]),_ft(f[3])) if K32.GetThreadTimes(h,*[ctypes.byref(x) for x in f]) else None
def process_sample(pid):
    h=K32.OpenProcess(PROCESS_QUERY_LIMITED,False,pid)
    if not h:return None
    try:
        f=[W.FILETIME() for _ in range(4)];io=IO_COUNTERS()
        cpu=_ft(f[2])+_ft(f[3]) if K32.GetProcessTimes(h,*[ctypes.byref(x) for x in f]) else None
        rw=(int(io.ReadTransferCount),int(io.WriteTransferCount)) if K32.GetProcessIoCounters(h,ctypes.byref(io)) else (None,None)
        return [cpu,rw[0],rw[1],_ft(f[0])]
    finally:K32.CloseHandle(h)

def calibrate_cycles_per_s(trials=5,span=0.2):
    h=K32.GetCurrentThread();rates=[]
    for _ in range(trials):
        c0=thread_cycles(h);t0=time.perf_counter()
        while time.perf_counter()-t0<span:pass
        c1=thread_cycles(h);t1=time.perf_counter();rates.append((c1-c0)/(t1-t0))
    mhz=registry_mhz();best=max(rates)
    return dict(cycles_per_s=best,trials=[round(r) for r in rates],registry_mhz=mhz,ratio_to_registry=round(best/(mhz*1e6),5) if mhz else None)

class ThreadSampler:
    def __init__(self,interval=0.1):
        self.interval=interval;self.pid=None;self.samples=[];self.desc={};self.created={};self.errors=[];self._h={};self._stop=threading.Event();self._th=None;self._last_enum=0.0
    def set_pid(self,pid):self.pid=pid
    def _enum(self):
        for tid in threads_of(self.pid):
            if tid in self._h:
                if not self.desc.get(tid):self.desc[tid]=thread_description(self._h[tid])
                continue
            h=K32.OpenThread(THREAD_QUERY_LIMITED,False,tid)
            if not h:continue
            self._h[tid]=h;self.desc[tid]=thread_description(h);tt=thread_times(h);self.created[tid]=tt[0] if tt else None
    def _loop(self):
        while not self._stop.is_set():
            if self.pid:
                try:
                    now=time.perf_counter()
                    if now-self._last_enum>=(0.25 if 'GameThread' not in self.desc.values() else 1.0) or not self._h:self._enum();self._last_enum=now
                    cyc={};gt=None
                    for tid,h in self._h.items():
                        v=thread_cycles(h)
                        if v is not None:cyc[tid]=v
                        if self.desc.get(tid)=='GameThread':gt=thread_times(h)
                    self.samples.append(dict(t=time.perf_counter(),cyc=cyc,gt_times=gt))
                except Exception as ex:
                    if len(self.errors)<5:self.errors.append(repr(ex)[:200])
            self._stop.wait(self.interval)
    def start(self):self._th=threading.Thread(target=self._loop,name='c37-threads',daemon=True);self._th.start();return self
    def stop(self):
        self._stop.set()
        if self._th:self._th.join(5)
        for h in self._h.values():K32.CloseHandle(h)
        self._h={}
    def data(self):
        return dict(pid=self.pid,interval_s=self.interval,desc={str(k):v for k,v in self.desc.items()},created={str(k):v for k,v in self.created.items()},
                    samples=[dict(t=s['t'],cyc={str(k):v for k,v in s['cyc'].items()},gt_times=s['gt_times']) for s in self.samples],errors=self.errors)
class LoadSampler:
    def __init__(self,interval=1.0,root_pid=None):
        self.interval=interval;self.root=root_pid or os.getpid();self.samples=[];self.names={};self.errors=[];self._stop=threading.Event();self._th=None;self.pdh=None
    def _one(self):
        snap=snapshot_processes();procs={}
        for pid,(name,ppid) in snap.items():
            if pid in (0,):continue
            self.names.setdefault(pid,(name,ppid))
            v=process_sample(pid)
            if v:procs[pid]=v
        pd=self.pdh.read() if self.pdh else {}
        gpu=gpu_by_pid(pd.get('\\GPU Engine(*)\\Utilization Percentage'))
        disk={k.split('\\')[-1]:list(v.values())[0] if v else None for k,v in pd.items() if 'PhysicalDisk' in k}
        return dict(t=time.perf_counter(),utc=time.time(),sys=system_times(),procs=procs,parents={p:v[1] for p,v in snap.items()},gpu3d=gpu,disk=disk)
    def _loop(self):
        try:self.pdh=Pdh()
        except Exception as ex:self.errors.append('pdh: '+repr(ex)[:200])
        while not self._stop.is_set():
            try:self.samples.append(self._one())
            except Exception as ex:
                if len(self.errors)<5:self.errors.append(repr(ex)[:200])
            self._stop.wait(self.interval)
        if self.pdh:self.pdh.close()
    def start(self):self._th=threading.Thread(target=self._loop,name='c37-load',daemon=True);self._th.start();return self
    def stop(self):
        self._stop.set()
        if self._th:self._th.join(10)
    def data(self):
        return dict(root_pid=self.root,interval_s=self.interval,names={str(k):v[0] for k,v in self.names.items()},pdh_errors=(self.pdh.errors if self.pdh else ['pdh not opened']),
                    samples=[dict(t=s['t'],utc=s['utc'],sys=s['sys'],procs={str(k):v for k,v in s['procs'].items()},parents={str(k):v for k,v in s['parents'].items()},
                                  gpu3d={str(k):v for k,v in s['gpu3d'].items()},disk=s['disk']) for s in self.samples],errors=self.errors)
def find_child(parent_pid,image,timeout_s=0.0):
    t0=time.time()
    while True:
        for pid,(name,ppid) in snapshot_processes().items():
            if ppid==parent_pid and name.lower()==image.lower():return pid
        if time.time()-t0>=timeout_s:return None
        time.sleep(0.2)
