def resolve(label, spec):
    s = json.loads(json.dumps(spec))
    rep = {'{RUNS}': str(h.WORK / 'runs'), '{LABEL}': label, '{OUT}': str(h.OUT / label)}
    def fix(v):
        if isinstance(v, str):
            for k, r in rep.items():
                v = v.replace(k, r)
            return v
        if isinstance(v, dict):
            return {k: fix(x) for k, x in v.items()}
        if isinstance(v, list):
            return [fix(x) for x in v]
        return v
    return fix(s)


if __name__ == '__main__':

    L = legs(build)
    for name in sys.argv[2:]:
        base = name.split('__')[0]
        label = f'{base}_{build}' + (('__' + name.split('__')[1]) if '__' in name else '')
        spec = resolve(label, L[base])
        rec = h.run(label, spec)
        print(json.dumps(h.brief(rec), default=str), flush=True)
