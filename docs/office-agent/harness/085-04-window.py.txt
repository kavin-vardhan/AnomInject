def probe_offset(log):
    """S = the GFrameCounter at which the ExecCmds batch ran: each DEFER probe prints its cvar at (S + count); the log
    prefix gives GFrameCounter % 1000. Returns (S, readings) or (None, readings)."""
    got=[]
    for n,cv in L.PROBES:
        for ln in (log or '').splitlines():
            if re.search(re.escape(cv)+r' = ',ln):
                f=EV.frame_of(ln)
                if f is not None:got.append((n,cv,f,f-n));break
    offs=sorted(set(x[3] for x in got))
    return (offs[0] if len(offs)==1 else (min(offs) if offs and max(offs)-min(offs)<=2 else None)),got
def pick_hosts(per,pref=None,cap_limit=1<<30):
    """P-HOST under ND-4. Per family (uv / normal): H1, H2 = the actors with final=APPLY at the delivered 128 MiB, an actor eligible for
    BOTH families first (the plan's host), then the rock, then name order. If a family has none at 128 MiB, the actors whose only
    refusal is over_budget are eligible under a raise: the DECLARED bench-only cap is the next power of two >= the chosen hosts'
    required_bytes (<= 1 GiB). A family with no host at all leaves its MainWorld legs NOT-RUN-PREMISE (a shortfall for chat)."""
    pref=pref or [LG.ROCK,LG.ROCK2]
    fams={'U':'uv','N':'normal'}
    both=[a for a,v in per.items() if all((v.get(f) or {}).get('final')=='APPLY' for f in fams.values())]
    def rank(a):return (0 if a in both else 1,pref.index(a) if a in pref else len(pref),a)
    out=dict(cap_declared=False,cap_bytes=None,eligible_both_at_128=sorted(both,key=rank));need=[];notes=[]
    for k,f in fams.items():
        el=sorted([a for a,v in per.items() if (v.get(f) or {}).get('final')=='APPLY'],key=rank)
        out[f'eligible_{f}_at_128']=el
        if el:out[f'H1{k}']=el[0];out[f'H2{k}']=el[1] if len(el)>1 else None;continue
        ob=sorted([(rank(a),v[f]['required'],a) for a,v in per.items() if (v.get(f) or {}).get('final')=='over_budget' and 0<v[f]['required']<=cap_limit])
        if ob:
            out[f'H1{k}']=ob[0][2];out[f'H2{k}']=ob[1][2] if len(ob)>1 else None;need+=[x[1] for x in ob[:2]]
            notes.append(f'{f}: none at 128 MiB, over_budget-only hosts {ob[0][2]} ({ob[0][1]} B)')
        else:
            out[f'H1{k}']=None;out[f'H2{k}']=None;notes.append(f'{f}: NO host at any cap <= 1 GiB (refusals are not budget): {f} MainWorld legs NOT-RUN-PREMISE (chat)')
    if need:
        cap=1<<max(20,math.ceil(math.log2(max(need))))
        out.update(cap_declared=True,cap_bytes=cap)
        nd4=f'bench-only cap raise DECLARED before any gate leg: {cap} B on the MainWorld and COST legs; the delivered default stays 134217728. '+'; '.join(notes)
    else:nd4='no cap raise: every family with a host has one eligible at the delivered 128 MiB. '+'; '.join(notes)
    same=out.get('H1U') and out.get('H1U')==out.get('H1N')
    out['_nd4']=nd4
    out['_hosts']=(f"H1 uv {out.get('H1U')} / normal {out.get('H1N')} ({'one host for both families' if same else 'PER-FAMILY hosts: no MainWorld actor is eligible for both families - flagged for chat'}); "
                   f"H2 uv {out.get('H2U')} / normal {out.get('H2N')}")
    return out

def lock_step():

    draft=c.RUN/'premise-amendment-draft.md'

def bm53_summary(lg):
    per=collections.defaultdict(lambda:dict(onsets=0,offsets=0,fails=0,legs=[]))
    for k,v in lg.items():
        if not isinstance(v,dict) or v.get('group') not in ('BM53F','BM53MW') or v.get('role')=='NULL' or not v.get('edges'):continue
        key=f"{v['group']}:{v.get('mode')}:{k.split('_')[-1] if not k.endswith('AAOFF') else 'NAT_AAOFF'}"
        e=v['edges'];p=per[key];p['onsets']+=e['onsets'];p['offsets']+=e['offsets'];p['fails']+=len(e['fails']);p['legs'].append(k)
    out={}
    for key,p in per.items():
        floor=16 if key.startswith('BM53F') else 8
        out[key]=dict(p,floor=floor,verdict=('FAIL' if p['fails'] else ('PASS' if p['onsets']>=floor and p['offsets']>=floor else 'INSUFFICIENT')))
    legv={k:v.get('m53_verdict') for k,v in lg.items() if isinstance(v,dict) and v.get('group') in ('BM53F','BM53MW')}
    return dict(per_mode_condition=out,leg_verdicts=legv,rule='t50: 0 off frames at onset and offset; >= 16 judged per mode per AA condition (fixture), >= 8 (MainWorld H1), fewer = INSUFFICIENT (to chat), never PASS')

def mode_gates(sess):

    out['verdict']='PASS' if vs and all(v=='PASS' for v in vs) and nulls_ok else ('FAIL' if any(v=='FAIL' for v in vs) else 'INCOMPLETE')
    return out

def postflight():
    problems=[]
    def chk(ok,msg):
        if not ok:problems.append(msg)
    ours,foreign=c.classify()
    chk(not [p for p in ours if p['name']!='UnrealTraceServer.exe'],'our engine/game processes present at close')
    st=c.staged_state();chk(st==c.M53,'staged set is not the m53 S1 set: '+json.dumps(st))
    chk(c.git(c.RP,'rev-parse','HEAD')==c.M51_HEAD and c.git(c.RP,'branch','--show-current')=='m51','main checkout moved off m51 53bf725')
    led=L.ledger();acc=[x for x in led if x.get('status')=='ACCEPTED'];links=0
    for x in acc:
        a=c.BANK/x['label'];al=c.BANK/(c.BANK_PREFIX+x['leg'])
        for p in [p for p in a.rglob('*') if p.is_file()]:
            q=al/p.relative_to(a)
            if not q.exists() or os.stat(q).st_ino!=os.stat(p).st_ino:problems.append(f'alias not a hardlink: {q}');break
            links+=1
    left=[d.name for d in c.EXE_SIDE.iterdir() if d.is_dir() and d.name.startswith(c.BANK_PREFIX)] if c.EXE_SIDE.exists() else []
    chk(not left,f'leg copies left after banking: {left[:5]}')
    out=dict(utc=c.now(),problems=problems,accepted_legs=len(acc),attempts=len([x for x in led if x.get('label')]),not_accepted=[x for x in led if x.get('status')!='ACCEPTED'],
             alias_hardlinks_verified=links,left=left,staged=c.which_set(st),foreign_present=[(p['pid'],p['name']) for p in foreign],verdict='PASS' if not problems else 'FAIL')
    L.jsave(c.RUN/'postflight.json',out);return out
def window(phases):
    code=0;why=None;t0=time.time();restore=None;staged=None;lk=None;stage_start='FIX'
    try:
        staged=L.stage('DC','01-dc',stub_start=stage_start)
        if 'premise' in phases:
            run_groups(PREMISE_GROUPS);lk=build_lock()
            if 'lock' in phases:lock_step()
        if 'live' in phases:
            pf=preflight(need_lock=True)
            if pf['problems']:raise L.Stop(10,'gate preflight: '+'; '.join(pf['problems']))
            run_groups(GATE_GROUPS)
    except BaseException as ex:
        if isinstance(ex,KeyboardInterrupt):why='interrupted'
        code=getattr(ex,'code',99) if isinstance(ex,L.Stop) else 99;why=(why or '')+f' {type(ex).__name__}: {getattr(ex,"msg",ex)}';c.log_event('window-stop',code=code,why=why)
    finally:
        if 'live' in phases or code!=0:
            try:restore=L.restore_m53()
            except BaseException as ex:
                code=code or 12;why=(why or '')+' | RESTORE FAILED: '+str(getattr(ex,'msg',ex));c.log_event('restore-failed',why=why)
    rep=evaluate();elapsed=round(time.time()-t0,1)
    pf2=postflight() if (not c.STUB and 'live' in phases) else dict(verdict='SKIPPED (dry or premise-only: DC stays staged for the gate phase)')
    led=L.ledger()
    summ=dict(result='COMPLETE' if code==0 else 'STOPPED',code=code,why=why,elapsed_s=elapsed,run=str(c.RUN),postflight=pf2.get('verdict'),
              staged=dict(before=(staged or {}).get('before_set'),receipt=str(c.RUN/'receipt-01-dc.json')),
              restore=dict(attempts=(restore or {}).get('restore_attempts'),copies=[(Path(x['dst']).name,x['attempt']) for x in (restore or {}).get('copies',[])],injected_lock=(restore or {}).get('injected_lock')),
              launches=len([x for x in led if x.get('label')]),accepted=len([x for x in led if x.get('status')=='ACCEPTED']),
              not_accepted=[(x['leg'],x['status'],(x.get('reason') or '')[:120]) for x in led if x.get('status')!='ACCEPTED'],
              label_rule_all_pass=rep.get('label_rule_all_pass'),gates={k:(v.get('verdict') if isinstance(v,dict) and 'verdict' in v else None) for k,v in rep['gates'].items()})
    L.jsave(c.RUN/'summary.json',summ);print(json.dumps(summ,default=str)[:8000]);return code
