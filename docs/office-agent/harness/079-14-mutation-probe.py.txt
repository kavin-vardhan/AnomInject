def run(name, old, new):
    d = os.path.join(WORK, name)
    if os.path.isdir(d):
        shutil.rmtree(d)
    os.makedirs(d)
    for f in ("verify_capture.py", "measure_label_offset.py",
              "test_verify_capture_consistency.py"):
        shutil.copy2(os.path.join(SRC, f), os.path.join(d, f))
    applied = "control"
    if old is not None:
        p = os.path.join(d, "verify_capture.py")
        text = open(p, "r", encoding="utf-8").read()
        if old not in text:
            return name, "PATTERN-NOT-FOUND", "-", "-"
        text = text.replace(old, new, 1)
        open(p, "w", encoding="utf-8", newline="").write(text)
        applied = "mutated"
    st = subprocess.run([sys.executable, "-B", os.path.join(d, "verify_capture.py"),
                         "--label-pixel-gate", "--selftest"],
                        capture_output=True, text=True)
    ct = subprocess.run([sys.executable, "-B",
                         os.path.join(d, "test_verify_capture_consistency.py")],
                        capture_output=True, text=True)
    return name, applied, st.returncode, ct.returncode
