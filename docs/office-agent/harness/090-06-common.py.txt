    b=pin()['binaries'][name];p=Path(b['path']);out=[]

def gpu_pmon():
    """Per-process SM utilisation from nvidia-smi pmon (two 1 s samples, the minimum kept). None when nvidia-smi is unavailable (then only names classify)."""
    exe=shutil.which('nvidia-smi') or r'C:\Windows\system32\nvidia-smi.exe'
    if not Path(exe).exists():return None
    try:out=subprocess.run([exe,'pmon','-c','2','-s','u'],capture_output=True,text=True,timeout=30).stdout
    except Exception as ex:return dict(error=f'{type(ex).__name__}: {ex}')
    per={}
    for ln in out.splitlines():
        if ln.startswith('#') or not ln.strip():continue
        f=ln.split()
        if len(f)<4 or not f[1].isdigit():continue
        sm=int(f[3]) if f[3].isdigit() else 0
        p=per.setdefault(int(f[1]),dict(pid=int(f[1]),type=f[2],samples=[],name=f[-1]));p['samples'].append(sm)
    rows=[dict(p,sm=min(p['samples'])) for p in per.values()]
    return dict(rows=rows,rule='sustained = the MINIMUM SM % over two 1 s samples')
