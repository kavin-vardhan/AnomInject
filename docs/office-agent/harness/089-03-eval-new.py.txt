def gt_align(s,null,T,NT):
    import bisect
    nr=sorted((null.by_si[si]['t'],NT[si]) for si in NT if si in null.by_si and null.by_si[si].get('t') is not None)
    nt=np.array([x[0] for x in nr],dtype=np.float64);nv=np.array([x[1] for x in nr],dtype=np.float64)
    out=dict(null_t=[round(float(nt[0]),4),round(float(nt[-1]),4)] if len(nt) else None,rho={},NTa={},oor=set(),gap=set(),t={})
    for si,v in T.items():
        r=s.by_si.get(si);tv=r.get('t') if r else None
        if tv is None:out['gap'].add(si);continue
        tv=float(tv);out['t'][si]=tv
        if not len(nt) or tv<nt[0]-1e-9 or tv>nt[-1]+1e-9:out['oor'].add(si);continue
        i=int(bisect.bisect_left(nt,tv))
        if i<len(nt) and abs(nt[i]-tv)<=1e-9:val=float(nv[i])
        elif i<=0 or i>=len(nt) or nt[i]-nt[i-1]>GT_GAP_S+1e-9:out['gap'].add(si);continue
        else:val=float(np.interp(tv,nt,nv))
        if val<=0:out['gap'].add(si);continue
        out['NTa'][si]=val;out['rho'][si]=v/val
    return out

def trim_offset_reference(roff,V,ctx,thD_on,thP_on):
    info=dict(candidates=list(roff),dropped_leading=[],dropped_trailing=[],settled=False,rule='office kit: leading frames tested leave-one-out against the frames after them, then trailing frames against the frames before them')
    views=[V(x) for x in roff]
    if len(views)<2 or any(v is None for v in views):return [],False,info
    k=0;settled=False
    while len(views)-k>=3:
        rest=views[k+1:];_m1,thD1,thP1,_o1=ls086._noise(rest,ctx)
        d,p,_o,_f=ls086.metrics(views[k],ls086.median_view(rest),ctx)
        if d<=thD1 and p<=thP1:settled=True;break
        k+=1
    if not settled and len(views)-k==2:
        d,p,_o,_f=ls086.metrics(views[k],views[k+1],ctx);settled=bool(d<=thD_on and p<=thP_on)
    info['dropped_leading']=list(roff[:k]);use=list(roff[k:])
    if settled:
        while len(use)>=3:
            head=use[:-1];hv=[V(x) for x in head];_mh,thDh,thPh,_oh=ls086._noise(hv,ctx)
            d,p,_o,_f=ls086.metrics(V(use[-1]),ls086.median_view(hv),ctx)
            if d<=thDh and p<=thPh:break
            info['dropped_trailing'].insert(0,use[-1]);use=head
    info['settled']=settled;info['used']=list(use) if settled else []
    return (use if settled else []),settled,info

def cc_gate_confirmed(on_sd,null_sd,on_log,null_log,nullB_sd=None,nullB_log=None,frames=200):
    r=cc_pair(on_sd,null_sd,on_log,null_log,nullB_sd,nullB_log,frames)
    if not r.get('valid'):return dict(verdict='INVALID',errors=r.get('errors'),rule=CC_CONFIRMED_RULE)
    on=sess(on_sd);D={int(k):v for k,v in r['D'].items()}
    fr,armed=cc_schedule(on_log);seg={k:fr[k]['seg'] for k in range(frames)};des={k:fr[k]['designed'] for k in range(frames)}
    unc={k:_flag(on.by_si[k],'camera_clipping.unconfirmed') for k in range(frames)}
    lab={k:bool(on.by_si[k].get('camera_clipping.slab')) for k in range(frames)}
    pos=POS_HOLDS+(('pawn_mesh',) if (armed or {}).get('pawn_mesh') else ())
    conf=[k for k in range(frames) if seg[k] in pos and not unc[k]]
    F=r['F'];P=min(D[k] for k in conf) if conf else 0
    theta=float(np.sqrt(max(F,1)*max(P,1)));clipped={k:D[k]>theta for k in range(frames)}
    dis=[k for k in range(frames) if not unc[k] and lab[k]!=clipped[k]]
    edges=[]
    for k0 in [k for k in range(1,frames) if des[k]!=des[k-1]]:
        lo=max(0,k0-6);hi=min(frames-1,k0+6)
        lf=next((k for k in range(lo+1,hi+1) if lab[k]!=lab[k-1]),None);of=next((k for k in range(lo+1,hi+1) if clipped[k]!=clipped[k-1]),None)
        edges.append(dict(designed=k0,to=des[k0],label_flip=lf,pixel_flip=of,off=None if lf is None or of is None else lf-of))
    unj=P<=2*F
    return dict(rule=CC_CONFIRMED_RULE,F=F,P_confirmed=P,theta=round(theta,1),confirmed_positive_frames=len(conf),confirmed_disagreeing=dis,
                unconfirmed_frames=sum(unc.values()),unconfirmed_D=[D[k] for k in range(frames) if unc[k]][:40],
                unconfirmed_labelled_without_pixel_change=[k for k in range(frames) if unc[k] and lab[k] and not clipped[k]],
                edges=edges,edge_offsets=[e['off'] for e in edges],unjudgeable=unj,
