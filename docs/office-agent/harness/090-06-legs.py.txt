def labelled_window(k,cfg=CFG_EDGE):
    """Fire k (1-based) of a targeted capture run under IAI.Capture.Config <settleK> <pre> <positive> <post> <bursts>
    (AnomalyCaptureSubsystem.cpp:993-1065): LeadIn captures si 0..pre-1 and BeginFire runs on the tick that captures si pre-1, so that
    frame is labelled (the label is sampled at OnWorldTickEnd, after BeginFire); settleK ticks are not captured; Positives captures
    positive frames and BeginRevert runs on the tick of the last one (that frame is NOT labelled); SettleAfterRevert is not captured;
    PostGap captures post frames and the next BeginFire runs on the tick of the last one. Labelled window = [first, first+positive-1],
    first = (pre-1) + (positive+post)*(k-1); the revert tick is si first+positive."""
    K,pre,pos,post,_=(int(x) for x in cfg.split())
    first=(pre-1)+(pos+post)*(k-1)
    return first,first+pos-1
