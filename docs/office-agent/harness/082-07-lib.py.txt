def _pp_prongs(span,streams,monitor,game,names):
    lo,hi=span;force=[]
    inside=lambda x:x is None or lo<=x<=hi
    nongame=lambda s:'fg' in s and s.get('fg') not in game
    name=lambda s:s.get('fg_name') or names.get(str(s.get('fg'))) or ''
    streams=[[s for s in st if inside(s.get('x'))] for st in streams]
    for st in streams:force+=[(s['x']-PP_FORCE_S,s['x']+PP_FORCE_S) for s in st if s.get('x') is not None and nongame(s)]
    near=lambda a,b:any(f0<=b and a<=f1 for f0,f1 in force)
    shell=collections.Counter();moves=[];inputs=[];h_moves=0;h_inputs=0
    for st in streams:
        for s in st:
            if nongame(s) and name(s).lower() in SHELL_PROCS:shell[name(s)]+=1
        prev=None
        for i,s in enumerate(st):
            if not s.get('cursor'):continue
            if prev is not None and st[prev]['cursor']!=s['cursor']:
                if s.get('x') is None:h=any(nongame(t) for t in st[max(0,i-PP_FORCE_SAMPLES):i+PP_FORCE_SAMPLES+1])
                else:h=near(st[prev]['x'] if st[prev].get('x') is not None else s['x'],s['x'])
                if h:h_moves+=1
                else:moves.append([st[prev]['cursor'],s['cursor']])
            prev=i
    tick=sorted((s for s in monitor if s.get('input_tick') is not None and s.get('x') is not None),key=lambda s:s['x'])
    for a,b in zip(tick,tick[1:]):
        if a['input_tick']==b['input_tick']:continue
        x=b['x']-b['idle_ms']/1000.0
        if not lo<=x<=hi:continue
        if near(x,x):h_inputs+=1
        else:inputs.append(round(x,3))
    reasons=[]
    if shell:reasons.append('foreground taken by a shell process: '+', '.join(f'{k} x{v}' for k,v in shell.items()))
    if moves:reasons.append(f'cursor moved while the harness was not moving it: {len(moves)} move(s)')
    if inputs:reasons.append(f'non-harness input registered: {len(inputs)} input event(s)')
    return dict(person_evidence=bool(reasons),reasons=reasons,prongs=dict(shell=dict(shell),cursor_moves=len(moves),cursor_moves_near_focus_action=h_moves,input_events=len(inputs),input_events_near_focus_action=h_inputs))

def g3_pixel_clause(app,null,anomaly,tol=TOL):

def rd_drift(ap,evs,period=DITHER_PERIOD):
    drift=[];short=[];compared=[];diag=[]
    for e in evs:
        f0=e['si'][0];reg=region([(ap,f0,e['mv'][f0])])
        if reg is None or not reg.any():drift.append(dict(start_frame=e['start_frame'],why='no mask on the first frame'));continue
        ref={};n_cmp=0;first=None
        for si in e['si']:
            if first is None and si!=f0:
                k=int((absdiff(ap.img(si),ap.img(f0))[reg]>0).sum())
                if k:first=dict(si=si,differing_px=k,frame_index_delta=ap.by_si[si]['frame_index']-ap.by_si[f0]['frame_index'])
            ph=ap.by_si[si]['frame_index']%period
            if ph not in ref:ref[ph]=si;continue
            k=int((absdiff(ap.img(si),ap.img(ref[ph]))[reg]>0).sum());n_cmp+=1
            if k:drift.append(dict(start_frame=e['start_frame'],si=si,ref_si=ref[ph],differing_px=k));break
        compared.append(n_cmp);diag.append(dict(start_frame=e['start_frame'],first_frame_n_vs_1_differing=first))
        if not drift and n_cmp<len(e['si'])-period:short.append(dict(start_frame=e['start_frame'],compared=n_cmp,frames=len(e['si'])))
    return dict(rule=f'AMENDMENT 2 (G318): each labelled frame against the first frame of the event with the same engine frame_index mod {period}, 0 differing pixels in the frame-1 mask',
                drift=drift,short=short,compared=compared,diagnostic=dict(rule='frame N against frame 1 (082-07a) - reported, never judged; it reads the tonemapper dither',events=diag))

def ev_p21(legs):

def admission_needs(g,B=None):
    B=B or LG.by_id()
    if g.get('cls')!='Q' or g.get('fn') in ('STATIC','DECLARED-CANNOT'):return []
    legs=[B[l] for l in g['legs'] if l in B]
    if not legs or any(l['fixture']!='F-SYN' for l in legs):return []
    if g['gate'] in ADMIT_GATES:pass
    elif g['gate']=='G-REASON' and all((l.get('expect') or {}).get('final')=='APPLY' for l in legs):pass
    else:return []
    need={(l['target'],FAMKEY[l['anomaly']]) for l in legs if l['anomaly'] in FAMKEY}
    for l in legs:
        for x in l.get('after_start') or []:
            m=re.match(r'IAI\.Apply (\S+) =(\S+)$',x)
            if m and m.group(1) in FAMKEY:need.add((m.group(2),FAMKEY[m.group(1)]))
    return sorted(need)

def evaluate_ready(final=False,stop=True):

        row=dict(gate=g['gate'],row=g['row'],cls=g['cls'],stop=g['stop'],legs=g['legs'],verdict=r['verdict'],readings=r,utc=c.now());gate_add(row);out.append(row)
        if g.get('save'):jsave(c.RUN/g['save'],r)

def run_sequence(runner=real_runner,**kw):
