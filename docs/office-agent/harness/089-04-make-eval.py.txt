    on=M53R.Sess(on_sd);vn=M53R.Sess(vsd);per=[];lows=[];miss=0
    for e in M53R.events(on,anomaly):

            if reg is None or not reg.any():continue

            if v<thr:lows.append(dict(start_frame=e['start_frame'],si=si,max=v))
        per.append(dict(start_frame=e['start_frame'],measured=len(vals),min_frame_max=min(vals) if vals else None))
    me=sum(1 for p in per if p['measured']>0)
    return dict(verdict='READS-WRONG' if me>0 and not lows else ('BELOW-16' if lows else 'NO-MEASURED-EVENT'),events=per,measured_events=me,below=lows[:10],unaligned_labelled=miss)
def gt_adjudicate(on_sd,vsd,e1_events,anomaly):
    """A reading for any E1 FAIL: on each frame E1 calls wrong (an unlabelled frame judged visible without a transition, or a
    labelled frame judged invisible), ON vs the null at the same game time inside the event mask."""
    on=M53R.Sess(on_sd);vn=M53R.Sess(vsd);evs={e['start_frame']:e for e in M53R.events(on,anomaly)};out=[]
    for r in e1_events or []:
        p=((r.get('per') or {}).get(RELEASE) or {})
        if p.get('verdict')!='FAIL':continue
        e=evs.get(r.get('start_frame'))
        if not e:continue
        lab=set(M53R.labelled_sis(e));vis=set(p.get('visible') or []);tg=p.get('transition_gate') or {}
        frames=sorted(set(tg.get('unlabelled_visible_without_transition') or [])|(lab-vis))
        rd=[]
        for si in frames:
            mv=e['mv'].get(si) or next((e['mv'][x] for x in e['si'] if e['mv'].get(x)),None)
            if si not in on.by_si or si not in vn.by_si or not mv:rd.append(dict(si=si,max=None));continue
            reg=M53R.region([(on,x,mv) for x in e['si'] if e['mv'].get(x)][:1])
            rd.append(dict(si=si,labelled=si in lab,max=int(M53R.absdiff(on.img(si),vn.img(si))[reg].max()) if reg is not None and reg.any() else None))
        out.append(dict(start_frame=r.get('start_frame'),frames=rd))
    return out
