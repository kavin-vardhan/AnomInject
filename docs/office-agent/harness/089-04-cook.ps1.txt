param([string]$Tag = 'rb1', [string]$StagingDir = '')
$ev = 'D:\IntrusiveAnomalies\_reviews\089-04-evidence\cook'
New-Item -ItemType Directory -Force $ev | Out-Null
$pak = 'D:\UESource\UnrealEngine\Engine\Binaries\Win64\UnrealPak.exe'
$maps = '/Game/CaptureBenchGate/CB_GateLevel+/Game/StackOBot/UI/MainMenu/MainMenu+/Game/StackOBot/Maps/MainWorld+/Game/CaptureBenchTexCorrupt/CB_TexCorruptLevel'
$env:MSBUILDDISABLENODEREUSE = '1'

if (Test-Path $out) { Add-Content "$ev\cooks.progress.txt" "$Tag EXISTS - stop"; exit 3 }

$a = @('BuildCookRun', '-project=E:\IA_BuildCache\_r53_host\StackOBot\StackOBot.uproject', '-platform=Win64', '-clientconfig=Development', '-cook', '-stage', '-pak', '-archive', "-archivedirectory=$out", '-build', '-utf8output', '-nocompileeditor', "-map=$maps")

Add-Content "$ev\cooks.progress.txt" ("{0} success={1} secs={2} list={3} at {4}" -f $Tag, $ok, [int]((Get-Date) - $t0).TotalSeconds, $le, (Get-Date -Format HH:mm:ss))
