def sha(p):
    with Path(p).open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()
def now():return datetime.datetime.now(datetime.timezone.utc).isoformat()
def git(r,*a):return subprocess.check_output(['git','-C',str(r),*a]).decode().strip()
def state(r):
    return {'head':git(r,'rev-parse','HEAD'),'branch':git(r,'branch','--show-current'),
            'refs':git(r,'for-each-ref','--format=%(refname) %(objectname)','refs/heads','refs/remotes'),
            'untracked':{p:sha(r/p) for p in git(r,'ls-files','--others','--exclude-standard').splitlines()}}
def classify():
    names=",".join("'"+n+"'" for n in BENCH_NAMES)
    s=subprocess.check_output(['powershell','-NoProfile','-Command',"@(Get-CimInstance Win32_Process | Where-Object { $_.Name -in "+names+" } | Select-Object ProcessId,ParentProcessId,Name,ExecutablePath,CommandLine,CreationDate) | ConvertTo-Json -Compress"]).decode().strip()
    v=[] if not s or s=='[]' else json.loads(s) if s.startswith('[') else [json.loads(s)]
    ours=[];foreign=[]
    for x in v:
