def table(headers, rows):
    def line(v):
        return "| " + " | ".join(str(x).replace("|", "/").replace("\n", " ") for x in v) + " |"
    return "\n".join([line(headers), line(["---"] * len(headers)), *(line(r) for r in rows)])
