Scratch configuration (device profile, saved config) is placed only for the legs that need it and removed by receipt."""

def dp_text(profile):
    lines = ['[%s DeviceProfile]' % profile, 'DeviceType=Windows', 'BaseProfileName=%s' % ('Windows' if profile != 'Windows' else '')]
    for g in GROUPS:
        lines.append('+TextureLODGroups=(Group=%s,MinLODSize=1,MaxLODSize=16384,LODBias=1,MinMagFilter=aniso,MipFilter=point,'
                     'MipGenSettings=TMGS_SimpleAverage)' % g)
    return '\n'.join(lines) + '\n'


def place(path, text, note):
    path = Path(path)
    if path.exists():
        raise SystemExit(f'refusing to overwrite an existing {path}')
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(text, encoding='utf-8')
    return C.receipt(path, 'PLACED ' + note)


def remove(path, note):
    path = Path(path)
    if path.exists():
        rec = C.receipt(path, 'REMOVING ' + note)
        path.unlink()
        C.receipt(path, 'REMOVED ' + note)
        return rec
    return None


def manifest(tag):
    out = W / f'manifest_{tag}.json'
    r = subprocess.run([sys.executable, str(R / '085-03-content-manifest.py'), 'hash', str(out)], capture_output=True, text=True)
    return dict(tag=tag, rc=r.returncode, out=str(out), tail=r.stdout.strip()[-300:])


def census_spec(build, map_, stress=False, texmods=None, extra=None, inventory=None, first_only=False, pre=None):
    cmds = []
    if stress:
        cmds += STRESS
    if texmods:
        cmds += [f'PYSTEP {PY / "texmods.py"}']
    if inventory:
        cmds += [f'PYSTEP {PY / "inventory.py"}']
    cmds += pre or []
    cmds += ['WAIT 10']
    if first_only:
        cmds += ['IAI.Bench.TexCorruptAllReasonsFirstOnly 1']
    cmds += ['IAI.TexCorrupt.Census allreasons', 'WAIT 45']
    pe = {}
    if texmods:
        pe['texmods'] = texmods
        pe['texmods_wait_s'] = 12.0
    return dict(mode='pie', build=build, map=map_, arbiter=False, pre=cmds, hold_s=5.0, wait=420,
                extra=(extra or []) + ['-IAIBench'], plan_extra=pe)


def run(label, spec, inventory=False, texmods=False):
    if inventory:
        spec.setdefault('plan_extra', {})['inventory_out'] = str(W / 'inv' / f'{label}.json')
        (W / 'inv').mkdir(parents=True, exist_ok=True)
    if texmods:
        spec.setdefault('plan_extra', {})['texmods_out'] = str(W / 'inv' / f'{label}.texmods.json')
        (W / 'inv').mkdir(parents=True, exist_ok=True)
    rec = C.run(label, spec)
    print(json.dumps(dict(label=label, status=rec.get('status'), rc=(rec.get('run') or {}).get('rc'),
                          census=C.grep(label, r'IAI-TEXCORRUPT-CENSUS v1 (id=|subs|allreasons id=\S+ objects)', 12)), indent=1), flush=True)
    return rec

UV, NM = 'uv_corruption', 'normal_corruption'
FIX, FIX_N, STREAM = 'StaticMeshActor_0', 'StaticMeshActor_12', 'StaticMeshActor_74'
FOCUS = ['IAI.Capture.FocusGate 0']


def variant(v, mode):
    """Per-variant pre commands, plan extras, extra args and scratch config for a mode (pie|staged)."""
    pre, pe, extra, dp, target_uv = [], {}, [], None, FIX
    if v in ('V1', 'V3'):
        bias, cin = (1, 0) if v == 'V1' else (0, 1)
        if mode == 'pie':
            mods = [dict(actors=[FIX, FIX_N], lod_bias=bias if v == 'V1' else None, cinematic=cin if v == 'V3' else None)]
            pre += [f'PYSTEP {PY / "texmods.py"}']
            pe.update(texmods=mods, texmods_wait_s=10.0)
        else:
            pre += [f'IAI.Bench.TexCorruptHostTexBias {FIX} {bias} {cin}', f'IAI.Bench.TexCorruptHostTexBias {FIX_N} {bias} {cin}']
    elif v == 'V2':
        dp = 'pie' if mode == 'pie' else 'staged'
    elif v == 'V4':
        pre += ['r.MipMapLODBias 1']
    elif v == 'V5a':
        pre += ['r.Streaming.UsePerTextureBias 1', 'r.Streaming.MipBias 1']
    elif v == 'V5b':
        pre += ['r.Streaming.UsePerTextureBias 0', 'r.Streaming.MipBias 1']
    elif v == 'V6':
        pre += ['r.Streaming.PoolSize 40']
        target_uv = STREAM
    return pre, pe, extra, dp, target_uv


def with_dp(where, fn):
    if where is None:
        return fn()
    if where == 'pie':
        place(HOST_DP, dp_text('WindowsEditor'), 'scratch texture-group LODBias=1 (V2, PIE)')
        try:
            return fn()
        finally:
            remove(HOST_DP, 'scratch texture-group LODBias=1 (V2, PIE)')
    bak = None
    if STAGED_DP.exists():
        bak = W / 'staged_DeviceProfiles.ini.bak'
        shutil.copyfile(STAGED_DP, bak)
        C.receipt(bak, 'BACKUP of the staged build saved DeviceProfiles.ini before V2')
        STAGED_DP.unlink()
    place(STAGED_DP, dp_text('Windows'), 'scratch texture-group LODBias=1 (V2, staged saved config)')
    try:
        return fn()
    finally:
        remove(STAGED_DP, 'scratch texture-group LODBias=1 (V2, staged)')
        if bak:
            shutil.copyfile(bak, STAGED_DP)
            C.receipt(STAGED_DP, 'RESTORED the staged build saved DeviceProfiles.ini after V2')


def proof_leg(label, build, mode, v, kind):
    """kind: uv | normal | identity | canfail (census both targets + a uv capture)."""
    pre, pe, extra, dp, tuv = variant(v, mode)
    spec = dict(mode=mode, build=build, map=c.MAP_FIX, census=False, extra=extra + ['-IAIBench'], frames=150, cfg='2 4 8 14 0')
    pre = FOCUS + pre
    if kind == 'uv':
        spec.update(anomaly=UV, target=tuv, args=['tile'])
    elif kind == 'normal':
        spec.update(anomaly=NM, target=FIX_N, args=['invert'])
    elif kind == 'identity':
        pre += ['IAI.Bench.TexCorruptIdentity 1']
        spec.update(anomaly=UV, target=tuv)
    elif kind == 'canfail':
        pre += [f'IAI.Bench.TexCorruptCensus {tuv}', f'IAI.Bench.TexCorruptCensus {FIX_N}']
        spec.update(anomaly=UV, target=tuv, args=['tile'])
    if mode == 'pie':
        pre = pre[:1] + pre[1:]
        pe.setdefault('texmods_out', str(W / 'inv' / f'{label}.texmods.json'))
        spec['plan_extra'] = pe
        spec['wait'] = 600
    spec['pre'] = pre
    return with_dp(dp, lambda: run(label, spec))
