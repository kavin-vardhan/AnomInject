def sess_of(leg_id):
    sd=L.leg_session(leg_id)
    if sd is None and c.STUB:
        a=[x for x in L.ledger() if x['leg']==leg_id and x['status']=='ACCEPTED']
        if a:sd=L.find_session(c.BANK/a[-1]['label'])
    return sd
def premise_verdict(leg_id):
    sd=sess_of(leg_id)
    if sd is None:return dict(pass_=False,reason='premise leg not accepted')
    d=E.diff_leg(sd,log=L.leg_log(leg_id));ev=d['events']
    elig=[r for r in ev if r.get('status')=='OK']
    meas=[r for r in elig if r.get('measurable')]
    frac=len(meas)/len(elig) if elig else 0.0
    out=dict(pass_=bool(elig) and frac>=0.8,eligible=len(elig),measurable=len(meas),fraction=round(frac,3),
             status=dict(collections.Counter(r.get('status') for r in ev)),rule='>= 80 % of non-warm-up, camera-still (status OK) events MEASURABLE')
    if c.STUB and DRY_FORCE_PREMISE and not out['pass_']:
        out=dict(out,pass_=True,forced=True,real_pass=False,note='DRY ONLY: premise FORCED to pass so the dry run launches and evaluates the gate legs on stand-in data; the real stand-in verdict is real_pass')
    L.jsave(c.RUN/f'premise-{leg_id}.json',out);return out

def lod_fixture_summary(g):

                      N=None if not gn else {k:gn[k] for k in ('verdict','judged','off_frames_t50_outside_transition','fade_edges','edges','fade_control','intermediate_hist')},
                      compare=E.lod_fade_compare(gd,gn))

            rep['legs'][Lg['leg']]=dict(status=(nr[-1]['status'] if nr else 'NOT ACCEPTED'),group=Lg['group']);continue
        log=L.leg_log(Lg['leg']);r=dict(group=Lg['group'],role=Lg['role'],session=str(sd))
        try:
            r.update(common_checks(sd))
            if Lg['group']=='M52':
                if Lg['role']=='NULL':
                    on=sess_of('B0');r['null']=E.null_check_v2(sd,on,LG.ROCK)
                else:r.update(eval_m52_leg(Lg,sd,log,sess_of(Lg['null']) if Lg.get('null') else None))
            elif Lg['group']=='BCC':
                if Lg['role']=='CC_ON':
                    na,nb=Lg['pair'];sa=sess_of(na);sb=sess_of(nb) if nb else None
                    r['cc']=E.cc_gate(sd,sa,log,L.leg_log(na),sb,L.leg_log(nb) if nb else None,Lg['frames']) if sa else dict(valid=False,gate='INVALID',errors=[f'null {na} not accepted'])
                    e2=E.ls086_verdict(sd,types={'camera_clipping'});r['e2']=E.summarize_ls086(e2)
                else:r['role_note']='null leg: read through its ON pair'
            elif Lg['group']=='BCCMW':
                if Lg['role']=='CC_ON':
                    na,nb=Lg['pair'];sa=sess_of(na);sb=sess_of(nb) if nb else None
                    r['cc_mw']=E.cc_mw_gate(sd,sa,sb,Lg['frames']) if sa else dict(gate='INVALID',verdict='NO-NULL')
                else:r['role_note']='null leg: read through its ON pair'
            elif Lg['group']=='MASK55V2':
                r['mask']=E.mask_gate(sd,log);r.update(eval_diff_leg(Lg,sd,log))
            else:
                r.update(eval_diff_leg(Lg,sd,log))
                if Lg['role']=='PREMISE':r['premise']=premise_verdict(Lg['leg'])
                elif Lg['group'] in ('BLOD','BLODF'):r['lod']=E.lod_gate(r['e1_events'],variant=Lg.get('variant'));r['lod']['cond']=Lg.get('cond')
        except Exception:
            import traceback;r['error']=traceback.format_exc()[-1500:]
        rep['legs'][Lg['leg']]=r
    g=rep['legs']
    if 'e1' in (g.get('B0L') or {}):rep['can_fail']['B0L_timing_lever_must_FAIL']=dict(e1=g['B0L']['e1']['leg_verdict'],proven=g['B0L']['e1']['leg_verdict']=='FAIL')
    if 'purity' in (g.get('P1') or {}):rep['can_fail']['P1_purity_lever_must_FAIL']=dict(observed=g['P1']['purity']['verdict'],proven=g['P1']['purity']['verdict'].startswith('FAIL'))
    for k,v in g.items():
        if isinstance(v,dict) and (v.get('cc') or {}).get('valid'):rep['can_fail'][f'{k}_sphere_proxy_must_disagree']=dict(proven=v['cc']['gate2_canfail_proven'])
    rep['lod_fixture']=lod_fixture_summary(g)
    rep['label_rule_all_pass']=all((v.get('label_rule') or {}).get('pass_') for v in g.values() if isinstance(v,dict) and v.get('label_rule'))
    rep['harness']={p.name:c.sha(p)[:16] for p in sorted(R.glob('084-09-*')) if p.is_file() and p.suffix in ('.py','.ps1','.md')}

def postflight():
    problems=[]
    def chk(ok,msg):
        if not ok:problems.append(msg)
    ours,foreign=c.classify()
    chk(not [p for p in ours if p['name']!='UnrealTraceServer.exe'],'our engine/game processes present at close')
    st=c.staged_state();chk(st==c.M53,'staged set is not the m53 set: '+json.dumps(st))
    chk(c.git(c.RP,'rev-parse','HEAD')==c.M51_HEAD and c.git(c.RP,'branch','--show-current')=='m51','main checkout moved off m51 53bf725')
    fh=c.git(c.RP,'rev-parse',c.FIX_BRANCH);fo=c.git(c.RP,'rev-parse','refs/remotes/origin/'+c.FIX_BRANCH);chk(fh==fo,f'fix branch {fh[:7]} != origin {fo[:7]}')
    led=L.ledger();acc=[x for x in led if x.get('status')=='ACCEPTED'];links=0
    for x in acc:
        a=c.BANK/x['label'];al=c.BANK/(c.BANK_PREFIX+x['leg'])
        for p in [p for p in a.rglob('*') if p.is_file()]:
            q=al/p.relative_to(a)
            if not q.exists() or os.stat(q).st_ino!=os.stat(p).st_ino:problems.append(f'alias not a hardlink: {q}');break
            links+=1
    left=[d.name for d in c.EXE_SIDE.iterdir() if d.is_dir() and d.name.startswith(c.BANK_PREFIX)] if c.EXE_SIDE.exists() else []
    chk(not left,f'leg copies left after banking: {left[:5]}')
    bk={k:c.sha(c.M53_BACKUP/k) for k in c.KEYS if (c.M53_BACKUP/k).exists()};chk(bk==c.M53,'m53 backup does not verify')
    out=dict(utc=c.now(),problems=problems,accepted_legs=len(acc),attempts=len([x for x in led if x.get('label')]),not_accepted=[x for x in led if x.get('status')!='ACCEPTED'],
             alias_hardlinks_verified=links,left=left,staged=st,fix_head=fh,foreign_present=[(p['pid'],p['name']) for p in foreign],verdict='PASS' if not problems else 'FAIL')
    L.jsave(c.RUN/'postflight.json',out);return out
