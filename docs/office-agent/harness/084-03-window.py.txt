    done=[]

def write_variant(s,e,window,record,settle_extra,tag,tmask,out_dir,mask_mode='normal',drop_si=(),jump_from=None,jump=0,image_src=None):

    for r in s.rows:

    ev_rows=set(e['rows'])|W|RS|SX;rows=[];px=int((tmask==tag).sum())
    for r in s.rows:

        if jump_from is not None and si>=jump_from and nr.get('frame_index') is not None:nr['frame_index']=nr['frame_index']+jump
        an=[a for a in (r.get('anomalies') or []) if not (a.get('id')==L.ID and a.get('start_frame')==e['start_frame'])]

def template_mask(s,e):
    import numpy as np
    for si in e['rows']:
        mv=e['mv'].get(si);m=s.mask(si)
        if mv and m is not None and int((m==mv).sum())>0:return int(mv),m,'delivered mask png of this event'
    bb=[b for b in e['bbox'].values() if b and len(b)==4]
    if not bb:return None,None,None
    x,y,w,h=(int(v) for v in bb[len(bb)//2]);H,Wd=s.shape();m=np.zeros((H,Wd),dtype=np.uint8);m[max(0,y):min(H,y+h),max(0,x):min(Wd,x+w)]=200
    return 200,m,'synthesised from bbox_px (no delivered mask for this event)'
VARIANTS=['repaired','onset_plus2','offset_minus5','early_onset_overlabel_minus2','late_offset_overlabel_plus4','masks_missing','mask_png_absent',
          'gap_missing_si_after_offset','gap_engine_jump_in_suffix','gap_engine_jump_before_onset','gap_engine_jump_inside_window',

def proof_summary(cases):
    checks={}
    for n in VARIANTS:
        checks[n]=dict(expect=next((cs['variants'][n]['expect'] for cs in cases if n in cs['variants']),None),required=n not in OPTIONAL_VARIANTS,
                       proven_on=[f"{cs['source']}:{cs['start_frame']}" for cs in cases if n in cs['variants'] and cs['variants'][n]['ok']],
                       contradicted_on=[f"{cs['source']}:{cs['start_frame']}" for cs in cases if n in cs['variants'] and not cs['variants'][n]['ok']])
    return dict(checks=checks,proven=bool(cases) and all((d['proven_on'] or not d['required']) and not d['contradicted_on'] for d in checks.values()),
                not_constructed=[n for n,d in checks.items() if not d['proven_on'] and not d['contradicted_on']],
                base_edited_cases=[f"{cs['source']}:{cs['start_frame']}" for cs in cases if cs.get('base_pixel_edits')])

    for i,e in enumerate(evs):

                out['skipped'].append(f"{who}: a visible RUN {L.runs_of(fo)} outside the pixel window that the null does not show; under rule (b) no label can pass, so it cannot show the PASS half");continue

            m=L.measure_event(sb,eb,nf,fref)
            if m.get('status')!='OK' or m['censored'] or m['first_vis']!=fv or m['last_vis']!=lv or L.gate_sync(sb,eb,m,False,ctx)['visible_outside_window_failing']:
                out['skipped'].append(f"{who}: the despiked oracle does not reproduce the window {[fv,lv]} cleanly");continue
        deep=sorted((si for si in vis if fv<=si<=lv),key=lambda x:T0[x])

            if prec is not None:kr=n_;break

            if m2.get('status')!='OK':
                case['variants'][name]=dict(expect=expect_txt,ok=False,sync_verdict=None,window_gate=None,detail=f"re-measure status {m2.get('status')}");return

            gp=L.gate_sync(s2,e2,m2,False,cx);gr=L.gate_sync(s2,e2,m2,True,cx);gw=L.gate_window(s2,e2,m2)
            ok,detail=expect(gp,gr,gw,m2)
            case['variants'][name]=dict(expect=expect_txt,ok=bool(ok),detail=detail,label=[min(win),max(win)] if win else None,sync_verdict=gp['verdict'],
                                        two_sided_pixel_gate=gp['pass'],with_record_check=gr['pass'],window_gate=gw['pass_'],old_one_sided_gate=old_gate_sync(e2,m2),
                                        masks_on_disk=len(L.mask_window(s2,e2)),fail_reasons=gp['fail_reasons'],censored_edges=gp['censored_edges'],
                                        visible_outside_window_failing=gp['visible_outside_window_failing'],null_explained=gp['visible_outside_window_null_explained'],
                                        mask_misses=gw['labelled_without_mask_reasons'])

        cen=lambda side:(lambda gp,gr,gw,m2:(gp['verdict']!='PASS' and bool(m2.get(side+'_censored')),f"verdict {gp['verdict']}; {side}: {m2.get(side+'_censored')}"))
        mfail=lambda gp,gr,gw,m2:(gp['pass'] and not gw['pass_'] and bool(gw['labelled_without_mask']),f"mask misses {gw['labelled_without_mask_reasons']}")

        if base_edit:
            run('original_spike_unedited',pix,pix,(),lambda gp,gr,gw,m2:(gp['verdict']=='FAIL' and gp['visible_outside_window_failing']==sorted(base_edit),f"failing {gp['visible_outside_window_failing']}"),
                'FAIL (i): the real, unedited isolated visible frame(s) outside the window (G345 withdrawn, correction b)',no_base=True,oracle=m_orig)

                'PASS: a jump between two visible frames is not adjacent to an edge (correction a is not over-broad)',remeasure=True,jump_from=jin,jump=3)
        else:case['notes'].append('gap_engine_jump_inside_window not built: no two consecutive visible frames in fv+2..lv')
        if prec is not None:
            src={prec+j:deep[j] for j in range(kr)};inj=list(range(prec,prec+kr))
            run('recurrence_real_pixels',pix,pix,(),lambda gp,gr,gw,m2:(gp['verdict']=='FAIL' and set(inj)<=set(gp['visible_outside_window_failing']),f"injected {inj} from {[deep[j] for j in range(kr)]}; failing {gp['visible_outside_window_failing']}"),
                'FAIL (i): a later visible run outside the window (correction b)',remeasure=True,image_src=src)

            if k<len(ne):

            run('recurrence_shown_by_null',pix,pix,(),lambda gp,gr,gw,m2:(gp['pass'] and set(str(x) for x in inj)<=set(gp['visible_outside_window_null_explained']),f"null carrier frames {sorted(amap)} carry the same pixels; explained {gp['visible_outside_window_null_explained']}"),
                'PASS: the same run is shown by the null at the aligned frames, a different cause (correction b)',remeasure=True,nctx=dict(sess=ns2,ordinal=k),image_src=src)

        if psp is not None:
            run('single_spike_real_pixels',pix,pix,(),lambda gp,gr,gw,m2:(gp['verdict']=='FAIL' and psp in gp['visible_outside_window_failing'],f"spike at si {psp} from si {deep[0]}"),

        else:case['notes'].append('single_spike_real_pixels not built: no clean contiguous room')
        v=case['variants'];case['proven']=all(x['ok'] for x in v.values())
        case['old_gate_missed']=[n for n in v if n!='repaired' and v[n].get('old_one_sided_gate')]
        out['cases'].append(case)
    out.update(proof_summary(out['cases']))
    out['note']=('each variant is a REAL session copy on disk (labels.jsonl rewritten, mask PNGs written, frames = the banked pixels). '
                 'Label and mask variants are judged against the FROZEN pixel oracle measured once on the original; gap and recurrence '

                 're-points the paired null\'s aligned frames at the same PNGs. old_gate_missed lists the variants the 084-02 one-sided '
                 'gate passed.')
    return out
