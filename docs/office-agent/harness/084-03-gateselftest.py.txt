def build(vis=(),n=60,drop=(),jumps=None,evr=range(5,41),lab=(),rec=None,mv=7,tp=100,masks='png',tp_at=None,contam=(),null=False):
    jumps=jumps or {};T={};rows=[];off=0;M={};rec=lab if rec is None else rec;tp_at=tp_at or {}
    for si in range(n):
        off+=jumps.get(si,0)
        if si in drop:continue
        T[si]=90.0 if si in vis else 100.0
        r=dict(session_index=si,frame_index=100+si+off,anomalies=[])
        if si in evr:
            held=(si in lab) and not null
            a={'id':L.ID,'start_frame':SF,'target_name':'Rock','stuck_mip.held':held,'stuck_mip.render_state':('held' if (si in rec and not null) else 'baseline'),
               'mask_value':mv,'target_pixels':tp_at.get(si,tp if held else 0)}
            if si in contam:a['stuck_mip.contaminated']=1
            r['anomalies'].append(a)
            if held and masks=='png':M[si]=np.full((4,4),mv,dtype=np.uint8)
        rows.append(r)
    return FS(rows,T,M)
