param(
 [Parameter(Mandatory=$true)][string]$Label,
 [Parameter(Mandatory=$true)][string]$EvDir,
 [Parameter(Mandatory=$true)][string]$Map,
 [Parameter(Mandatory=$true)][string]$ExecCmds,
 [string]$Kind='capture',
 [string]$OutLabel='',
 [int]$WaitSec=240,
 [string]$ClosePattern='',
 [int]$CloseDelayMs=1000,
 [int]$CensusCount=0,
 [int]$ResX=1280,
 [int]$ResY=720,
 [switch]$Stub
)
$ErrorActionPreference='Stop'
$exe='D:\IntrusiveAnomalies\StackOBot\Builds\BenchGate\Windows\StackOBot\Binaries\Win64\StackOBot.exe'
$exeSide=Split-Path $exe
if(@('capture','close_after','close_mid','census','cancel') -notcontains $Kind){throw "unknown kind $Kind"}
if(($Kind -in @('capture','close_after','close_mid','cancel')) -and -not $OutLabel){throw 'OutLabel required for a capture kind'}
if(Test-Path -LiteralPath $EvDir){throw "evidence dir exists: $EvDir"}
New-Item -ItemType Directory -Path $EvDir | Out-Null
$log=Join-Path $EvDir 'game.log'
$argsList=@($Map,'-windowed',"-ResX=$ResX","-ResY=$ResY","-ExecCmds=`"$ExecCmds`"",'-unattended','-nosplash',"-abslog=$log")
$record=[ordered]@{label=$Label;kind=$Kind;map=$Map;out_label=$OutLabel;exec_cmds=$ExecCmds;args=$argsList;stub=[bool]$Stub;started_utc=[DateTime]::UtcNow.ToString('o');focus=@();capture_focus=@();errors=@()}
function Save-Record{$record|ConvertTo-Json -Depth 8|Set-Content -LiteralPath (Join-Path $EvDir 'launch.json') -Encoding utf8}
if($Stub){$record.status='STUB';$record.exe_exists=(Test-Path -LiteralPath $exe);$record.cmd_chars=$ExecCmds.Length;Save-Record;Write-Output "STUB $Label kind=$Kind chars=$($ExecCmds.Length)";exit 0}
Add-Type @"
using System; using System.Runtime.InteropServices;
public class M53Fg {
 [DllImport("user32.dll")] public static extern IntPtr GetForegroundWindow();
 [DllImport("user32.dll")] public static extern uint GetWindowThreadProcessId(IntPtr h,out uint pid);
 [DllImport("user32.dll")] public static extern bool SetForegroundWindow(IntPtr h);
 [DllImport("user32.dll")] public static extern bool BringWindowToTop(IntPtr h);
 [DllImport("user32.dll")] public static extern bool ShowWindow(IntPtr h,int cmd);
 [DllImport("user32.dll")] public static extern void keybd_event(byte k,byte s,uint f,UIntPtr e);
 [StructLayout(LayoutKind.Sequential)] public struct Point { public int X; public int Y; }
 [DllImport("user32.dll")] public static extern bool GetCursorPos(out Point p);
 public static uint FgPid(){uint p;GetWindowThreadProcessId(GetForegroundWindow(),out p);return p;}
 public static string Cursor(){Point p;GetCursorPos(out p);return p.X+","+p.Y;}
 public static void Force(IntPtr h){ShowWindow(h,9);keybd_event(0x12,0,0,UIntPtr.Zero);keybd_event(0x12,0,2,UIntPtr.Zero);SetForegroundWindow(h);BringWindowToTop(h);}
}
"@
$busy=@(Get-Process -Name StackOBot,UnrealEditor,UnrealEditor-Cmd,LyraGame,LyraEditor -ErrorAction SilentlyContinue)
if($busy.Count){$record.status='REFUSED-NOT-IDLE';$record.busy=@($busy|ForEach-Object{@{pid=$_.Id;name=$_.ProcessName}});Save-Record;throw 'bench not idle - refused (nothing is killed)'}
$outDir=if($OutLabel){Join-Path $exeSide $OutLabel}else{''}
if($outDir -and (Test-Path -LiteralPath $outDir)){throw "exe-side output exists: $outDir"}
$record.exe_sha256=(Get-FileHash -LiteralPath $exe -Algorithm SHA256).Hash.ToLower()
$record.cursor_before=[M53Fg]::Cursor()
$record.out_dir=$outDir
Save-Record
$p=$null
function Log-Text{if(Test-Path -LiteralPath $log){return [string](Get-Content -LiteralPath $log -Raw -ErrorAction SilentlyContinue)}return ''}
function Check-Foreign($where){$fe=@(Get-Process -Name UnrealEditor,UnrealEditor-Cmd,LyraEditor -ErrorAction SilentlyContinue);if($fe.Count){$record.foreign_editor=@($fe|ForEach-Object{@{pid=$_.Id;path=$_.Path}});$record.foreign_where=$where;throw "FOREIGN-EDITOR-APPEARED $where"}}
function Hold-Focus($sh){try{$sh.AppActivate($p.Id)|Out-Null}catch{};try{$p.Refresh();if($p.MainWindowHandle -ne 0){[M53Fg]::Force($p.MainWindowHandle)}}catch{}}
function Close-Ours{
 $record.close_requested_utc=[DateTime]::UtcNow.ToString('o')
 $record.close_sent=$p.CloseMainWindow()
 $record.exited_after_close=$p.WaitForExit(30000)
 if($record.exited_after_close){$record.exit_code=$p.ExitCode}
}
try{
 $p=Start-Process -FilePath $exe -ArgumentList $argsList -PassThru
 $record.pid=$p.Id;Save-Record
 $sh=New-Object -ComObject WScript.Shell
 $sw=[Diagnostics.Stopwatch]::StartNew();$stable=0;$focusAt=-1.0
 for($i=0;$i -lt 90;$i++){
  Start-Sleep -Milliseconds 400
  if($p.HasExited){throw 'game exited during startup'}
  Check-Foreign 'startup'
  Hold-Focus $sh
  $fg=[M53Fg]::FgPid()
  $record.focus+=@{utc=[DateTime]::UtcNow.ToString('o');foreground_pid=$fg;cursor=[M53Fg]::Cursor()}
  if($fg -eq $p.Id){if($focusAt -lt 0){$focusAt=[math]::Round($sw.Elapsed.TotalSeconds,1)};$stable++}else{$stable=0}
  if($stable -ge 8){break}
 }
 $record.focus_at_s=$focusAt;$record.focus_held=([M53Fg]::FgPid() -eq $p.Id)
 $record.sampling_started_utc=[DateTime]::UtcNow.ToString('o')
 $record.anchor_start=@{utc=[DateTime]::UtcNow.ToString('o')}
 $run=[Diagnostics.Stopwatch]::StartNew();$nextForeign=0.0;$session=$null
 if($Kind -eq 'census'){
  while($run.Elapsed.TotalSeconds -lt $WaitSec){
   if($p.HasExited){throw 'game exited during census'}
   if($run.Elapsed.TotalSeconds -ge $nextForeign){$nextForeign+=1.0;Check-Foreign 'census'}
   $n=([regex]::Matches((Log-Text),'TEXCORRUPT-CENSUS END')).Count
   if($n -ge $CensusCount){$record.census_end_lines=$n;break}
   Start-Sleep -Milliseconds 250
  }
  if(-not $record.census_end_lines){throw "CENSUS-INCOMPLETE ($CensusCount END lines expected)"}
  Start-Sleep -Milliseconds 2500
 } elseif($Kind -eq 'cancel'){
  while($run.Elapsed.TotalSeconds -lt 60){
   if($p.HasExited){throw 'game exited before the cancel line'}
   if((Log-Text) -match 'Capture run CANCELLED before focus'){$record.cancel_seen=$true;break}
   Start-Sleep -Milliseconds 250
  }
  if(-not $record.cancel_seen){throw 'CANCEL-LINE-NOT-SEEN'}
  Start-Sleep -Milliseconds 2500
  $record.session_dir_exists=(Test-Path -LiteralPath $outDir)
 } else {
  $closeDone=$false
  while($run.Elapsed.TotalSeconds -lt $WaitSec){
   if($p.HasExited){if($closeDone){break};throw 'game exited during capture'}
   $t=$run.Elapsed.TotalSeconds
   if($t -ge $nextForeign){$nextForeign+=1.0;Check-Foreign ('capture t='+[math]::Round($t,1))}
   $fg=[M53Fg]::FgPid();$record.capture_focus+=@{seconds=$t;foreground_pid=$fg}
   if($fg -ne $p.Id){Hold-Focus $sh}
   if($Kind -eq 'close_mid' -and -not $closeDone){
    if((Log-Text).Contains($ClosePattern)){$record.close_pattern_seen_utc=[DateTime]::UtcNow.ToString('o');Start-Sleep -Milliseconds $CloseDelayMs;Close-Ours;$closeDone=$true;break}
   } else {
    $session=Get-ChildItem -LiteralPath $outDir -Directory -ErrorAction SilentlyContinue|Where-Object{Test-Path -LiteralPath (Join-Path $_.FullName 'run_summary.json')}|Select-Object -First 1
    if($session){break}
   }
   Start-Sleep -Milliseconds 100
  }
  $record.anchor_end=@{utc=[DateTime]::UtcNow.ToString('o')}
  if($Kind -eq 'close_mid'){
   if(-not $closeDone){throw 'CLOSE-PATTERN-NOT-SEEN'}
   $session=Get-ChildItem -LiteralPath $outDir -Directory -ErrorAction SilentlyContinue|Select-Object -First 1
  } else {
   if(-not $session){throw 'NO-RUN-SUMMARY within the wait'}
   $prevN=-1;$prevZ=-1;$st=0
   for($w=0;$w -lt 40;$w++){
    $ff=@(Get-ChildItem -LiteralPath (Join-Path $session.FullName 'Actual_Frames') -File -ErrorAction SilentlyContinue)
    $n=$ff.Count;$z=@($ff|Where-Object{$_.Length -eq 0}).Count
    if($n -eq $prevN -and $z -eq $prevZ){$st++}else{$st=0}
    if($st -ge 4 -and $z -eq 0){break}
    $prevN=$n;$prevZ=$z;Start-Sleep -Milliseconds 250
   }
   Start-Sleep -Milliseconds 2500
   if($Kind -eq 'close_after'){Close-Ours}
  }
  if($session){$record.session=$session.FullName}
 }
 $record.status='COMPLETE'
}catch{
 $record.status='PROTOCOL-FAIL';$record.error=$_.Exception.Message
}finally{
 if($p -and -not $p.HasExited){Stop-Process -Id $p.Id -Force;$p.WaitForExit();$record.killed_by_pid=$p.Id}
 elseif($p -and $null -eq $record.exit_code){try{$record.exit_code=$p.ExitCode}catch{}}
 $record.finished_utc=[DateTime]::UtcNow.ToString('o');Save-Record
}
Write-Output "$Label kind=$Kind status=$($record.status) session=$($record.session) error=$($record.error)"
if($record.status -ne 'COMPLETE'){exit 1}
exit 0
