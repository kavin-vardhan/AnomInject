def kat_log(seed,attempts,extra_draw_at=None):
    st=EV.KAT.KatStream(seed);lines=[]
    for i,(ne,nc,is_m53) in enumerate(attempts):
        sb=st.seed;n=0
        if ne>0:
            n=1+((2+(1 if is_m53 else 0)) if nc>0 else 0)
        if extra_draw_at==i:n+=1
        for _ in range(n):st.mutate()
        lines.append(draw_line(i,sb,'uv_corruption' if is_m53 else 'blinking',f'{ne}:ab12','-','StaticMeshActor_0' if nc else '-',f'{nc}' if nc else '-','3.5','tile' if is_m53 else '-',st.seed,'fired' if ne else 'all_excluded'))
    return '\n'.join(lines)

               delay_known_answer,doctor_cases,restore_cases,module_selftests):
