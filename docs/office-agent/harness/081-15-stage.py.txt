from pathlib import Path
import hashlib,json,shutil,subprocess,sys
base=Path(r'D:\IntrusiveAnomalies');ev=base/'_reviews/081-15-evidence'
def sha(p):
    with Path(p).open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()
prior=json.loads(Path(sys.argv[1]).read_text());resume='--resume' in sys.argv
assert sha(prior['archive'])==prior['sha256']
assert sha(prior['staged'])==prior['sha256'] or (resume and sha(prior['staged'])==sha(prior['built']))
for r in prior['containers']:assert sha(r['path'])==r['sha256']
built=Path(prior['built']);h=sha(built)
archive=base/f'_binary_baselines/StackOBot.exe.m55-stage2-{h[:8].upper()}'
if archive.exists():assert resume and sha(archive)==h
else:shutil.copy2(built,archive)
assert sha(archive)==h
lyra=Path(r'E:\UEProjects\LyraStarterGame\Plugins\AnomalyInjector\Binaries\Win64')
ldst=base/f'_binary_baselines/m55-stage2-r15-lyra-{sha(lyra/"UnrealEditor-AnomalyCapture.dll")[:8]}'
ldst.mkdir(exist_ok=resume);mods=[]
for p in lyra.iterdir():
    if p.is_file():
        shutil.copy2(p,ldst/p.name);assert sha(p)==sha(ldst/p.name)
        mods.append(dict(name=p.name,size=p.stat().st_size,sha256=sha(p)))
shutil.copy2(built,prior['staged']);assert sha(prior['staged'])==h
source=subprocess.check_output(['git','-C',str(base/'StackOBot/Plugins/AnomalyInjector'),'rev-parse','HEAD']).decode().strip()
tokens=['IAI-INPUT-LOCK ON','IAI.Bench.PlaceView','chg_gt8_max_sofar','no_labelled_frames','ChangeCase:','M55_INVENTED_NOT_A_REAL_SYMBOL']
binary=Path(prior['staged']).read_bytes()
scan={t:dict(ascii=binary.count(t.encode()),utf16=binary.count(t.encode('utf-16le'))) for t in tokens}
assert all(scan[t]['utf16']>0 for t in tokens[:-1]) and scan[tokens[-1]]['utf16']==0
result=dict(source_commit=source,built=str(built),staged=prior['staged'],archive=str(archive),size=built.stat().st_size,
    sha256=h,containers=prior['containers'],lyra_archive=str(ldst),lyra_modules=mods,a44=scan)
output=ev/sys.argv[2];assert not output.exists();output.write_text(json.dumps(result,indent=2))
print(json.dumps({k:v for k,v in result.items() if k not in ('containers','lyra_modules')},indent=2))
