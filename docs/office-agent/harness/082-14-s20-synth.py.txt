def committed(leg,m):

def override_len(leg,comp):
    T=template();n=[x['idx'] for x in T['slots'] if x['comp']==comp]
    if leg['role']=='C':return len(n)
    if leg['role']=='IDP':
        for c,s,br in T['idp']:
            if c==comp:return br
        return 0
    if leg['role']=='IDF':return max(n)+1 if n else 0
    return 0

def render(L):return '\n'.join(t if k is None else f'[2026.09.28-00.00.00:000][{k%1000:3d}]{t}' for k,t in L)+'\n'
def rows(leg,s,m,fi=None):
