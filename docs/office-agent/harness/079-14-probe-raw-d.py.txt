#!/usr/bin/env python3
"""Print the raw per-pair d series the gate computes, using the gate's OWN
machinery, so a NO-TRANSITION / suppressed-peak claim can be checked against
numbers rather than inferred."""
import os
import sys

sys.path.insert(0, r"C:\ClaudeTemp\079-14\wt\tools")
import verify_capture as V

cap = sys.argv[1]
lo, hi = int(sys.argv[2]), int(sys.argv[3])
mlo = V._offset_module()
events = V._load_gate_events(cap, mlo)
rows, _ = mlo.read_labels(os.path.join(cap, "labels.jsonl"))
paths = V._frame_paths(cap, mlo)
cache = mlo.FrameCache(paths, 0, limit=64)
size = cache.frame_size()
hot = V._HotCache(cache, V.DIFF_THRESH_DEFAULT)

ev = events[0]
node = ev["node"]
_a, _r, entry = V._anchor_entry(rows, ev["indices"], node, mlo)
runs = V._runs_of(ev["indices"])
print("event type=%s node=%s indices=%s runs=%s" % (ev["type"], node, ev["indices"], runs))
for (a, b) in runs:
    sig = V._PairSignal(cap, rows, node, entry, "mask", hot, paths, size, mlo,
                        V.MIN_VISIBLE_PX_DEFAULT, (a, b), V.EDGE_WINDOW_DEFAULT)
    print("\nrun [%d..%d] raw d per pair:" % (a, b))
    vals = {}
    for k in range(lo, hi + 1):
        p = sig.pair(k)
        vals[k] = p["d"]
        print("   pair %2d (frames %2d->%2d)  d=%s  err=%s"
              % (k, k - 1, k, "None" if p["d"] is None else "%.6f" % p["d"], p["error"]))
    tau = V.mlo.SIGNAL_FLOOR if False else 0.0040
    print("   above tau=%.4f : %s" % (tau, [k for k, d in sorted(vals.items()) if d and d > tau]))
    peaks = [k for k, d in sorted(vals.items())
             if d and d > tau and d >= V.PEAK_NEIGHBOUR_RATIO * min(vals.get(k - 1) or 0.0,
                                                                   vals.get(k + 1) or 0.0)]
    print("   qualify as one-sided peaks over this range: %s" % peaks)
    print("   ABOVE TAU BUT SUPPRESSED: %s"
          % [k for k, d in sorted(vals.items()) if d and d > tau and k not in peaks])
