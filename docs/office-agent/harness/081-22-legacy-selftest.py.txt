def load(lab):
    s=next((bank/('M55S2_'+lab)).glob('session*'));att=next(t for t in bank.glob('M55S2_'+lab+'_try*') if (t/s.name).exists());return s,att

    (tmp/f'reading-{name}.json').write_text(json.dumps(rec,default=str));(tmp/f'flat-{name}.json').write_text(json.dumps(dict(flat=fl,frames=fr),default=str))

    native_pairs_unequal=[(p['a'],p['b'],[f for f,v in p['fields_equal'].items() if not v]) for p in out['pairs']],colour=[(p['a'],p['b'],p['colour_frames_byte_equal']) for p in out['pairs']])

real=Path(r'D:\IntrusiveAnomalies\_reviews\081-22-evidence\legacy-historical-run-unique.json')

out,recs=lg.compare('N',[('K1',pre),('K2',b3),('K3',b3),('K4',pre)])
def rules(out):

                strict_reference=all(x['count']==0 for x in out['strict_vs_reference_pair']))
