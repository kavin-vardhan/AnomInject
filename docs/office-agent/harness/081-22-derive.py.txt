def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def derive(src,dst,pairs):
    t=(R/src).read_text(encoding='utf-8');done=[]
    for a,b in pairs:
        n=t.count(a);assert n==1,(src,a,n);t=t.replace(a,b);done.append(a)
    assert not (R/dst).exists(),dst
    (R/dst).write_bytes(t.encode('utf-8'))
    return dict(src=src,src_sha256=sha(R/src),dst=dst,dst_sha256=sha(R/dst),replacements=done)
