def judged_edges(events,thr=RELEASE):
    on=off=0;fails=[];nm=0;warm=0;conf=0
    for r in events:
        st=r.get('status')
        if st=='WARMUP':warm+=1
        if st=='CONFOUNDED-REFERENCE':conf+=1;continue
        if st not in ('OK','WARMUP'):continue
        if not r.get('measurable'):nm+=1;continue
        p=(r.get('per') or {}).get(thr) or {};ed=p.get('edges') or {}
        if isinstance(ed,list):ed=dict(start=ed[0].get('start'),end=ed[-1].get('end'),runs=ed) if ed else {}
        if ed.get('start') is not None:on+=1
        if ed.get('end') is not None:off+=1
        if p.get('verdict')=='FAIL':fails.append(dict(start_frame=r.get('start_frame'),edges=ed,reasons=p.get('fail_reasons')))
    return dict(onsets=on,offsets=off,fails=fails,not_measurable=nm,warmup=warm,confounded=conf)

def scramble_invisible(seed,ordinal,k=8):
    """Predicted visibility of a scramble pair on TC_UC1 (8x8 cells of the 4-colour pal(0) pattern): the number of cells
    whose source cell has a different colour. 0 changed cells = a predicted-invisible pair, excluded by name (DC plan 5 rule 3)."""
    a,b,ai=ORC.scramble_pair(seed,ordinal,k) if hasattr(ORC,'scramble_pair') else ORC.scramble_pair_py(seed,ordinal,k)
    col=lambda cx,cy:(cx+3*cy)%4
    changed=0
    for cy in range(k):
        for cx in range(k):
            j=cy*k+cx;i=(ai*((j-b)%(k*k)))%(k*k)
            if col(i%k,i//k)!=col(cx,cy):changed+=1
    return changed

def _count(fp):

def replay(draws,seed):
    """Independent replay of every Auto.Draw line: the KAT stream seeded like the capture, and per attempt the number of
    draws the logged counts imply (id over eligible, target over candidates, the hold, and one mode draw for m53 ids);
    the replayed seed_after must equal the logged one on every line."""
    if not draws:return dict(pass_=False,reason='no Auto.Draw lines')
    st=KAT.KatStream(int(seed));bad=[]
    for x in draws:
        if int(x['sb'])!=st.seed:bad.append(dict(attempt=x['attempt'],why='seed_before',want=st.seed,got=int(x['sb'])));st.seed=int(x['sb'])
        ne=_count(x['elig']);nc=_count(x['cand']);n=0
        if x['result']!='all_excluded' and ne>0:
            n=1
            if nc>0:n+=2+(1 if x['id'] in M53 else 0)
        for _ in range(n):st.mutate()
        if int(x['sa'])!=st.seed:
            bad.append(dict(attempt=x['attempt'],why='seed_after',want=int(x['sa']),got=st.seed,draws_modelled=n));st.seed=int(x['sa'])
    return dict(pass_=not bad,lines=len(draws),mismatches=bad[:6],note='per attempt: id over eligible, then target + hold + (m53) one mode draw when candidates > 0; the stream starts at the capture seed')

GRAMMAR=re.compile(r'^(scope=(view|all) candidates=\d+ cap_bytes=\d+ uv_modes=[a-z_+!]+ normal_modes=[a-z_+!]+|id=(uv_corruption|normal_corruption) eligible=\d+ refused=\d+ reasons=([a-z_]+:\d+(,[a-z_]+:\d+)*)?|end stats_unchanged=[01])\s*$')
def census_product(log):
    body=census_lines(log);blocks=[];cur=None
    for b in body:
        if b.startswith('scope='):cur=dict(head=b,ids={},end=None);blocks.append(cur)
        elif b.startswith('id=') and cur is not None:
            m=re.match(r'id=(\S+) eligible=(\d+) refused=(\d+) reasons=(\S*)',b)
            if m:cur['ids'][m.group(1)]=dict(eligible=int(m.group(2)),refused=int(m.group(3)),reasons={k:int(v) for k,v in (x.split(':') for x in m.group(4).split(',') if ':' in x)})
        elif b.startswith('end') and cur is not None:cur['end']=b
    bad=[b for b in body if not GRAMMAR.match(b)]
    return dict(blocks=blocks,grammar_violations=bad[:5],grammar_pass=not bad and bool(body))

def census_join(prod,bench):
    """The product census counts equal the bench census per-actor lines aggregated on the same view (default scope)."""
    if not prod['blocks']:return dict(pass_=False,reason='no product census block')
    b0=prod['blocks'][0];fam={'uv_corruption':'uv','normal_corruption':'normal'};out={}
    for pid,v in b0['ids'].items():
        f=fam.get(pid,pid)
        rows=[a[f] for a in bench['per_actor'].values() if f in a]
        el=sum(1 for r in rows if r['final']=='APPLY')
        out[pid]=dict(product=v,bench_rows=len(rows),bench_eligible=el,match=(len(rows)==v['eligible']+v['refused'] and el==v['eligible']))
    return dict(per_id=out,pass_=all(x['match'] for x in out.values()) if out else False)

def nd5_rate(legs):
    """ND-5: per mode, the share of MainWorld events judged NOT-MEASURABLE, at t50 and strict (the definition is declared
    in the README before any leg runs): an event is NOT-MEASURABLE when E1 finds no visible change inside its mask over its
    labelled frames at the threshold (its measurable flag is false), warm-up and confounded-reference events excluded."""
    per=collections.defaultdict(lambda:dict(events=0,not_measurable=0))
    for k,v in legs.items():
        if not isinstance(v,dict) or v.get('group')!='BM53MW' or v.get('role')=='NULL':continue
        mode=v.get('mode')
        for r in v.get('e1_events') or []:

    return {m:dict(d,rate=round(d['not_measurable']/d['events'],3) if d['events'] else None) for m,d in per.items()}

def doctor_copy(sd,variant,root):
    """A doctored copy of a session: images hardlinked, labels.jsonl and annotation.json rewritten so every m53 event's
    labelled set is moved by the variant (the can-fail (b) shapes). The original is never written."""
    import shutil
    sd=Path(sd);dst=Path(root)/f'{sd.name}__{variant}';
    if dst.exists():shutil.rmtree(dst)
    dst.mkdir(parents=True)
    for p in sd.rglob('*'):
        q=dst/p.relative_to(sd)
        if p.is_dir():q.mkdir(parents=True,exist_ok=True);continue
        if p.name in ('labels.jsonl','annotation.json'):continue
        q.parent.mkdir(parents=True,exist_ok=True)
        try:os.link(p,q)
        except OSError:shutil.copy2(p,q)
    an=json.loads((sd/'annotation.json').read_text(encoding='utf-8-sig'));rows=E9.read_jsonl(sd/'labels.jsonl');by={r['session_index']:r for r in rows}
    for a in an.get('anomalies') or []:
        if a.get('anomaly_type') not in M53:continue
        inj=(a.get('injected_frames') or {}).get('frame_indices') or []
        new=[x for x in _doctor_list(inj,variant) if x in by]
        for key in ('injected_frames','affected_frames'):
            if isinstance(a.get(key),dict):a[key]=dict(a[key],frame_indices=new,start_frame=min(new) if new else None,end_frame=max(new) if new else None,frame_count=len(new))
        tmpl=None;sf=None
        for si in inj:
            for e in by[si].get('anomalies') or []:
                if isinstance(e,dict) and e.get('id')==a['anomaly_type']:tmpl=dict(e);sf=e.get('start_frame');break
            if tmpl:break
        for si,r in by.items():
            ents=[e for e in (r.get('anomalies') or []) if isinstance(e,dict) and e.get('id')==a['anomaly_type'] and e.get('start_frame')==sf]
            if si in new and not ents and tmpl:r.setdefault('anomalies',[]).append(dict(tmpl,labelled=True));r['anomaly_present']=True
            for e in ents:e['labelled']=si in new
    (dst/'annotation.json').write_text(json.dumps(an),encoding='utf-8')
    (dst/'labels.jsonl').write_text('\n'.join(json.dumps(by[k]) for k in sorted(by))+'\n',encoding='utf-8')

def hosts_from_probes(cands,probes):
    """H1 / H2 per family = the candidates whose identity probe APPLIED and whose last decision is APPLY; a probe that
    applied on every decision ('clean') ranks before one that refused early (e.g. not_fully_resident while streaming),
    then C1 before C2. A family's cap is the largest raise among its chosen hosts, else the delivered 128 MiB."""
    out={};disp={}
    for k in ('U','N'):
        ok=[]
        for i in (1,2):
            cd=cands.get(f'C{i}{k}');pr=probes.get(f'P_HOST_FIRE_{i}{k}')
            if cd and pr and pr.get('usable'):ok.append((0 if pr.get('clean') else 1,i,cd))
        ok=[x[2] for x in sorted(ok,key=lambda x:(x[0],x[1]))]
        out[f'H1{k}']=ok[0]['actor'] if ok else None;out[f'H2{k}']=ok[1]['actor'] if len(ok)>1 else None
        rz=[x['cap_bytes'] for x in ok if x.get('tier')=='raise']
        out[f'cap_{k}']=max(rz) if rz else None
        disp[k]=dict(hosts=[(x['actor'],x['tier'],x.get('cap_bytes')) for x in ok],probes={f'C{i}{k}':(probes.get(f'P_HOST_FIRE_{i}{k}') or {}).get('heads') for i in (1,2)})
    out['cap_declared']=bool(out.get('cap_U') or out.get('cap_N'));out['_disp']=disp
    return out

def gt_map(on,nu):
    nt=sorted((float(r['t']),r['session_index']) for r in nu.rows if r.get('t') is not None);ts=[x[0] for x in nt]
    m={};dt={};gap=[]

        if r.get('t') is None or not ts:gap.append(si);continue
        t=float(r['t']);i=bisect.bisect_left(ts,t);best=None
        for j in (i-1,i):
            if 0<=j<len(ts):

        else:gap.append(si)

def arbiter_gt(on_sd,vsd,anomaly,thr=16):
    """G1-NAT arbiter at the same game time (AA-off legs): every labelled frame reads max |d| >= 16 inside the event mask
    against the null frame at the same t; the minimum per event is reported."""
    on=M53R.Sess(on_sd);vn=M53R.Sess(vsd);per=[];lows=[];miss=0
    for e in M53R.events(on,anomaly):
        vals=[]
        for si in M53R.labelled_sis(e):
            if si not in vn.by_si:miss+=1;continue
            reg=M53R.region([(on,si,e['mv'][si])])
            if reg is None or not reg.any():continue
            v=int(M53R.absdiff(on.img(si),vn.img(si))[reg].max());vals.append(v)
            if v<thr:lows.append(dict(start_frame=e['start_frame'],si=si,max=v))
        per.append(dict(start_frame=e['start_frame'],measured=len(vals),min_frame_max=min(vals) if vals else None))
    me=sum(1 for p in per if p['measured']>0)
    return dict(verdict='READS-WRONG' if me>0 and not lows else ('BELOW-16' if lows else 'NO-MEASURED-EVENT'),events=per,measured_events=me,below=lows[:10],unaligned_labelled=miss)
