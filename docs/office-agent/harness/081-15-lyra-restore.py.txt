from pathlib import Path
import hashlib,json,shutil,subprocess
B=Path(r'D:\IntrusiveAnomalies');E=B/'_reviews/081-15-evidence';L=Path(r'E:\UEProjects\LyraStarterGame\Plugins\AnomalyInjector')
def sha(p):
 with Path(p).open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()
def git(*a):return subprocess.check_output(['git','-C',str(L),*a]).decode().strip()
idle=subprocess.check_output(['powershell','-NoProfile','-Command','@(Get-Process -Name UnrealEditor,StackOBot,LyraGame,UnrealBuildTool -ErrorAction SilentlyContinue) | Select-Object -ExpandProperty Id']).decode().strip();assert not idle
s=json.loads((E/'lyra-staged.json').read_text());assert git('rev-parse','HEAD')==s['source_commit'];assert not git('diff','--name-only') and not git('diff','--cached','--name-only')
target=(L/'Binaries/Win64').resolve();original=B/'_binary_baselines/m55-stage1-lyra-entry-caa68c6'
for r in s['modules']:assert sha(target/r['name'])==r['sha256'] and sha(Path(s['archive'])/r['name'])==r['sha256']
for r in s['entry_modules']:assert sha(original/r['name'])==r['sha256']
git('checkout','--detach',s['entry_source'])
for r in s['entry_modules']:shutil.copy2(original/r['name'],target/r['name']);assert sha(target/r['name'])==r['sha256']
removed=[]
for name in ('UnrealEditor-AnomalyBench.dll','UnrealEditor-AnomalyBench.pdb'):
 p=(target/name).resolve();assert p.parent==target and p.name==name;p.unlink();removed.append(str(p))
assert not git('diff','--name-only') and not git('ls-files','--others','--exclude-standard')
(E/'lyra-restored.json').write_text(json.dumps(dict(source=git('rev-parse','HEAD'),restored_modules=s['entry_modules'],removed_exact_files=removed),indent=2))
print('Lyra entry source/original9modules restored; exact2candidate-only Bench files removed.')
