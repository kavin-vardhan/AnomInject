from pathlib import Path
import json,collections
base=Path(r'D:/IntrusiveAnomalies');ev=base/'_reviews/081-06-evidence'
prior=json.loads((ev/'prior-accepted-and-failed.json').read_text());attempts=json.loads((ev/'qualification-attempts.json').read_text())
accepted=prior['accepted_by_chat'][:]
for r in attempts:
 if r['status']=='PASS':
  r=dict(r);r['fixture']='LYRA';r['qualification_status']='PASS-L1-and-gate';accepted.append(r)
missing=[];table=['| Gate | Stack N/S valid pairs | Lyra N/S valid pairs |','|---|---:|---:|']
for g in range(1,16):
 cells=[]
 for fixture in ('STACK','LYRA'):
  cell=[]
  for order in ('N','S'):
   r=next((r for r in accepted if r['fixture']==fixture and r['gate']==g and r['order']==order),None)
   if not r:missing.append(dict(fixture=fixture,gate=g,order=order))
   cell.append(str(r['reasons'].get('identity_valid',0)) if r else 'UNQUALIFIED')
  cells.append('/'.join(cell))
 table.append(f'| {g} | {cells[0]} | {cells[1]} |')
out=dict(accepted=len(accepted),new_accepted=sum(r['status']=='PASS' for r in attempts),new_attempts=len(attempts),missing=missing,prior_accepted=prior['accepted_by_chat'],original_failed=prior['original_failed'],attempts=attempts)
(ev/'final-ledger.json').write_text(json.dumps(out,indent=2));(ev/'matrix.md').write_text('\n'.join(table)+'\n')
print(json.dumps({k:v for k,v in out.items() if k not in ('prior_accepted','original_failed','attempts')},indent=2))
