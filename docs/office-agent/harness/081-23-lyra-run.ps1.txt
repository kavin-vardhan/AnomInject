param([Parameter(Mandatory=$true)][string]$Label, [string]$Anomaly='null_effect', [int]$MaxFrames=90, [string]$ExtraExecCmds='', [string]$Config='2 4 16 4 0', [int]$Order=0, [string]$SceneMode="", [int]$DelayFrames=0, [int]$ResX=1280, [int]$ResY=720, [string]$Target="Cube4", [int]$Seed=4242, [switch]$AutoPool)
$FixtureOn=$true
$ErrorActionPreference='Stop'
$root=$env:IAI_R23_ROOT;if(-not $root -or -not (Test-Path -LiteralPath $root)){throw 'IAI_R23_ROOT unset or missing'}
$dir=Join-Path $root $Label
if(Test-Path -LiteralPath $dir){throw 'Existing evidence label'}
if(Get-Process -Name UnrealEditor,StackOBot,LyraGame,UnrealBuildTool -ErrorAction SilentlyContinue){throw 'Bench not idle'}
New-Item -ItemType Directory -Path $dir | Out-Null
$exe='D:\UESource\UnrealEngine\Engine\Binaries\Win64\UnrealEditor.exe'
$log=Join-Path $dir 'game.log'
function Check-Foreign($where){$fe=@(Get-Process -Name UnrealEditor -ErrorAction SilentlyContinue|Where-Object{(-not $p) -or ($_.Id -ne $p.Id)});if($fe.Count){$record.foreign_editor=@($fe|ForEach-Object{@{pid=$_.Id;path=$_.Path;start=$_.StartTime.ToString('o')}});$record.foreign_where=$where;throw "FOREIGN-EDITOR-APPEARED $where"}}
Add-Type @"
using System; using System.Runtime.InteropServices;
public class R1Focus {
 [DllImport("user32.dll")] public static extern IntPtr GetForegroundWindow();
 [DllImport("user32.dll")] public static extern uint GetWindowThreadProcessId(IntPtr h,out uint pid);
 [DllImport("user32.dll")] public static extern bool SetForegroundWindow(IntPtr h);
 [DllImport("user32.dll")] public static extern bool BringWindowToTop(IntPtr h);
 [DllImport("user32.dll")] public static extern bool ShowWindow(IntPtr h,int cmd);
 [DllImport("user32.dll")] public static extern void keybd_event(byte k,byte s,uint f,UIntPtr e);
 [StructLayout(LayoutKind.Sequential)] public struct Point { public int X; public int Y; }
 [DllImport("user32.dll")] public static extern bool GetCursorPos(out Point p);
 [DllImport("kernel32.dll")] public static extern bool GetSystemTimes(out long idle,out long kernel,out long user);
 public static string SysTimes(){long i,k,u;GetSystemTimes(out i,out k,out u);return i+","+k+","+u;}
 public static uint FgPid(){uint p;GetWindowThreadProcessId(GetForegroundWindow(),out p);return p;}
 public static string Cursor(){Point p;GetCursorPos(out p);return p.X+","+p.Y;}
 public static void Force(IntPtr h){ShowWindow(h,9);keybd_event(0x12,0,0,UIntPtr.Zero);keybd_event(0x12,0,2,UIntPtr.Zero);SetForegroundWindow(h);BringWindowToTop(h);}
}
"@
$cmds='r.DefaultFeature.AutoExposure, r.EyeAdaptationQuality, r.AntiAliasingMethod, r.ScreenPercentage, r.CustomDepth, r.Nanite, IAI.Capture.SVE 1, IAI.Capture.Delivery 0, IAI.Capture.Pace 1, IAI.Capture.Fps 30, IAI.Capture.Config 2 4 16 4 0, IAI.Bench.ChangeEvidenceCases 1, IAI.Bench.ChangeGate 14, IAI.Bench.SynthTickOrder 0, IAI.Server.Start'
$cmds=$cmds.Replace('IAI.Capture.Config 2 4 16 4 0',"IAI.Capture.Config $Config").Replace('IAI.Bench.SynthTickOrder 0',"IAI.Bench.SynthTickOrder $Order")
if($ExtraExecCmds){$cmds+=', '+$ExtraExecCmds}
$argsList=@('E:\UEProjects\LyraStarterGame\LyraStarterGame.uproject','/ShooterCore/Maps/L_ShooterGym','-game','-windowed',"-ResX=$ResX","-ResY=$ResY","-ExecCmds=`"$cmds`"",'-unattended','-nosplash','-IAIBenchFixture',"-abslog=$log")
$record=@{label=$Label;args=$argsList;exe_sha256=(Get-FileHash -LiteralPath $exe -Algorithm SHA256).Hash.ToLower();started_utc=[DateTime]::UtcNow.ToString('o');cursor_before=[R1Focus]::Cursor();capture_start_sent=$false;fixture_on=[bool]$FixtureOn;focus=@()}
$record | ConvertTo-Json -Depth 8 | Set-Content -LiteralPath (Join-Path $dir 'launch.json') -Encoding utf8
$p=$null;$ws=$null;$count=0
try {
 # The bench's foreground-focus gate requires a visible game window.
 $p=Start-Process -FilePath $exe -ArgumentList $argsList -PassThru
 $record.pid=$p.Id
 $clock=[Diagnostics.Stopwatch]::StartNew();$token=$null;$sh=New-Object -ComObject WScript.Shell
 while($clock.Elapsed.TotalSeconds -lt 120){
  if($p.HasExited){throw 'Game exited during startup'}
  Check-Foreign 'startup'
  try{$sh.AppActivate($p.Id)|Out-Null}catch{}
  $p.Refresh()
  if($p.MainWindowHandle -ne 0 -and [R1Focus]::FgPid() -ne $p.Id){[R1Focus]::Force($p.MainWindowHandle)}
  if(Test-Path -LiteralPath $log){$t=[string](Get-Content -LiteralPath $log -Raw);$m=[regex]::Match($t,'Control server token:\s*(\S+)\s*\(');if($m.Success -and $t -match 'Bringing World .*up for play'){$token=$m.Groups[1].Value;break}}
  Start-Sleep -Milliseconds 200
 }
 if(-not $token){throw 'No server token / world readiness'}
 $record.ready_seconds=$clock.Elapsed.TotalSeconds
 $ct=[Threading.CancellationToken]::None;$ws=[Net.WebSockets.ClientWebSocket]::new()
 $ws.ConnectAsync([Uri]'ws://127.0.0.1:8077',$ct).GetAwaiter().GetResult()
 function Send($o){$b=[Text.Encoding]::UTF8.GetBytes(($o|ConvertTo-Json -Compress));[void]$ws.SendAsync([ArraySegment[byte]]::new($b),[Net.WebSockets.WebSocketMessageType]::Text,$true,$ct).GetAwaiter().GetResult()}
 function Recv {
  $buf=[byte[]]::new(262144);$mem=[IO.MemoryStream]::new();$cts=[Threading.CancellationTokenSource]::new(5000)
  try {do {$r=$ws.ReceiveAsync([ArraySegment[byte]]::new($buf),$cts.Token).GetAwaiter().GetResult();if($r.MessageType -eq [Net.WebSockets.WebSocketMessageType]::Close){throw 'Server closed connection'};$mem.Write($buf,0,$r.Count)}while(-not $r.EndOfMessage);return [Text.Encoding]::UTF8.GetString($mem.ToArray())}finally{$cts.Dispose();$mem.Dispose()}
 }
 Send @{type='hello';token=$token}
 $auth=(Recv)|ConvertFrom-Json
 if($auth.type -ne 'welcome'){throw 'Authentication failed'}
 function BenchRequest($request) {
  Send $request
  for($j=0;$j -lt 8;$j++){$reply=(Recv)|ConvertFrom-Json;if($reply.type -eq $request.type){if(-not $reply.executed){throw "Bench command not executed: $($request.type)"};return}}
  throw "Bench command receipt missing: $($request.type)"
 }
 if($FixtureOn){
  BenchRequest @{type='bench_input_lock';enabled=$true}
  $lockLog=[string](Get-Content -LiteralPath $log -Raw)
  if($lockLog -notmatch 'IAI-INPUT-LOCK ON .*lookIgnored=1 moveIgnored=1'){throw 'Input lock not observed'}
  $record.lock_confirmed_utc=[DateTime]::UtcNow.ToString('o')
 }
 $settle=[Diagnostics.Stopwatch]::StartNew()
 while($settle.Elapsed.TotalSeconds -lt 45){try{$sh.AppActivate($p.Id)|Out-Null}catch{};$p.Refresh();if([R1Focus]::FgPid() -ne $p.Id){[R1Focus]::Force($p.MainWindowHandle)};Start-Sleep -Milliseconds 200}
 $record.focus_before=([R1Focus]::FgPid() -eq $p.Id);$record.cursor_before_snapshots=[R1Focus]::Cursor()
 # Record a focus failure, but still collect the requested diagnostic pose/B1 data.
 if($FixtureOn){
  BenchRequest @{type='bench_place_view';request=1}
  $placeLog=[string](Get-Content -LiteralPath $log -Raw)
  if($placeLog -notmatch 'IAI-L2 PLACED request=1 ok=1'){throw 'Placement success not observed'}
  $record.placement_confirmed_utc=[DateTime]::UtcNow.ToString('o')
 }
 Start-Sleep -Milliseconds 700
 $beforeSampling=[string](Get-Content -LiteralPath $log -Raw)
 if($beforeSampling -match '=== Handled ensure: ==='){throw 'Engine handled ensure before sampling'}
 $record.sampling_started_utc=[DateTime]::UtcNow.ToString('o')
 Send @{type='subscribe';channels=@('snapshot');snapshotHz=1}
 $samples=[IO.StreamWriter]::new((Join-Path $dir 'snapshots.jsonl'),$false,[Text.UTF8Encoding]::new($false))
 try {
  while($count -lt 1){
   $raw=Recv;$row=$raw|ConvertFrom-Json
   if($row.type -ne 'snapshot'){continue}
   $samples.WriteLine(($row|ConvertTo-Json -Depth 20 -Compress));$samples.Flush()
   $record.focus+=@{sample_index=$count;foreground_pid=[R1Focus]::FgPid();cursor=[R1Focus]::Cursor()}
   if($row.capture.running -or $row.capture.framesWritten -ne 0 -or @($row.active).Count -ne 0){throw 'Capture or injection active during preflight'}
   $count++
   if([R1Focus]::FgPid() -ne $p.Id){try{$sh.AppActivate($p.Id)|Out-Null}catch{};$p.Refresh();[R1Focus]::Force($p.MainWindowHandle)}
  }
 }finally{$samples.Dispose()}
 if($count -ne 1){throw 'Incomplete snapshot sample count'}
 Send @{type='subscribe';channels=@()}
 # Independent L1 camera snapshot and full timestamp L2 audit before capture.
 $record.status='COMPLETE';$record.snapshot_count=$count
 $record|ConvertTo-Json -Depth 12|Set-Content -LiteralPath (Join-Path $dir 'launch.json') -Encoding utf8
 & C:\Python313\python.exe D:\IntrusiveAnomalies\_reviews\081-23-lyra-pre.py $Label $ResX $ResY
 if($LASTEXITCODE -ne 0){throw 'PRECAPTURE-L1-INVALID'}
 BenchRequest @{type='bench_log_barrier'}
 & C:\Python313\python.exe D:\IntrusiveAnomalies\_reviews\081-08-l2.py $log (Join-Path $dir '_l2_pre.json') pre
 if($LASTEXITCODE -ne 0){throw 'PRECAPTURE-L2-INVALID'}
 if($SceneMode){
  BenchRequest @{type='bench_scene_fixture';mode=$SceneMode}
  $sceneLog=[string](Get-Content -LiteralPath $log -Raw)
  if($sceneLog -notmatch ('IAI-SCENE '+$SceneMode.ToUpper()+' ')){throw 'Scene fixture success not observed'}
 }
 Check-Foreign 'before capture_start'
 $captureDir="D:\IntrusiveAnomalies\_bench_sessions_bank\M55B3R23_$Label"
 if(Test-Path -LiteralPath $captureDir){throw 'Capture directory already exists'}
 $record.capture_dir=$captureDir;$record.capture_focus=@();$record.capture_start_sent=$true
 $record.capture_start_utc=[DateTime]::UtcNow.ToString('o')
 $record.capture_focus+=@{seconds=0;foreground_pid=[R1Focus]::FgPid()}
 $captureRequest=@{type='capture_start';dir=$captureDir;format='png';seed=$Seed;maxFrames=$MaxFrames}
 if(-not $AutoPool -and $Anomaly -and $Target){$captureRequest.anomaly=$Anomaly;$captureRequest.target=$Target}
 Send $captureRequest
 $record.anchor_start=@{qpc=[Diagnostics.Stopwatch]::GetTimestamp();utc=[DateTime]::UtcNow.ToString('o');freq=[Diagnostics.Stopwatch]::Frequency}
 $record.capture_samples=@();$nextSample=0.0;$nextForeign=0.0
 $runClock=[Diagnostics.Stopwatch]::StartNew();$session=$null;$stable=0;$previousShape='';$summarySeen=$false
 while($runClock.Elapsed.TotalSeconds -lt 600){
  if($p.HasExited){throw 'Game exited during capture'}
  if($runClock.Elapsed.TotalSeconds -ge $nextForeign){$nextForeign+=1.0;Check-Foreign ('capture t='+[math]::Round($runClock.Elapsed.TotalSeconds,1))}
  if($runClock.Elapsed.TotalSeconds -ge $nextSample){$nextSample+=1.0;$gp=Get-Process -Id $p.Id -ErrorAction SilentlyContinue;$scw=@(Get-Process -Name ShaderCompileWorker -ErrorAction SilentlyContinue);$scwMs=0.0;foreach($x in $scw){try{$scwMs+=$x.TotalProcessorTime.TotalMilliseconds}catch{}};$row=@{s=$runClock.Elapsed.TotalSeconds;qpc=[Diagnostics.Stopwatch]::GetTimestamp();utc=[DateTime]::UtcNow.ToString('o');sys=[R1Focus]::SysTimes();scw=$scw.Count;scw_cpu_ms=$scwMs};if($gp){$row.cpu_ms=$gp.TotalProcessorTime.TotalMilliseconds;$row.ws=$gp.WorkingSet64;$row.priv=$gp.PrivateMemorySize64;$row.threads=$gp.Threads.Count};$record.capture_samples+=$row}
  $fg=[R1Focus]::FgPid();$record.capture_focus+=@{seconds=$runClock.Elapsed.TotalSeconds;foreground_pid=$fg}
  if($fg -ne $p.Id){try{$sh.AppActivate($p.Id)|Out-Null}catch{};$p.Refresh();[R1Focus]::Force($p.MainWindowHandle)}
  $session=Get-ChildItem -LiteralPath $captureDir -Directory -ErrorAction SilentlyContinue | Where-Object {Test-Path -LiteralPath (Join-Path $_.FullName 'run_summary.json')} | Select-Object -First 1
  if($session){
   $summarySeen=$true
   $files=@(Get-ChildItem -LiteralPath (Join-Path $session.FullName 'Actual_Frames') -File)
   $zeros=@($files|Where-Object {$_.Length -eq 0}).Count
   $shape="$($files.Count):$(($files|Measure-Object -Property Length -Sum).Sum)"
   if($shape -eq $previousShape -and $zeros -eq 0){$stable++}else{$stable=0};$previousShape=$shape
   if($stable -ge 25){break}
  }
  Start-Sleep -Milliseconds 50
 }
 $record.anchor_end=@{qpc=[Diagnostics.Stopwatch]::GetTimestamp();utc=[DateTime]::UtcNow.ToString('o');freq=[Diagnostics.Stopwatch]::Frequency}
 if(-not $summarySeen -or $stable -lt 25){throw 'Capture summary/writer tail did not complete'}
 $record.session=$session.FullName;$record.capture_finished_utc=[DateTime]::UtcNow.ToString('o')
 & C:\Python313\python.exe D:\IntrusiveAnomalies\_reviews\081-08-l2.py $log (Join-Path $dir '_l2_post.json') post
 $record.post_l2_ok=($LASTEXITCODE -eq 0)
 # Run-end/teardown must release the owned input increment. Game log is flushed by that release.
 $endLog=[string](Get-Content -LiteralPath $log -Raw)
 if($endLog -notmatch 'IAI-INPUT-LOCK OFF reason=(run_end|world_cleanup)'){throw 'Run-end input unlock absent'}
 $record.status='COMPLETE';$record.finished_seconds=$clock.Elapsed.TotalSeconds
}catch{$record.status='PROTOCOL-FAIL';$record.error=$_.Exception.Message;throw}
finally {
 if($ws){$ws.Dispose()}
 if($p -and -not $p.HasExited){Stop-Process -Id $p.Id -Force;$p.WaitForExit()}
 $record.snapshot_count=$count
 $record.finished_utc=[DateTime]::UtcNow.ToString('o');$record|ConvertTo-Json -Depth 12|Set-Content -LiteralPath (Join-Path $dir 'launch.json') -Encoding utf8
}
Write-Output "$Label snapshots=$count capture_start_sent=$($record.capture_start_sent)"
