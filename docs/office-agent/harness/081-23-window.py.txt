from pathlib import Path

R=Path(r'D:\IntrusiveAnomalies\_reviews');os.environ.update(IAI_R23_WINDOW=W,IAI_R23_BUDGET_TAG=tag,PYTHONDONTWRITEBYTECODE='1');env=dict(os.environ);PY=sys.executable
