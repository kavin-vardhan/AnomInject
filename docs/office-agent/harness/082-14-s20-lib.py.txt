    res=dict(readable=False,complete=False,why='',eff_mismatch=[],raw_changed=[],len_changed=[],resolved_changed=[],n_slots=0,comps=[])

        res['why']=f"slot keys differ or incomplete (pre {len(p['slots'])}/{p['slots_declared']}, post {len(q['slots'])}/{q['slots_declared']})";return res

def read_cell(get,s):
    frames=s['frames'];per=[];ev0=s['event'][0];t0=s['tail'][0]
    for si in range(frames):
        a=get('arm',si);n1=get('N1',si);n2=get('N2',si)
        per.append(dict(si=si,k_arm_n1=kpair(a,n1),k_arm_n2=kpair(a,n2),k_n1_n2=kpair(n1,n2),r_arm_n1=rpair(a,n1),r_arm_n2=rpair(a,n2),r_n1_n2=rpair(n1,n2)))
    tail=[p for p in per if p['si']>=t0];nwin=[p for p in per if p['si']>=ev0]
    nulls_ok=all(p['k_n1_n2'][0]<=TOL for p in nwin)
    present=all(p['k_arm_n1'][1]>=MIN_PX and p['k_arm_n2'][1]>=MIN_PX for p in tail)
    absent=all(p['k_arm_n1'][0]<=TOL and p['k_arm_n2'][0]<=TOL for p in tail)
    state='INCONCLUSIVE' if not nulls_ok else 'present' if present else 'absent' if absent else 'partial'
    fr=next((p for p in per if p['r_arm_n1'][1]>=MIN_PX and p['r_arm_n2'][1]>=MIN_PX and p['r_n1_n2'][1]<MIN_PX),None)
    fu=next((p for p in per if p['r_arm_n1'][1]>=MIN_PX and p['r_arm_n2'][1]>=MIN_PX),None)
    return dict(state=state,nulls_ok=nulls_ok,null_window=[ev0,frames-1],tail=[t0,frames-1],region_r_rule=REGION_R_RULE,
                region_r_first_unconditioned=fu['si'] if fu else None,
                null_max_window=max(p['k_n1_n2'][0] for p in nwin),null_max_all=max(p['k_n1_n2'][0] for p in per),
                tail_min_px16=min(min(p['k_arm_n1'][1],p['k_arm_n2'][1]) for p in tail),tail_max_max=max(max(p['k_arm_n1'][0],p['k_arm_n2'][0]) for p in tail),
                partial_si=[p['si'] for p in tail if not (p['k_arm_n1'][1]>=MIN_PX and p['k_arm_n2'][1]>=MIN_PX)] if state=='partial' else [],
                region_r_first=dict(si=fr['si'],arm_n1=fr['r_arm_n1'],arm_n2=fr['r_arm_n2'],n1_n2=fr['r_n1_n2']) if fr else None,per_si=per)

def flags(cells,cross,states,fps,disp):
    st=lambda k:(cells.get(k) or {}).get('state')
    def ok(*ks):return all(st(k) not in (None,'INCONCLUSIVE') for k in ks)
    F={};notes=[]
    if ok('A_IDP_ON'):F['REPRODUCED']=st('A_IDP_ON')=='present';F['NOT-REPRODUCED']=st('A_IDP_ON')=='absent'
    for arm,on,off in (('IDP','A_IDP_ON','A_IDP_OFF'),('IDF','B_IDF_ON','B_IDF_OFF'),('C','B_C_ON','B_C_OFF'),('R','B_R_ON','B_R_OFF')):
        if ok(on,off):F[f'CACHE-DEPENDENT[{arm}]']=st(on)=='present' and st(off)=='absent'
    if ok('B_C_ON'):F['REPRODUCED-BY-SWAP']=st('B_C_ON')=='present'
    fmatch=(fps.get('B_IDF_ON',{}).get('committed')==30 and fps.get('B_C_ON',{}).get('slots')==30 and fps.get('B_C_ON',{}).get('components')==12)
    for cache in ('ON','OFF'):
        ci,cc=f'B_IDF_{cache}',f'B_C_{cache}';x=cross.get(f'C_vs_IDF_{cache}')
        if ok(ci,cc) and x is not None:
            F[f'SAME-TRACE-AS-ID-F[{cache}]']=st(cc)=='present' and st(ci)=='present' and x['same_le2']
            F[f'DIFFERENT-TRACE[{cache}]']=st(cc)=='present' and not x['same_le2']
    F['FOOTPRINT-MATCHED']=fmatch
    if not fmatch:notes.append('FOOTPRINT-UNMATCHED: ID-F did not commit 30/30 or C did not override 30 slots on 12 components; the ID-F-vs-C comparison is reported but cannot carry a class')
    xr=cross.get('R_vs_IDF_ON')
    if ok('B_R_ON','B_IDF_ON') and xr is not None:F['REPRODUCED-BY-REFRESH']=st('B_R_ON')=='present' and xr['same_le2']
    if ok('B_IDF_ON','B_C_ON','B_R_ON'):F['SPECIFIC-TO-M53-PATH']=st('B_IDF_ON')=='present' and st('B_C_ON')=='absent' and st('B_R_ON')=='absent'
    cls=classify(F,states,disp,fmatch);return dict(flags=F,classification=cls,notes=notes)

def classify(F,states,disp,fmatch):
    out=[];why={}
    if F.get('NOT-REPRODUCED'):out.append('NOT-REPRODUCED');why['NOT-REPRODUCED']='ID-P cache on reads absent; report and stop, the bank reading stays a banked observation'
    swap=F.get('REPRODUCED-BY-SWAP');same=F.get('SAME-TRACE-AS-ID-F[ON]');refr=F.get('REPRODUCED-BY-REFRESH')
    if swap:
        conj=dict(same_trace=bool(same),footprint_matched=bool(fmatch),not_refresh=(refr is False),c_effective_mismatch=eff_mismatch(states,'B_C_ON_C'),idf_effective_mismatch=eff_mismatch(states,'B_IDF_ON_IDF'))
        if all(conj.values()):out.append('PLUGIN-WIDE-RESTORE-DEFECT');why['PLUGIN-WIDE-RESTORE-DEFECT']=conj
        else:
            out.append('REPRODUCED-BY-SWAP, CAUSE UNRESOLVED')
            why['REPRODUCED-BY-SWAP, CAUSE UNRESOLVED']=dict(failed=[k for k,v in conj.items() if not v],left_or_unresolved_counts={k:left_or_unresolved(disp.get(k) or {}) for k in ('B_C_ON_C','B_IDF_ON_IDF')},rule=M1_RULE)
    if refr:
        if eff_equal(states,'B_IDF_ON_IDF'):out.append('STALE-NULL');why['STALE-NULL']='REPRODUCED-BY-REFRESH and ID-F post-revert Effective == pre-event on every slot'
        else:why['STALE-NULL (not established)']='REPRODUCED-BY-REFRESH, but ID-F state is '+('mismatched' if eff_mismatch(states,'B_IDF_ON_IDF') else 'unreadable or incomplete')
    if F.get('SPECIFIC-TO-M53-PATH'):
        if eff_mismatch(states,'B_IDF_ON_IDF'):out.append('M53-RESTORE-DEFECT');why['M53-RESTORE-DEFECT']=dict(eff_mismatch=states['B_IDF_ON_IDF']['eff_mismatch'][:5])
        elif eff_equal(states,'B_IDF_ON_IDF'):
            out.append('M53-SPECIFIC, CAUSE NOT IN SLOT STATE');why['M53-SPECIFIC, CAUSE NOT IN SLOT STATE']=dict(left_or_unresolved_count=left_or_unresolved(disp.get('B_IDF_ON_IDF') or {}),rule=M1_RULE)
        else:out.append('M53-SPECIFIC, CAUSE UNRESOLVED (state unreadable)')
    if not out:out.append('UNCLASSIFIED')
    order=['NOT-REPRODUCED','PLUGIN-WIDE-RESTORE-DEFECT','M53-RESTORE-DEFECT','STALE-NULL','M53-SPECIFIC, CAUSE NOT IN SLOT STATE','M53-SPECIFIC, CAUSE UNRESOLVED (state unreadable)','REPRODUCED-BY-SWAP, CAUSE UNRESOLVED','UNCLASSIFIED']
    return dict(primary=sorted(out,key=order.index)[0],all=sorted(out,key=order.index),why=why,m1_rule=M1_RULE)
