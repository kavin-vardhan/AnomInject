"""090-10b label edges from pixels, independent of masks. For each event: the first frame whose drawn box differs from the clean
reference (two frames before the label) by >= the event threshold, and the first frame after that which no longer differs; the
onset offset is that first changed frame minus the first labelled frame, the end offset is the first clean frame minus the
frame after the last labelled one. Threshold per event = max(4.0, 4 x the clean-to-clean null), as in 090-10b-common.pixels."""
import importlib.util, json, sys
from pathlib import Path
sys.dont_write_bytecode = True
_s = importlib.util.spec_from_file_location('c910b', str(Path(__file__).with_name('090-10b-common.py')))
c = importlib.util.module_from_spec(_s); _s.loader.exec_module(c)


def edges(sess):
    sess = Path(sess)
    ann = json.loads((sess / 'annotation.json').read_text(encoding='utf-8'))
    rows = {}
    for ln in (sess / 'labels.jsonl').read_text(encoding='utf-8').splitlines():
        if ln.strip():
            r = json.loads(ln); rows[r['session_index']] = r
    out = []
    for e in ann.get('anomalies', []):
        fr = (e.get('affected_frames') or {}).get('frame_indices') or []
        tgt = ((e.get('affected_objects') or {}).get('nodes') or [{}])[0].get('name')
        rec = dict(type=e.get('anomaly_type'), target=tgt, label=[fr[0], fr[-1]] if fr else None)
        boxes = []
        for i in fr:
            for a in (rows.get(i) or {}).get('anomalies', []):
                if a.get('target_name') == tgt:
                    bx = a.get('bbox_drawn_px') or (a.get('bbox_px') if a.get('bbox_valid') else None)
                    if bx and bx[2] > 0 and bx[3] > 0:
                        boxes.append(bx)
        if not fr or not boxes or (fr[0] - 2) not in rows or (fr[-1] + 3) not in rows:
            rec['verdict'] = 'UNJUDGED'; out.append(rec); continue
        x0 = min(b[0] for b in boxes); y0 = min(b[1] for b in boxes)
        box = (x0, y0, max(b[0] + b[2] for b in boxes) - x0, max(b[1] + b[3] for b in boxes) - y0)
        ref = c._img(sess / rows[fr[0] - 2]['image'])
        null = c._bbox_diff(c._img(sess / rows[fr[0] - 1]['image']), ref, box)
        thr = max(c.T_ABS, c.T_REL * max(null or 0.0, c.NULL_FLOOR))
        span = list(range(fr[0] - 1, fr[-1] + 4))
        d = {i: c._bbox_diff(c._img(sess / rows[i]['image']), ref, box) for i in span if i in rows}
        changed = [i for i in span if d.get(i) is not None and d[i] >= thr]
        if not changed:
            rec.update(verdict='NO-CHANGE', thr=round(thr, 2)); out.append(rec); continue
        first = changed[0]
        after = next((i for i in span if i > first and d.get(i) is not None and d[i] < thr), None)
        rec.update(thr=round(thr, 2), onset=first - fr[0], end=(after - (fr[-1] + 1)) if after is not None else None,
                   diffs=[round(d[i], 1) if d.get(i) is not None else None for i in span])
        rec['verdict'] = 'IN-SYNC' if rec['onset'] == 0 and rec['end'] == 0 else 'OFFSET'
        out.append(rec)
    return out


if __name__ == '__main__':
    for p in sys.argv[1:]:
        res = edges(p)
        for r in res:
            print(json.dumps({k: v for k, v in r.items() if k != 'diffs'}))
        n = len(res); ok = sum(1 for r in res if r.get('verdict') == 'IN-SYNC')
        print('EDGES %s: %d of %d events in sync at both edges (onset 0, end 0)' % (Path(p).parent.name, ok, n))
