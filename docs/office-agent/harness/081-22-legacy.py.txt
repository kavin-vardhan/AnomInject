def keys(v):
    if isinstance(v,dict):return sorted((k,keys(x)) for k,x in v.items())
    if isinstance(v,list):return sorted(set(json.dumps(keys(x)) for x in v))
    return None
def flat(v,p,out):
    if isinstance(v,dict):
        for k,x in v.items():flat(x,p+'/'+k,out)
    elif isinstance(v,list):
        for i,x in enumerate(v):flat(x,p+f'[{i}]',out)
    else:out[p]=v

FIELDS_D=['counts','camera','onsets','label_keys','annotation_keys','summary_keys','run_keys','event_ranges','mask_ties','masks']
FIELDS_C=['frame_index_set','frames_written','counts','camera','onsets','label_keys','annotation_keys','summary_keys_nonchange','run_keys','event_ranges','mask_ties','occupancy','sve_want_cmp']
def compare(group,legs):
    recs={l:json.loads((E/f'reading-{l}.json').read_text()) for l,_ in legs}
    fl={l:json.loads((E/f'flat-{l}.json').read_text()) for l,_ in legs}
    for l in recs:recs[l]['sve_want_cmp']=[{k:v for k,v in x.items() if k in ('marksIssued','wantedMatches','submitsIssued','framesWritten','pendingWantedAtEnd','maxPendingDepth')} for x in recs[l]['sve_want']]
    names=[l for l,_ in legs];A=[l for l,w in legs if w is pre];Bn=[l for l,w in legs if w is not pre]
    fields=FIELDS_D if MODE=='d' else FIELDS_C
    pairs=[]
    for x,y in itertools.combinations(names,2):
        same=(x in A)==(y in A)
        eq={f:recs[x][f]==recs[y][f] for f in fields}
        dx,dy=fl[x]['flat'],fl[y]['flat']
        diff=sorted(p for p in set(dx)|set(dy) if dx.get(p)!=dy.get(p))
        fx,fy=fl[x]['frames'],fl[y]['frames']
        frames_equal=sum(1 for k in fx if fy.get(k)==fx[k])
        pairs.append(dict(a=x,b=y,same_binary=same,fields_equal=eq,diff_paths=diff,colour_frames_byte_equal=[frames_equal,len(fx),len(fy)]))
    ctrl=[p for p in pairs if p['same_binary']];cross=[p for p in pairs if not p['same_binary']]
    run_unique=set();unstable_fields=set()
    for p in ctrl:
        run_unique|=set(p['diff_paths'])
        unstable_fields|={f for f,v in p['fields_equal'].items() if not v}
    norm=lambda s:re.sub(r'\[\d+\]','[]',s)
    ru_norm={norm(p) for p in run_unique}
    res=[]
    for p in cross:
        extra=[q for q in p['diff_paths'] if q not in run_unique]
        extra_norm=[q for q in extra if norm(q) not in ru_norm]
        gated={f:v for f,v in p['fields_equal'].items() if f not in unstable_fields}
        res.append(dict(a=p['a'],b=p['b'],extra_paths=extra[:40],extra_paths_count=len(extra),extra_paths_not_even_by_pattern=extra_norm[:40],gated_fields_unequal=[f for f,v in gated.items() if not v]))
    skip=(lambda q:q.startswith('run_summary/change_')) if MODE=='c' else (lambda q:False)
    npaths={l:{norm(q) for q in fl[l]['flat'] if not skip(q)} for l in names}
    pre_union=set().union(*[npaths[l] for l in A]);b3_union=set().union(*[npaths[l] for l in Bn])
    key_presence=dict(added_in_b3={l:sorted(npaths[l]-pre_union)[:40] for l in Bn},removed_in_b3={l:sorted(npaths[l]-b3_union)[:40] for l in A})
    key_presence['ok']=not any(key_presence['added_in_b3'].values()) and not any(key_presence['removed_in_b3'].values())
    ref_pair=[p for p in pairs if p['a'] in A and p['b'] in A]
    ref_unique=set(ref_pair[0]['diff_paths']) if ref_pair else set()
    hist_p=E/'legacy-historical-run-unique.json'
    if MODE=='d' and group=='N' and hist_p.exists():ref_unique|=set(json.loads(hist_p.read_text())['paths'])
    strict=[dict(a=p['a'],b=p['b'],extra_vs_reference_pair=[q for q in p['diff_paths'] if q not in ref_unique][:40],count=len([q for q in p['diff_paths'] if q not in ref_unique])) for p in pairs if not (p['a'] in A and p['b'] in A)]
    out=dict(group=group,legs={l:recs[l]['binary'] for l in names},pairs=[{k:v for k,v in p.items() if k!='diff_paths'}|dict(diff_paths_count=len(p['diff_paths']),diff_paths_sample=p['diff_paths'][:30]) for p in pairs],
             run_unique_paths=sorted(run_unique)[:200],run_unique_count=len(run_unique),unstable_fields=sorted(unstable_fields),cross=res,
             key_presence=key_presence,reference_pair_unique_count=len(ref_unique),strict_vs_reference_pair=strict)
    return out,recs
def evaluate():
    results={}
    for group,extra,legs in GROUPS:
        if not all((E/f'reading-{l}.json').exists() for l,_ in legs):results[group]=dict(complete=False);continue
        out,recs=compare(group,legs);verdict=[]
        if MODE=='d':
            for l,w in legs:
                r=recs[l]
                if w is not pre and (r['change_sidecar'] or r['change_keys'] or r['effective_lines'] or r['cvar_echo']!=['0']):verdict.append(f'{l}: evidence OFF not read back ({r["change_sidecar"]},{r["change_keys"][:3]},{r["effective_lines"]},{r["cvar_echo"]})')
                if r['ensures']:verdict.append(f'{l}: handled ensure')
            for x in out['cross']:
                if x['extra_paths_count']:verdict.append(f"{x['a']}~{x['b']}: {x['extra_paths_count']} cross-binary paths outside the run-unique set")
                if x['gated_fields_unequal']:verdict.append(f"{x['a']}~{x['b']}: stable 081-12 fields unequal {x['gated_fields_unequal']}")
            if not out['key_presence']['ok']:verdict.append(f"key presence: build 3 adds/removes paths {out['key_presence']}")
            if group=='N':
                for x in out['strict_vs_reference_pair']:
                    if x['count']:verdict.append(f"native {x['a']}~{x['b']}: {x['count']} paths differ that the pre-m55 pair does not vary")
                for p in out['pairs']:
                    bad=[f for f,v in p['fields_equal'].items() if not v]
                    if bad:verdict.append(f"native {p['a']}~{p['b']}: 081-12 field set not byte-identical {bad}")
        else:
            b3l=[l for l,w in legs if w is not pre]
            for l in b3l:
                r=recs[l]
                if not (r['summary_selected'].get('change_multi_view_families') or 0)>0:verdict.append(f'{l}: change_multi_view_families not > 0')
                if (r.get('m55_pairs') or {}).get('measured',1)!=0:verdict.append(f'{l}: measured m55 rows present')
                if not (r['summary_selected'].get('change_reason_unsupported_delivery') or 0)>0:verdict.append(f'{l}: no unsupported_delivery')
                if r.get('oracle_exit')!=0:verdict.append(f"{l}: oracle exit {r.get('oracle_exit')}")
            for l,_ in legs:
                if recs[l]['ensures']:verdict.append(f'{l}: handled ensure')
            for x in out['cross']:
                if x['gated_fields_unequal']:verdict.append(f"{x['a']}~{x['b']}: stable legacy fields unequal {x['gated_fields_unequal']}")
            if not out['key_presence']['ok']:verdict.append(f"key presence (change_* excepted): {out['key_presence']}")
            pw={l:recs[l]['sve_want'] for l,_ in legs}
            pre_pend=[x['pendingWantedAtEnd'] for l,w in legs if w is pre for x in pw[l]];b3_pend=[x['pendingWantedAtEnd'] for l,w in legs if w is not pre for x in pw[l]]
            pre_max=[x['maxPendingDepth'] for l,w in legs if w is pre for x in pw[l]];b3_max=[x['maxPendingDepth'] for l,w in legs if w is not pre for x in pw[l]]
            if not (pre_pend and b3_pend and set(b3_pend)<=set(pre_pend) and max(b3_max)<=max(pre_max)):verdict.append(f'pending arms: pre {pre_pend}/{pre_max} b3 {b3_pend}/{b3_max}')
            out['pending']=dict(pre_pending_at_end=pre_pend,b3_pending_at_end=b3_pend,pre_max_depth=pre_max,b3_max_depth=b3_max)
        out['verdict']='PASS' if not verdict else 'FAIL';out['problems']=verdict
        results[group]=out
    (E/f'part1-{MODE}-comparison.json').write_text(json.dumps(results,indent=2,default=str))
    return results
