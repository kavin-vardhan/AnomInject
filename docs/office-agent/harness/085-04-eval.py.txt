def judged_edges(events,thr=RELEASE):
    on=off=0;fails=[];nm=0;warm=0;conf=0
    for r in events:
        st=r.get('status')
        if st=='WARMUP':warm+=1
        if st=='CONFOUNDED-REFERENCE':conf+=1;continue
        if st not in ('OK','WARMUP'):continue
        if not r.get('measurable'):nm+=1;continue
        p=(r.get('per') or {}).get(thr) or {};ed=p.get('edges') or {}
        if isinstance(ed,list):ed=dict(start=ed[0].get('start'),end=ed[-1].get('end'),runs=ed) if ed else {}
        if ed.get('start') is not None:on+=1
        if ed.get('end') is not None:off+=1
        if p.get('verdict')=='FAIL':fails.append(dict(start_frame=r.get('start_frame'),edges=ed,reasons=p.get('fail_reasons')))
    return dict(onsets=on,offsets=off,fails=fails,not_measurable=nm,warmup=warm,confounded=conf)

def scramble_invisible(seed,ordinal,k=8):
    """Predicted visibility of a scramble pair on TC_UC1 (8x8 cells of the 4-colour pal(0) pattern): the number of cells
    whose source cell has a different colour. 0 changed cells = a predicted-invisible pair, excluded by name (DC plan 5 rule 3)."""
    a,b,ai=ORC.scramble_pair(seed,ordinal,k) if hasattr(ORC,'scramble_pair') else ORC.scramble_pair_py(seed,ordinal,k)
    col=lambda cx,cy:(cx+3*cy)%4
    changed=0
    for cy in range(k):
        for cx in range(k):
            j=cy*k+cx;i=(ai*((j-b)%(k*k)))%(k*k)
            if col(i%k,i//k)!=col(cx,cy):changed+=1
    return changed

def replay(draws,seed):
    """Independent replay of every Auto.Draw line: the KAT stream seeded like the capture, and per attempt the number of
    draws the logged counts imply (id over eligible, target over candidates, the hold, and one mode draw for m53 ids);
    the replayed seed_after must equal the logged one on every line."""
    if not draws:return dict(pass_=False,reason='no Auto.Draw lines')
    st=KAT.KatStream(int(seed));bad=[]
    for x in draws:
        if int(x['sb'])!=st.seed:bad.append(dict(attempt=x['attempt'],why='seed_before',want=st.seed,got=int(x['sb'])));st.seed=int(x['sb'])
        ne=_count(x['elig']);nc=_count(x['cand']);n=0
        if x['result']!='all_excluded' and ne>0:
            n=1
            if nc>0:n+=2+(1 if x['id'] in M53 else 0)
        for _ in range(n):st.mutate()
        if int(x['sa'])!=st.seed:
            bad.append(dict(attempt=x['attempt'],why='seed_after',want=int(x['sa']),got=st.seed,draws_modelled=n));st.seed=int(x['sa'])
    return dict(pass_=not bad,lines=len(draws),mismatches=bad[:6],note='per attempt: id over eligible, then target + hold + (m53) one mode draw when candidates > 0; the stream starts at the capture seed')

def census_join(prod,bench):
    """The product census counts equal the bench census per-actor lines aggregated on the same view (default scope)."""
    if not prod['blocks']:return dict(pass_=False,reason='no product census block')
    b0=prod['blocks'][0];fam={'uv_corruption':'uv','normal_corruption':'normal'};out={}
    for pid,v in b0['ids'].items():
        f=fam.get(pid,pid)
        rows=[a[f] for a in bench['per_actor'].values() if f in a]
        el=sum(1 for r in rows if r['final']=='APPLY')
        out[pid]=dict(product=v,bench_rows=len(rows),bench_eligible=el,match=(len(rows)==v['eligible']+v['refused'] and el==v['eligible']))
    return dict(per_id=out,pass_=all(x['match'] for x in out.values()) if out else False)

def cost_reading(logs):
    per={};rows=[]
    for k in ('COST_A1','COST_B1','COST_B2','COST_A2'):
        P=parse_log(logs.get(k,''));am=[x['apply_ms'] for x in P['applied'] if x.get('apply_ms') is not None]
        rv=[float(x['ms']) for x in P['revert']];gp=[float(x['gpu']) for x in P['cost']];gt=[float(x['gt']) for x in P['cost']]
        med=lambda v:sorted(v)[len(v)//2] if v else None
        per[k]=dict(fires=len(P['applied']),apply_ms_med=med(am),revert_ms_med=med(rv),gpu_ms_med=med(gp),gt_ms_med=med(gt),cost_lines=len(P['cost']))
    a=[per[k]['gpu_ms_med'] for k in ('COST_A1','COST_A2') if per[k]['gpu_ms_med'] is not None]
    b=[per[k]['gpu_ms_med'] for k in ('COST_B1','COST_B2') if per[k]['gpu_ms_med'] is not None]
    if len(a)==2 and len(b)==2:
        diff=sum(b)/2-sum(a)/2;spread=abs(a[0]-a[1])
        verdict=f'B-A {diff:+.3f} ms GPU per frame' if abs(diff)>spread else f'below the resolution of this instrument (|B-A| {abs(diff):.3f} <= A spread {spread:.3f})'
    else:verdict='NOT READ'
    return dict(per_leg=per,reading=verdict,rule='A,B,B,A after a declared discard; no threshold (G169, G186)')
def gid_gate(g_sess):
    out={}
    for fam in ('UC1','NN1'):
        idl,nul,wr=g_sess.get(f'GID_{fam}_ID'),g_sess.get(f'GID_{fam}_NULL'),g_sess.get(f'GID_{fam}_WRONG')
        an='uv_corruption' if fam=='UC1' else 'normal_corruption'
        if not (idl and nul and wr):out[fam]=dict(verdict='INCOMPLETE');continue
        try:
            pc=M53R.pair_compare(M53R.Sess(idl),M53R.Sess(nul),an);fc=M53R.fault_compare(M53R.Sess(wr),M53R.Sess(nul),an)
            ok=str(pc.get('verdict')).startswith('PASS') and fc.get('verdict')=='READS-WRONG'
            out[fam]=dict(identity_vs_null=pc,wrong_vs_null=fc,verdict='PASS' if ok else 'FAIL')
        except Exception as ex:out[fam]=dict(verdict='ERROR',error=f'{type(ex).__name__}: {ex}')
    return out
def bind_gate(g_sess):
    t,tn=g_sess.get('BIND_TILE'),g_sess.get('BIND_TILENULL')

        pv=pf.get('per_frame_verdict') or pf.get('verdict')
        return dict(per_frame=pf,far_strip=fs,verdict='PASS' if (str(pv).startswith('PASS') and str(fs.get('verdict')).startswith('PASS')) else 'FAIL')

def nd5_rate(legs):
    """ND-5: per mode, the share of MainWorld events judged NOT-MEASURABLE, at t50 and strict (the definition is declared
    in the README before any leg runs): an event is NOT-MEASURABLE when E1 finds no visible change inside its mask over its
    labelled frames at the threshold (its measurable flag is false), warm-up and confounded-reference events excluded."""
    per=collections.defaultdict(lambda:dict(events=0,not_measurable=0))
    for k,v in legs.items():
        if not isinstance(v,dict) or v.get('group')!='BM53MW' or v.get('role')=='NULL':continue
        mode=v.get('mode')
        for r in v.get('e1_events') or []:
            if r.get('type') not in M53 or r.get('status')!='OK':continue
            per[mode]['events']+=1
            if not r.get('measurable'):per[mode]['not_measurable']+=1
    return {m:dict(d,rate=round(d['not_measurable']/d['events'],3) if d['events'] else None) for m,d in per.items()}
DOCTOR_VARIANTS=('onset_late_1','onset_early_1','offset_early_1','offset_late_1','shift_plus_1','shift_minus_1')
def _doctor_list(Lst,v):
    Lst=sorted(Lst)
    if not Lst:return Lst
    return {'exact':Lst,'onset_late_1':Lst[1:],'onset_early_1':[Lst[0]-1]+Lst,'offset_early_1':Lst[:-1],'offset_late_1':Lst+[Lst[-1]+1],
            'shift_plus_1':[x+1 for x in Lst],'shift_minus_1':[x-1 for x in Lst]}[v]
def doctor_copy(sd,variant,root):
    """A doctored copy of a session: images hardlinked, labels.jsonl and annotation.json rewritten so every m53 event's
    labelled set is moved by the variant (the can-fail (b) shapes). The original is never written."""
    import shutil
    sd=Path(sd);dst=Path(root)/f'{sd.name}__{variant}';
    if dst.exists():shutil.rmtree(dst)
    dst.mkdir(parents=True)
    for p in sd.rglob('*'):
        q=dst/p.relative_to(sd)
        if p.is_dir():q.mkdir(parents=True,exist_ok=True);continue
        if p.name in ('labels.jsonl','annotation.json'):continue
        q.parent.mkdir(parents=True,exist_ok=True)
        try:os.link(p,q)
        except OSError:shutil.copy2(p,q)
    an=json.loads((sd/'annotation.json').read_text(encoding='utf-8-sig'));rows=E9.read_jsonl(sd/'labels.jsonl');by={r['session_index']:r for r in rows}
    for a in an.get('anomalies') or []:
        if a.get('anomaly_type') not in M53:continue
        inj=(a.get('injected_frames') or {}).get('frame_indices') or []
        new=[x for x in _doctor_list(inj,variant) if x in by]
        for key in ('injected_frames','affected_frames'):
            if isinstance(a.get(key),dict):a[key]=dict(a[key],frame_indices=new,start_frame=min(new) if new else None,end_frame=max(new) if new else None,frame_count=len(new))
        tmpl=None;sf=None
        for si in inj:
            for e in by[si].get('anomalies') or []:
                if isinstance(e,dict) and e.get('id')==a['anomaly_type']:tmpl=dict(e);sf=e.get('start_frame');break
            if tmpl:break
        for si,r in by.items():
            ents=[e for e in (r.get('anomalies') or []) if isinstance(e,dict) and e.get('id')==a['anomaly_type'] and e.get('start_frame')==sf]
            if si in new and not ents and tmpl:r.setdefault('anomalies',[]).append(dict(tmpl,labelled=True));r['anomaly_present']=True
            for e in ents:e['labelled']=si in new
    (dst/'annotation.json').write_text(json.dumps(an),encoding='utf-8')
    (dst/'labels.jsonl').write_text('\n'.join(json.dumps(by[k]) for k in sorted(by))+'\n',encoding='utf-8')
    return dst
def doctored_canfail(sd,log,root,variants=DOCTOR_VARIANTS[:4]):
    """can-fail (b) on a real gate leg: every doctored variant must FAIL E1 at t50 while the exact labels PASS."""
    out={}
    base=e1(sd,log,types=M53)['e1'].get('leg_verdict')
    out['exact']=base
    for v in variants:
        d=doctor_copy(sd,v,root)
        try:out[v]=E9.summarize_diff(E9.diff_leg(d,types=M53,log=log)['events']).get('leg_verdict')
        finally:
            import shutil;shutil.rmtree(d,ignore_errors=True)
    out['proven']=base in ('PASS','PASS-WITH-CENSORED') and all(out[v]=='FAIL' for v in variants)
    return out
def _rgb(p):
    return E9.np.asarray(E9.Image.open(p).convert('RGB'))
def _maps_from_null(null_sd,acts_json):
    rows=ORC.load_labels(str(null_sd));pick=[r for r in rows[-3:]]
    stack=E9.np.stack([_rgb(Path(null_sd)/r['image']).astype('int16') for r in pick]);med=E9.np.median(stack,axis=0).astype('uint8')
    return ORC.build_maps(med,str(acts_json) if acts_json else None)
def mode_leg(on_sd,null_sd,acts_json,log,mode,seed):
    """G-MODE for one production leg: the maps from the leg's null (median of its last 3 frames), the events from the leg's
    labelled frames, the APPLIED lines attached (mode, ordinal, the logged scramble pair), then the oracle's judge."""
    anomaly='uv_corruption' if mode in ('tile','scramble') else 'normal_corruption'
    need={'tile':'uv16','scramble':'uv64'}.get(mode,'n')
    try:mp=_maps_from_null(null_sd,acts_json).get(need) or {}
    except Exception as ex:mp={'error':f'{type(ex).__name__}: {ex}'}
    if mp.get('A') is None:
        return dict(verdict='NO-MAP',reasons=[f'P-ORC: the null frames yield no {need} oracle map; G-MODE cannot judge (reported, never PASS)'],map=mp.get('error'))
    js=ORC.judge_session(str(on_sd),log_text=log or '',null_session_dir=str(null_sd) if null_sd else None,
                         dc_actors=str(acts_json) if acts_json else None,seed=seed)
    leg=dict((js.get('legs') or {}).get(f'{anomaly}/{mode}') or dict(verdict='INSUFFICIENT',reasons=[f'no {anomaly}/{mode} events judged'],legs_seen=sorted((js.get('legs') or {}).keys())))
    leg['porc']=js.get('porc');leg['seed']=js.get('seed');leg['seed_source']=js.get('seed_source');leg['tile_n']=js.get('tile_n')
    if js.get('error'):leg.setdefault('reasons',[]).append(js['error'])
    return leg
def null_modes(null_uv_sd,null_n_sd,acts_json,seed):
    out={}
    for sd,key in ((null_uv_sd,'uv'),(null_n_sd,'n')):
        if not sd:out[key]=dict(verdict='NOT RUN');continue
        maps=_maps_from_null(sd,acts_json);anomaly='uv_corruption' if key=='uv' else 'normal_corruption'
        bad=[k for k in (('uv16','uv64') if key=='uv' else ('n',)) if (maps.get(k) or {}).get('A') is None]
        if bad:out[key]=dict(verdict='NO-MAP',reasons=[f'the null frames yield no oracle map for {bad} (P-ORC)']);continue
        rows=ORC.load_labels(str(sd))
        fr=[(r['session_index'],_rgb(Path(sd)/r['image'])) for r in rows if any(e.get('id')==anomaly and e.get('labelled',True) for e in r.get('anomalies') or [])]
        out[key]=ORC.judge_null_as_modes(fr[:24],maps,dict(tile_n=8,seed=seed,ordinals=list(range(0,6))))
    return out
